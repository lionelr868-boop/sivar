   191→// ---------- seed helpers ----------
   192→/** تدوير الصفوف — seed 0 يعني بدون تغيير */
   193→function rot<T>(rows: T[], seed: number): T[] {
   194→  if (!seed || rows.length === 0) return [...rows]
   195→  const r = ((seed % rows.length) + rows.length) % rows.length
   196→  return [...rows.slice(r), ...rows.slice(0, r)]
   197→}
   198→
   199→/** اهتزاز رقمي بسيط — seed 0 يعني بدون تغيير */
   200→function jit(v: number, seed: number, pct = 0.08): number {
   201→  if (!seed) return v
   202→  const f = 1 + (((seed % 7) - 3) * pct) / 3
   203→  return Math.max(0, Math.round(v * f))
   204→}

   206→const NUMERIC_COLUMN_TYPES = new Set<string>(['number', 'currency', 'progress'])
   207→
   208→/** بناء كيان مع تطبيق seed (تدوير + اهتزاز الأرقام) — نسخة مستقلة عن الأعمدة */
   209→function makeEntity(
   210→  columns: EntityColumn[],
   211→  baseRows: Array<Record<string, unknown>>,
   212→  seed: number
   213→): EntityData {
   214→  const cols = columns.map((c) => ({ ...c }))
   215→  const rows = rot(baseRows, seed).map((row) => {
   216→    const out: Record<string, unknown> = {}
   217→    for (const col of cols) {
   218→      const v = row[col.key]
   219→      if (NUMERIC_COLUMN_TYPES.has(col.type) && typeof v === 'number') {
   220→        out[col.key] = jit(v, seed)
   221→      } else {
   222→        out[col.key] = v
   223→      }
   224→    }
   225→    return out
   226→  })
   227→  return { columns: cols, rows }
   228→}
