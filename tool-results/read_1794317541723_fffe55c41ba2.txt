    56→function strArg(args: Record<string, unknown>, key: string, fallback = ''): string {
    57→  const v = args[key]
    58→  return typeof v === 'string' && v.trim() ? v.trim() : fallback
    59→}
    60→function numArg(args: Record<string, unknown>, key: string): number | undefined {
    61→  const v = args[key]
    62→  if (typeof v === 'number' && Number.isFinite(v)) return v
    63→  if (typeof v === 'string' && v.trim() && Number.isFinite(Number(v))) return Number(v)
    64→  return undefined
    65→}
    66→function boolArg(args: Record<string, unknown>, key: string): boolean | undefined {
    67→  const v = args[key]
    68→  if (typeof v === 'boolean') return v
    69→  if (v === 'true') return true
    70→  if (v === 'false') return false
    71→  return undefined
    72→}

    74→/** إيجاد صفحة بالمُعرّف أو بالاسم */
    75→function findPageRef(schema: ProjectSchema, ref: unknown): ProjectPage | undefined {
    76→  if (typeof ref !== 'string' || !ref.trim()) return undefined
    77→  const r = ref.trim()
    78→  return schema.pages.find((p) => p.id === r) ?? schema.pages.find((p) => p.name === r || p.slug === r)
    79→}

    81→/** إيجاد مكوّن داخل صفحة (بالمُعرّف) */
    82→function findBlock(page: ProjectPage, blockId: unknown): Block | undefined {
    83→  if (typeof blockId !== 'string' || !blockId.trim()) return undefined
    84→  return page.blocks.find((b) => b.id === blockId)

   147→/**
   148→ * يُدرج عنصراً لصفحة في شريط لوحة التحكم (تعديل مباشر على المخطط المستنسخ) — بدقة:
   149→ * - لا تكرار: صفحة لها عنصر بالفعل لا تُضاف مرتين.
   150→ * - لا خلط: العنوان = اسم الصفحة، والأيقونة مطابقة لنوع الصفحة وكيانها.
   151→ * - ترتيب مستقر: صفحات «لوحة تحكم» جديدة تُدرج بعد آخر عنصر لوحة (يبقى الرئيسي أولاً)،
   152→ *   وبقية أنواع اللوحة تُلحق في نهاية القائمة دون المساس بالعناصر القائمة.
   153→ */
   154→function insertDashboardNavItem(schema: ProjectSchema, page: ProjectPage): void {
   155→  if (!isDashboardAreaPage(page)) return
   156→  if (schema.dashboardNav.items.some((it) => it.pageId === page.id)) return
   157→  const item: DashboardNavItem = { label: page.name, pageId: page.id, icon: dashboardNavIconFor(page) }
   158→  const items = schema.dashboardNav.items
   159→  if (page.type === 'dashboard') {
   160→    let at = 0
   161→    for (let i = items.length - 1; i >= 0; i--) {
   162→      const target = schema.pages.find((p) => p.id === items[i].pageId)
   163→      if (target?.type === 'dashboard') {
   164→        at = i + 1
   165→        break
   166→      }
   167→    }
   168→    items.splice(at, 0, item)
   169→  } else {
   170→    items.push(item)
   171→  }
   172→}

   174→/** يضمن ظهور صفحة قائمة في الشريط الجانبي للوحة التحكم (إن كانت من أنواع منطقة اللوحة) — دالة نقية */
   175→export function ensurePageInDashboardNav(schema: ProjectSchema, pageId: string): ProjectSchema {
   176→  const page = schema.pages.find((p) => p.id === pageId)
   177→  if (!page || !isDashboardAreaPage(page)) return schema
   178→  if (schema.dashboardNav.items.some((it) => it.pageId === pageId)) return schema
   179→  const next = clone(schema)
   180→  const target = next.pages.find((p) => p.id === pageId) as ProjectPage
   181→  insertDashboardNavItem(next, target)
   182→  return next
   183→}

   185→/**
   186→ * مزامنة عنصر صفحة في الشريط الجانبي بعد تعديلها (إعادة تسمية/تغيير كيان) —
   187→ * يُبقي العنصر في موضعه ويحدّث عنوانه وأيقونته فقط، بلا خلط مع بقية العناصر.
   188→ */
   189→function syncPageInDashboardNav(schema: ProjectSchema, page: ProjectPage): void {
   190→  const item = schema.dashboardNav.items.find((it) => it.pageId === page.id)
   191→  if (!item) return
   192→  if (item.label !== page.name) item.label = page.name
   193→  const icon = dashboardNavIconFor(page)
   194→  if (item.icon !== icon) item.icon = icon
   195→}

   197→/**
   198→ * (الجولة 13) مزامنة قوائم التنقل بعد تعديل صفحة من المحرر (إعادة تسمية) —
   199→ * يحدّث عنوان عنصر الشريط الجانبي وأيقونته وعنوان عنصر قائمة الموقع لنفس الصفحة فقط،
   200→ * دون المساس بترتيب العناصر أو بقية القوائم — لا خلط بين العناصر.
   201→ */
   202→export function syncPageNavigation(schema: ProjectSchema, pageId: string): ProjectSchema {
   203→  const page = schema.pages.find((p) => p.id === pageId)
   204→  if (!page) return schema
   205→  const next = clone(schema)
   206→  const target = next.pages.find((p) => p.id === pageId) as ProjectPage
   207→  syncPageInDashboardNav(next, target)
   208→  for (const navItem of next.navigation.items) {
   209→    if (navItem.pageId === target.id && navItem.label !== target.name) navItem.label = target.name
   210→  }
   211→  return next
   212→}

   214→/** كتل البداية لكل نوع صفحة (تُستعمل في create_page + factory) */
   215→export function starterBlocksFor(
   216→  schema: ProjectSchema,
   217→  type: PageType,
   218→  opts?: { entity?: string; authVariant?: AuthVariant; landingVariant?: string }
   219→): Block[] {
   220→  const d = (t: BlockType): Record<string, unknown> => clone(BLOCK_REGISTRY[t].defaults())
   221→  const mk = (t: BlockType, props: Record<string, unknown>): Block => ({ id: uidFn('b_'), type: t, props })
   222→
   223→  switch (type) {
   224→    case 'landing':
   225→      return buildLandingBlocks(schema, opts?.landingVariant)
   226→    case 'auth': {
   227→      const variant: AuthVariant = opts?.authVariant ?? 'login'
   228→      const props = d('auth-card')
   229→      props.variant = variant
   230→      if (variant === 'signup') {
   231→        props.title = 'إنشاء حساب'
   232→        props.subtitle = `انضم إلى ${schema.navigation.brand} وابدأ رحلتك اليوم`
   233→        props.buttonText = 'إنشاء الحساب'
   234→      } else if (variant === 'forgot') {
   235→        props.title = 'استعادة كلمة المرور'
   236→        props.subtitle = 'أدخل بريدك الإلكتروني وسنرسل لك رابط الاستعادة'
   237→        props.buttonText = 'إرسال الرابط'
   238→      }
   239→      return [mk('auth-card', props)]
   240→    }
   241→    case 'dashboard': {
   242→      const entity = opts?.entity ?? defaultEntityFor(schema.metadata.type)
   243→      const preset = defaultStatsPreset(schema.metadata.type)
   244→      const stats = (schema.mockData.stats[preset] ?? schema.mockData.stats.overview ?? []).map((s) => ({
   245→        ...s,
   246→      }))
   247→      return [
   248→        mk('dashboard-header', {
   249→          title: 'لوحة التحكم',
   250→          subtitle: 'نظرة عامة على أداء مشروعك',
   251→          actions: 'إضافة عنصر,تصدير التقرير',
   252→        }),
   253→        mk('stats-row', { items: stats.length ? stats : d('stats-row').items }),
   254→        mk('chart', { title: 'الإيرادات الشهرية', chartType: 'bar', dataKey: 'revenueMonthly' }),
   255→        mk('data-table', { entity, title: ENTITY_LABELS[entity] ?? 'العناصر', pageSize: 5 }),
   256→      ]
   257→    }
   258→    case 'pricing':
   259→      return [mk('pricing', d('pricing'))]
   260→    case 'table': {
   261→      const entity = opts?.entity ?? defaultEntityFor(schema.metadata.type)
   262→      const label = ENTITY_LABELS[entity] ?? 'العناصر'
   263→      return [
   264→        mk('dashboard-header', {
   265→          title: label,
   266→          subtitle: `إدارة ${label} ومتابعتها لحظة بلحظة`,
   267→          actions: `إضافة,تصدير`,
   268→        }),
   269→        mk('data-table', { entity, title: '', pageSize: 5 }),
   270→      ]
   271→    }
   272→    case 'form':
   273→      return [
   274→        mk('dashboard-header', { title: 'نموذج جديد', subtitle: 'املأ النموذج التالي', actions: '' }),
   275→        mk('form-fields', d('form-fields')),
   276→      ]
   277→    case 'settings':
   278→      return [mk('settings-form', d('settings-form'))]
   279→    case 'profile':
   280→      return [mk('profile-card', d('profile-card'))]
   281→    case 'faq':
   282→      return [mk('faq', d('faq'))]
   283→    default:
   284→      return [mk('empty-state', d('empty-state'))]
   285→  }
   286→}

   292→export function addPageToSchema(
   293→  schema: ProjectSchema,
   294→  opts: { name: string; type: PageType; entity?: string; authVariant?: AuthVariant; landingVariant?: string }
   295→): { schema: ProjectSchema; pageId: string } {
   296→  const next = clone(schema)
   297→  const type: PageType = isPageType(opts.type) ? opts.type : 'custom'
   298→  const name = opts.name?.trim() || PAGE_TYPE_LABELS[type]
   299→  const page: ProjectPage = {
   300→    id: uidFn('page_'),
   301→    name,
   302→    slug: slugifyFn(name),
   303→    type,
   304→    authVariant: opts.authVariant,
   305→    entity: opts.entity,
   306→    title: name,
   307→    subtitle: '',
   308→    blocks: starterBlocksFor(next, type, {
   309→      entity: opts.entity,
   310→      authVariant: opts.authVariant,
   311→      landingVariant: opts.landingVariant,
   312→    }),
   313→  }
   314→  next.pages.push(page)
   315→  // (الجولة 13) صفحات منطقة اللوحة تُدرج فوراً في الشريط الجانبي — بعنوان مطابق وأيقونة دقيقة
   316→  insertDashboardNavItem(next, page)
   317→  return { schema: next, pageId: page.id }
   318→}

   320→/** صفحة HTML مخصصة — كود من المستخدم داخل إطار آمن، مع هيكل الموقع أو لوحة التحكم حوله */

   361→/** تعقيم خصائص يرسلها الوكيل/المحرر لمكوّن جديد: قيم JSON بسيطة فقط (بلا دوال ولا ملفات data URL ضخمة) */
   362→function sanitizeInitialProps(raw: unknown): Record<string, unknown> | undefined {
   363→  if (!raw || typeof raw !== 'object' || Array.isArray(raw)) return undefined
   364→  const out: Record<string, unknown> = {}
   365→  const walk = (value: unknown): unknown => {
   366→    if (value === null || typeof value === 'string' || typeof value === 'number' || typeof value === 'boolean') {
   367→      return typeof value === 'string' && value.length > 3000 ? value.slice(0, 3000) : value
   368→    }
   369→    if (Array.isArray(value)) return value.slice(0, 24).map(walk)
   370→    if (value && typeof value === 'object') {
   371→      const rec: Record<string, unknown> = {}
   372→      for (const [k, v] of Object.entries(value).slice(0, 30)) rec[k] = walk(v)
   373→      return rec
   374→    }
   375→    return undefined
   376→  }
   377→  for (const [key, value] of Object.entries(raw as Record<string, unknown>).slice(0, 30)) {
   378→    const clean = walk(value)
   379→    if (clean !== undefined) out[key] = clean
   380→  }
   381→  return Object.keys(out).length > 0 ? out : undefined
   382→}

   384→export function addBlockToPage(
   385→  schema: ProjectSchema,
   386→  pageId: string,
   387→  blockType: BlockType,
   388→  position?: number,
   389→  initialProps?: Record<string, unknown>
   390→): { schema: ProjectSchema; blockId: string } {
   391→  if (!isBlockType(blockType)) {
   392→    throw new Error('نوع المكوّن غير معروف')
   393→  }
   394→  const next = clone(schema)
   395→  const page = next.pages.find((p) => p.id === pageId)
   396→  if (!page) {
   397→    throw new Error('الصفحة غير موجودة')
   398→  }
   399→  const def = getBlockDefinition(blockType)
   400→  // (الجولة 14) خصائص الوكيل/المحرر تُدمج فوق الافتراضيات — لا تُستبدل
   401→  // كي تكتمل النواقص من القالب ولا ينتج مكوّن مبتور
   402→  const defaults = def ? clone(def.defaults()) : {}
   403→  const props = initialProps ? mergeBlockProps(defaults, initialProps) : defaults
   404→  const block: Block = { id: uidFn('b_'), type: blockType, props }
   405→  const pos = typeof position === 'number' ? Math.max(0, Math.min(position, page.blocks.length)) : page.blocks.length
   406→  page.blocks.splice(pos, 0, block)
   407→  return { schema: next, blockId: block.id }
   408→}

   410→export function updateBlockPropsInSchema(
   411→  schema: ProjectSchema,
   412→  pageId: string,
   413→  blockId: string,
   414→  patch: Record<string, unknown>
   415→): ProjectSchema {
   416→  const next = clone(schema)
   417→  const page = next.pages.find((p) => p.id === pageId)
   418→  const block = page ? findBlock(page, blockId) : undefined
   419→  if (!page || !block) return schema
   420→  // (الجولة 14) دمج جراحي غير إتلافي: المصفوفات تُدمج بالفهرس فلا تضيع
   421→  // تعديلات المستخدم السابقة (صور/شارات/ميزات) عند كل تحديث
   422→  block.props = mergeBlockProps(block.props ?? {}, patch)
   423→  return next
   424→}

   436→export function moveBlockInSchema(
   437→  schema: ProjectSchema,
   438→  pageId: string,
   439→  blockId: string,
   440→  position: number
   441→): ProjectSchema {
   442→  const next = clone(schema)
   443→  const page = next.pages.find((p) => p.id === pageId)
   444→  if (!page) return schema
   445→  const idx = page.blocks.findIndex((b) => b.id === blockId)
   446→  if (idx === -1 || page.blocks.length < 2) return schema
   447→  const pos = Math.max(0, Math.min(Math.round(position), page.blocks.length - 1))
   448→  const [moved] = page.blocks.splice(idx, 1)
   449→  page.blocks.splice(pos, 0, moved)
   450→  return next

   453→/** تحويل أسماء الألوان العربية إلى HEX + التحقق */
   454→function normalizeThemeValue(key: string, value: unknown): { value?: unknown; error?: string } {
   455→  switch (key) {
   456→    case 'radius':
   457→      if (typeof value === 'number' && Number.isFinite(value)) return { value: Math.max(0, Math.min(40, value)) }
   458→      if (typeof value === 'string' && Number.isFinite(Number(value))) return { value: Math.max(0, Math.min(40, Number(value))) }
   459→      return { error: 'قيمة نصف القطر غير صالحة' }
   460→    case 'shadow': {
   461→      const v = String(value)
   462→      if (['none', 'sm', 'md', 'lg'].includes(v)) return { value: v }
   463→      return { error: 'قيمة الظل غير صالحة (none/sm/md/lg)' }
   464→    }
   465→    case 'fontFamily': {
   466→      const v = String(value)
   467→      if (['IBM Plex Sans Arabic', 'Tajawal', 'Cairo'].includes(v)) return { value: v }
   468→      return { error: 'الخط غير مدعوم (IBM Plex Sans Arabic / Tajawal / Cairo)' }
   469→    }
   470→    case 'mode': {
   471→      const v = String(value)
   472→      if (v === 'light' || v === 'dark') return { value: v }
   473→      return { error: 'نمط العرض غير صالح' }
   474→    }
   475→    default: {
   476→      // حقول الألوان
   477→      const v = String(value).trim()
   478→      if (/^#([0-9a-f]{3}|[0-9a-f]{6}|[0-9a-f]{8})$/i.test(v)) return { value: v.toLowerCase() }
   479→      const mapped = COLOR_NAME_TO_HEX[v]
   480→      if (mapped) return { value: mapped }
   481→      return { error: `لون غير صالح: «${v}»` }
   482→    }
   483→  }
   484→}

   502→export function updateThemeInSchema(schema: ProjectSchema, patch: Partial<ProjectTheme>): ProjectSchema {
   503→  const next = clone(schema)
   504→  for (const [key, raw] of Object.entries(patch ?? {})) {
   505→    if (!THEME_KEYS.includes(key)) continue
   506→    const { value, error } = normalizeThemeValue(key, raw)
   507→    if (error || value === undefined) continue
   508→    ;(next.theme as unknown as Record<string, unknown>)[key] = value
   509→  }
   510→  return next
   511→}

   530→export function updateDashboardNavInSchema(schema: ProjectSchema, items: DashboardNavItem[]): ProjectSchema {
   531→  const next = clone(schema)
   532→  if (!Array.isArray(items)) return schema
   533→  const seen = new Set<string>()
   534→  next.dashboardNav = {
   535→    items: items
   536→      .filter((it) => it && typeof it.label === 'string' && typeof it.pageId === 'string')
   537→      // (الجولة 13) لا تكرار لنفس الصفحة — أول ظهور يفوز فلا يحدث خلط بين العناصر
   538→      .filter((it) => {
   539→        if (seen.has(it.pageId)) return false
   540→        seen.add(it.pageId)
   541→        return true
   542→      })
   543→      .map((it) => {
   544→        // (الجولة 13) أيقونة دقيقة: صالحة كما أعطيت، وإلا أيقونة العنصر القائم لنفس الصفحة، وإلا أيقونة نوع الصفحة
   545→        const page = next.pages.find((p) => p.id === it.pageId)
   546→        const existing = schema.dashboardNav.items.find((nav) => nav.pageId === it.pageId)
   547→        const icon =
   548→          typeof it.icon === 'string' && it.icon.trim()
   549→            ? it.icon.trim()
   550→            : existing?.icon ?? (page ? dashboardNavIconFor(page) : 'file-text')
   551→        return { label: it.label.trim(), pageId: it.pageId, icon }
   552→      }),
   553→  }
   554→  return next
   555→}

   595→export function applyActions(schema: ProjectSchema, actions: AgentAction[]): ActionResult {
   596→  let current = clone(schema)
   597→  const applied: string[] = []
   598→  const errors: string[] = []
   599→
   600→  const pushError = (tool: string, msg: string) => errors.push(`${msg}`)
   601→
   602→  for (const action of actions ?? []) {
   603→    const args = (action?.args ?? {}) as Record<string, unknown>
   604→    try {
   605→      switch (action.tool as AgentToolName) {
   606→        // ---------- pages ----------
   607→        case 'create_page': {
   608→          const type = isPageType(args.type) ? args.type : null
   609→          if (!type) {
   610→            pushError(action.tool, 'نوع الصفحة غير صالح')
   611→            break
   612→          }
   613→          const name = strArg(args, 'name', PAGE_TYPE_LABELS[type])
   614→          const entity = strArg(args, 'entity') || undefined
   615→          const authVariantRaw = strArg(args, 'authVariant')
   616→          const authVariant: AuthVariant | undefined =
   617→            authVariantRaw === 'login' || authVariantRaw === 'signup' || authVariantRaw === 'forgot'
   618→              ? authVariantRaw
   619→              : undefined
   620→          const landingVariantRaw = strArg(args, 'landingVariant')
   621→          const landingVariant = LANDING_VARIANTS.some((v) => v.id === landingVariantRaw) ? landingVariantRaw : undefined
   622→          const { schema: next, pageId } = addPageToSchema(current, { name, type, entity, authVariant, landingVariant })
   623→          current = next
   624→          // (الجولة 13) صفحات منطقة اللوحة تُدرج تلقائياً في الشريط الجانبي — نذكر ذلك صراحة
   625→          const created = current.pages.find((p) => p.id === pageId)
   626→          applied.push(
   627→            landingVariant && type === 'landing'
   628→              ? `تمت إضافة صفحة هبوط «${name}» بنمط «${getLandingVariant(landingVariant).name}»`
   629→              : created && isDashboardAreaPage(created)
   630→                ? `تمت إضافة صفحة «${name}» وإدراجها في الشريط الجانبي للوحة التحكم`
   631→                : `تمت إضافة صفحة «${name}»`
   632→          )
   633→          void pageId
   634→          break
   635→        }
   636→
   637→        case 'delete_page': {
   638→          if (current.pages.length <= 1) {
   639→            pushError(action.tool, 'لا يمكن حذف الصفحة الوحيدة في المشروع')
   640→            break
   641→          }
   642→          const page = findPageRef(current, args.pageId ?? args.name ?? args.page)
   643→          if (!page) {
   644→            pushError(action.tool, 'الصفحة غير موجودة')
   645→            break
   646→          }
   647→          current = deletePageFromSchema(current, page.id)
   648→          applied.push(`تم حذف صفحة «${page.name}»`)
   649→          break
   650→        }
   651→
   652→        case 'update_page': {
   653→          const page = findPageRef(current, args.pageId ?? args.page ?? args.name)
   654→          if (!page) {
   655→            pushError(action.tool, 'الصفحة غير موجودة')
   656→            break
   657→          }
   658→          const entity = strArg(args, 'entity')
   659→          if (entity && !current.mockData.entities[entity]) {
   660→            pushError(action.tool, `الكيان «${entity}» غير معروف في البيانات التجريبية`)
   661→            break
   662→          }
   663→          const next = clone(current)
   664→          const target = next.pages.find((p) => p.id === page.id) as ProjectPage
   665→          const name = strArg(args, 'name')
   666→          if (name) {
   667→            target.name = name
   668→            target.slug = slugifyFn(name)
   669→            target.title = name
   670→          }
   671→          const title = strArg(args, 'title')
   672→          if (title) target.title = title
   673→          if (typeof args.subtitle === 'string') target.subtitle = args.subtitle
   674→          if (entity) target.entity = entity
   675→          // (الجولة 13) مزامنة دقيقة: عنوان الشريط الجانبي وقائمة الموقع يتبعان اسم الصفحة،
   676→          // وأيقونة الشريط تتبع نوع الصفحة/كيانها — بلا خلط بين العناصر القائمة
   677→          syncPageInDashboardNav(next, target)
   678→          for (const navItem of next.navigation.items) {
   679→            if (navItem.pageId === target.id && navItem.label !== target.name) navItem.label = target.name
   680→          }
   681→          current = next
   682→          applied.push(`تم تحديث صفحة «${target.name}»`)
   683→          break
   684→        }
   685→
   686→        // ---------- blocks ----------
   687→        case 'add_block': {
   688→          const page = findPageRef(current, args.pageId ?? args.page)
   689→          const blockType = args.blockType ?? args.type
   690→          if (!isBlockType(blockType)) {
   691→            pushError(action.tool, 'نوع المكوّن غير معروف')
   692→            break
   693→          }
   694→          if (!page) {
   695→            pushError(action.tool, 'الصفحة غير موجودة')
   696→            break
   697→          }
   698→          const position = numArg(args, 'position')
   699→          // (الجولة 14) خصائص الوكيل اختيارية وتُدمج فوق الافتراضيات
   700→          const initialProps = sanitizeInitialProps(args.props)
   701→          const { schema: next, blockId } = addBlockToPage(current, page.id, blockType, position, initialProps)
   702→          current = next
   703→          applied.push(`تمت إضافة مكوّن «${blockLabel(blockType)}» إلى صفحة «${page.name}»`)
   704→          void blockId
   705→          break
   706→        }
   707→
   708→        case 'update_block':
   709→        case 'update_content': {
   710→          const page = findPageRef(current, args.pageId ?? args.page)
   711→          if (!page) {
   712→            pushError(action.tool, 'الصفحة غير موجودة')
   713→            break
   714→          }
   715→          const block = findBlock(page, args.blockId ?? args.id)
   716→          if (!block) {
   717→            pushError(action.tool, 'المكوّن غير موجود')
   718→            break
   719→          }
   720→          const patch = (args.patch ?? args.props) as Record<string, unknown> | undefined
   721→          if (!patch || typeof patch !== 'object' || Array.isArray(patch)) {
   722→            pushError(action.tool, 'خصائص التحديث غير صالحة')
   723→            break
   724→          }
   725→          current = updateBlockPropsInSchema(current, page.id, block.id, patch)
   726→          const prefix = action.tool === 'update_content' ? 'محتوى' : 'مكوّن'
   727→          applied.push(`تم تحديث ${prefix} «${blockLabel(block.type)}»`)
   728→          break
   729→        }
   730→
   731→        case 'delete_block': {
   732→          const page = findPageRef(current, args.pageId ?? args.page)
   733→          if (!page) {
   734→            pushError(action.tool, 'الصفحة غير موجودة')
   735→            break
   736→          }
   737→          const block = findBlock(page, args.blockId ?? args.id)
   738→          if (!block) {
   739→            pushError(action.tool, 'المكوّن غير موجود')
   740→            break
   741→          }
   742→          current = deleteBlockFromSchema(current, page.id, block.id)
   743→          applied.push(`تم حذف مكوّن «${blockLabel(block.type)}»`)
   744→          break
   745→        }
   746→
   747→        case 'move_block': {
   748→          const page = findPageRef(current, args.pageId ?? args.page)
   749→          if (!page) {
   750→            pushError(action.tool, 'الصفحة غير موجودة')
   751→            break
   752→          }
   753→          const block = findBlock(page, args.blockId ?? args.id)
   754→          if (!block) {
   755→            pushError(action.tool, 'المكوّن غير موجود')
   756→            break
   757→          }
   758→          const idx = page.blocks.findIndex((b) => b.id === block.id)
   759→          let position = numArg(args, 'position')
   760→          const direction = strArg(args, 'direction')
   761→          if (position === undefined && (direction === 'up' || direction === 'down')) {
   762→            position = direction === 'up' ? idx - 1 : idx + 1
   763→          }
   764→          if (position === undefined || position < 0 || position >= page.blocks.length) {
   765→            pushError(action.tool, 'الموضع الجديد غير صالح')
   766→            break
   767→          }
   768→          current = moveBlockInSchema(current, page.id, block.id, position)
   769→          applied.push(`تم نقل مكوّن «${blockLabel(block.type)}»`)
   770→          break
   771→        }
   772→
   773→        // ---------- landing variant ----------
   774→        case 'change_landing_variant': {
   775→          const variantRaw = strArg(args, 'variant') || strArg(args, 'landingVariant') || strArg(args, 'variantId')
   776→          const variantMeta = LANDING_VARIANTS.find((v) => v.id === variantRaw)
   777→          if (!variantMeta) {
   778→            pushError(action.tool, 'نمط صفحة الهبوط غير معروف')
   779→            break
   780→          }
   781→          const ref = args.pageId ?? args.page ?? args.name
   782→          const page = ref ? findPageRef(current, ref) : current.pages.find((p) => p.type === 'landing')
   783→          if (!page || page.type !== 'landing') {
   784→            pushError(action.tool, 'لم أجد صفحة هبوط لتغيير نمطها')
   785→            break
   786→          }
   787→          current = changeLandingVariantInSchema(current, page.id, variantMeta.id)
   788→          applied.push(`تم تحويل صفحة «${page.name}» إلى نمط «${variantMeta.name}»`)
   789→          break
   790→        }
   791→
   792→        // ---------- theme / navigation ----------
   793→        case 'update_theme': {
   794→          const patch = (args.patch ?? args) as Record<string, unknown>
   795→          const themePatch: Record<string, unknown> = {}
   796→          let hadValid = false
   797→          for (const key of THEME_KEYS) {
   798→            if (patch[key] !== undefined) {
   799→              const { value, error } = normalizeThemeValue(key, patch[key])
   800→              if (error) {
   801→                pushError(action.tool, error)
   802→              } else if (value !== undefined) {
   803→                themePatch[key] = value
   804→                hadValid = true
   805→              }
   806→            }
   807→          }
   808→          if (!hadValid) {
   809→            if (!errors.length) pushError(action.tool, 'لا توجد قيم صالحة لتحديث الهوية')
   810→            break
   811→          }
   812→          current = updateThemeInSchema(current, themePatch as Partial<ProjectTheme>)
   813→          applied.push('تم تحديث هوية الألوان')
   814→          break
   815→        }
   816→
   817→        case 'update_navigation': {
   818→          const pageIds = new Set(current.pages.map((p) => p.id))
   819→          const items = args.items as Array<{ label?: string; pageId?: string }> | undefined
   820→          if (Array.isArray(items)) {
   821→            const bad = items.find((it) => !it || typeof it.pageId !== 'string' || !pageIds.has(it.pageId))
   822→            if (bad) {
   823→              pushError(action.tool, 'عنصر تنقل يشير إلى صفحة غير موجودة')
   824→              break
   825→            }
   826→          }
   827→          const cta = args.cta as { label?: string; pageId?: string } | undefined
   828→          if (cta && (typeof cta.pageId !== 'string' || !pageIds.has(cta.pageId))) {
   829→            pushError(action.tool, 'زر التنقل يشير إلى صفحة غير موجودة')
   830→            break
   831→          }
   832→          current = updateNavigationInSchema(current, {
   833→            brand: typeof args.brand === 'string' ? args.brand : undefined,
   834→            footerNote: typeof args.footerNote === 'string' ? args.footerNote : undefined,
   835→            items: Array.isArray(items) ? (items as ProjectNavigation['items']) : undefined,
   836→            cta: (cta as ProjectNavigation['cta']) ?? undefined,
   837→          })
   838→          applied.push('تم تحديث قائمة التنقل')
   839→          break
   840→        }
   841→
   842→        case 'update_dashboard_nav': {
   843→          const pageIds = new Set(current.pages.map((p) => p.id))
   844→          const items = args.items as Array<{ label?: string; pageId?: string; icon?: string }> | undefined
   845→          if (!Array.isArray(items)) {
   846→            pushError(action.tool, 'قائمة عناصر لوحة التحكم غير صالحة')
   847→            break
   848→          }
   849→          const bad = items.find((it) => !it || typeof it.pageId !== 'string' || !pageIds.has(it.pageId))
   850→          if (bad) {
   851→            pushError(action.tool, 'عنصر لوحة التحكم يشير إلى صفحة غير موجودة')
   852→            break
   853→          }
   854→          current = updateDashboardNavInSchema(current, items as DashboardNavItem[])
   855→          applied.push('تم تحديث قائمة لوحة التحكم')
   856→          break
   857→        }
   858→
   859→        // ---------- project ----------
   860→        case 'rename_project': {
   861→          const name = strArg(args, 'name')
   862→          if (!name) {
   863→            pushError(action.tool, 'اسم المشروع غير صالح')
   864→            break
   865→          }
   866→          current = renameProjectInSchema(current, name)
   867→          applied.push(`تمت إعادة تسمية المشروع إلى «${name}»`)
   868→          break
   869→        }
   870→
   871→        case 'generate_mock_data': {
   872→          const entity = strArg(args, 'entity')
   873→          const seed = Math.max(1, Math.floor(Math.random() * 97))
   874→          if (entity) {
   875→            if (!ENTITY_LABELS[entity]) {
   876→              pushError(action.tool, `الكيان «${entity}» غير معروف`)
   877→              break
   878→            }
   879→            if (!refreshEntityIn(current, entity, seed)) {
   880→              pushError(action.tool, 'تعذّر تحديث بيانات هذا الكيان')
   881→              break
   882→            }
   883→            applied.push(`تم تحديث بيانات «${ENTITY_LABELS[entity]}»`)
   884→          } else {
   885→            let count = 0
   886→            for (const key of Object.keys(current.mockData.entities)) {
   887→              if (refreshEntityIn(current, key, seed + count)) count++
   888→            }
   889→            applied.push(`تم تحديث البيانات التجريبية (${count} كياناً)`)
   890→          }
   891→          break
   892→        }
   893→
   894→        default:
   895→          pushError(String(action?.tool ?? '?'), 'إجراء غير معروف')
   896→          break
   897→      }
   898→    } catch (e) {
   899→      pushError(action?.tool ?? '?', (e as Error).message || 'حدث خطأ أثناء تنفيذ الإجراء')
   900→    }
   901→  }
   902→
   903→  return { schema: current, applied, errors }
   904→}

   919→ * وتُنقّى من نسخ الإصدارات التاريخية (المشروع الحي يحتفظ بها
   920→ * حتى التحميل كنسخة عمل للمعاينة والتصدير).
   921→ * نقية ومتوافقة رجعياً: مخطط بلا صور يُعاد كما هو (نفس المرجع).
   922→ */
   923→export function stripTransientImages(schema: ProjectSchema): ProjectSchema {
   924→  let changed = false
   925→
   926→  // ---- الكتل: خصائص الصورة على مستوى الكتلة + داخل عناصر القوائم ----
   927→  const pages = (schema.pages ?? []).map((page) => ({
   928→    ...page,
   929→    blocks: (page.blocks ?? []).map((block) => {
   930→      if (!block) return block
   931→      const props = block.props ?? {}
   932→      let nextProps: Record<string, unknown> | null = null
   933→      if (props.bgImage !== undefined || props.bgImageName !== undefined) {
   934→        const { bgImage: _bi, bgImageName: _bn, ...rest } = props
   935→        nextProps = rest
   936→        changed = true
   937→      }
   938→      // عناصر القوائم داخل أي خاصية مصفوفة (عناصر المعرض/الفريق وغيرها)
   939→      for (const [key, value] of Object.entries(nextProps ?? props)) {
   940→        if (!Array.isArray(value)) continue
   941→        let listChanged = false
   942→        const nextList = value.map((item) => {
   943→          if (!item || typeof item !== 'object' || Array.isArray(item)) return item
   944→          const rec = item as Record<string, unknown>
   945→          if (rec.image === undefined && rec.imageName === undefined) return item
   946→          listChanged = true
   947→          const { image: _im, imageName: _inm, ...restItem } = rec
   948→          return restItem
   949→        })
   950→        if (!listChanged) continue
   951→        changed = true
   952→        if (!nextProps) nextProps = { ...props }
   953→        nextProps[key] = nextList
   954→      }
   955→      return nextProps ? { ...block, props: nextProps } : block
   956→    }),
   957→  }))
   958→
   959→  // ---- branding: خلفية المشروع (الشعار يبقى كما هو) ----
   960→  let cleanedBranding: ProjectSchema['branding'] | null = null
   961→  const branding = schema.branding
   962→  if (branding) {
   963→    const b = branding as unknown as Record<string, unknown>
   964→    if (b.bgImage !== undefined || b.bgImageName !== undefined || b.bgOverlay !== undefined) {
   965→      const { bgImage: _bgi, bgImageName: _bgn, bgOverlay: _bgo, ...rest } = b
   966→      cleanedBranding = rest as unknown as ProjectSchema['branding']
   967→      changed = true
   968→    }
   969→  }
   970→
   971→  if (!changed) return schema
   972→  const next: ProjectSchema = { ...schema, pages }
   973→  if (cleanedBranding) next.branding = cleanedBranding
   974→  return next
   975→}
