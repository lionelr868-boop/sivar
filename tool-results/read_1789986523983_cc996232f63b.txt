   117→/** قائمة المكوّنات مع أغلفة وضع التحرير (outline) عند توفّر onBlockSelect */
   118→function BlockList({
   119→  page,
   120→  ctx,
   121→  selectedBlockId,
   122→  onBlockSelect,
   123→}: {
   124→  page: ProjectPage
   125→  ctx: BlockCtx
   126→  selectedBlockId?: string | null
   127→  onBlockSelect?: (blockId: string | null) => void
   128→}): ReactElement {
   129→  return (
   130→    <>
   131→      {page.blocks.map((block, index) => {
   132→        const label = BLOCK_REGISTRY[block.type]?.label ?? block.type
   133→        const selected = selectedBlockId === block.id
   134→        const dimmed = selectedBlockId != null && selectedBlockId !== block.id
   135→
   136→        if (!onBlockSelect) {
   137→          // حركة دخول متدرّجة في وضع المعاينة التفاعلي فقط (بلا وضع التحرير)
   138→          if (ctx.interactive) {
   139→            return (
   140→              <div key={block.id} className="pt-rise" style={{ animationDelay: `${Math.min(index, 6) * 60}ms` }}>
   141→                {renderBlock(block, ctx)}
   142→              </div>
   143→            )
   144→          }
   145→          return <div key={block.id}>{renderBlock(block, ctx)}</div>
   146→        }
   147→
   148→        return (
   149→          <div
   150→            key={block.id}
   151→            role="button"
   152→            tabIndex={0}
   153→            aria-pressed={selected}
   154→            onClick={(e) => {
   155→              e.stopPropagation()
   156→              onBlockSelect(selected ? null : block.id)
   157→            }}
   158→            onKeyDown={(e) => {
   159→              if (e.key === 'Enter' || e.key === ' ') {
   160→                e.preventDefault()
   161→                e.stopPropagation()
   162→                onBlockSelect(selected ? null : block.id)
   163→              }
   164→            }}
   165→            className={`relative group/block cursor-pointer transition-opacity duration-200 ${dimmed ? 'opacity-55' : ''}`}
   166→          >
   167→            {/* المحتوى معطّل المؤشر في وضع التحرير */}
   168→            <div className="pointer-events-none">{renderBlock(block, ctx)}</div>
   169→
   170→            {/* إطار التحديد */}
   171→            <span
   172→              aria-hidden
   173→              className={`pointer-events-none absolute inset-0 z-[6] rounded-[var(--pt-radius)] border-2 transition-opacity duration-150 ${
   174→                selected ? 'opacity-100' : 'opacity-0 group-hover/block:opacity-100'
   175→              }`}
   176→              style={{ borderColor: 'var(--pt-primary)' }}
   177→            />
   178→
   179→            {/* شارة اسم المكوّن */}
   180→            <span
   181→              className={`pointer-events-none absolute top-1 start-1 z-[7] rounded px-2 py-0.5 text-[11px] font-bold text-white transition-opacity duration-150 ${
   182→                selected ? 'opacity-100' : 'opacity-0 group-hover/block:opacity-100'
   183→              }`}
   184→              style={{ background: 'var(--pt-primary)' }}
   185→            >
   186→              {label}
   187→            </span>
   188→          </div>
   189→        )
   190→      })}
   191→    </>
   192→  )
   193→}

   217→function SiteShell({
   218→  schema,
   219→  page,
   220→  ctx,
   221→  selectedBlockId,
   222→  onBlockSelect,
   223→}: {
   224→  schema: ProjectSchema
   225→  page: ProjectPage
   226→  ctx: BlockCtx
   227→  selectedBlockId?: string | null
   228→  onBlockSelect?: (blockId: string | null) => void
   229→}): ReactElement {
   230→  const [menuOpen, setMenuOpen] = useState(false)
   231→  const nav = schema.navigation
   232→  const logo = ctx.schema.branding?.logo
   233→  // خلفية المشروع كلّه (Task 9) — الجذر يصبح شفافاً والصورة خلفه
   234→  const projectBg = ctx.schema.branding?.bgImage
   235→  const homeId = schema.pages.find((pg) => pg.type === 'landing')?.id ?? schema.pages[0]?.id
   236→  const go = (pageId: string) => {
   237→    if (ctx.onNavigate) ctx.onNavigate(pageId)
   238→  }
   239→  const isHome = page.type === 'landing' || page.id === homeId
   240→
   241→  const navLinks = (extraClass: string) => (
   242→    <>
   243→      {nav.items.map((item) => {
   244→        const active = item.pageId === page.id
   245→        return (
   246→          <button
   247→            key={item.pageId}
   248→            type="button"
   249→            onClick={() => {
   250→              go(item.pageId)
   251→              setMenuOpen(false)
   252→            }}
   253→            className={`min-h-10 cursor-pointer rounded-[calc(var(--pt-radius)-2px)] px-3.5 text-sm font-medium transition-colors duration-150 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-[var(--pt-primary)] ${
   254→              active ? 'text-[var(--pt-primary)]' : 'text-[var(--pt-text)] hover:text-[var(--pt-primary)]'
   255→            } ${extraClass}`}
   256→          >
   257→            {item.label}
   258→          </button>
   259→        )
   260→      })}
   261→    </>
   262→  )
   263→
   264→  return (
   265→    <div className={`flex min-h-screen w-full flex-col ${projectBg ? 'bg-transparent' : 'bg-[var(--pt-bg)]'} text-[var(--pt-text)]`}>
   266→      {/* الترويسة */}
   267→      <header className="sticky top-0 z-30 border-b border-[var(--pt-border)] bg-[var(--pt-bg)]/95 backdrop-blur-sm">
   268→        <div className="mx-auto flex w-full max-w-6xl items-center justify-between gap-3 px-4 py-3 md:px-8">
   269→          <button
   270→            type="button"
   271→            onClick={() => homeId && go(homeId)}
   272→            aria-label={`الذهاب إلى ${nav.brand}`}
   273→            className="flex min-h-10 cursor-pointer items-center gap-2 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-[var(--pt-primary)]"
   274→          >
   275→            {logo ? (
   276→              <img
   277→                src={logo}
   278→                alt={nav.brand}
   279→                loading="lazy"
   280→                decoding="async"
   281→                className="h-8 w-auto max-w-[120px] object-contain md:h-9"
   282→              />
   283→            ) : (
   284→              <>
   285→                <span className="size-8 rounded-[calc(var(--pt-radius)-2px)]" style={{ background: 'var(--pt-primary)' }} />
   286→                <span className="text-base font-extrabold text-[var(--pt-text)] md:text-lg">{nav.brand}</span>
   287→              </>
   288→            )}
   289→          </button>
   290→
   291→          {/* تنقل سطح المكتب */}
   292→          <nav aria-label="التنقل الرئيسي" className="hidden items-center gap-1 md:flex">
   293→            {navLinks('')}
   294→            {nav.cta ? (
   295→              <span className="ms-2">
   296→                <PButton size="sm" onClick={() => go(nav.cta!.pageId)}>
   297→                  {nav.cta.label}
   298→                </PButton>
   299→              </span>
   300→            ) : null}
   301→          </nav>
   302→
   303→          {/* زر القائمة (الهاتف) */}
   304→          <button
   305→            type="button"
   306→            aria-label={menuOpen ? 'إغلاق القائمة' : 'فتح القائمة'}
   307→            aria-expanded={menuOpen}
   308→            onClick={() => setMenuOpen((v) => !v)}
   309→            className="flex size-10 cursor-pointer items-center justify-center rounded-[calc(var(--pt-radius)-2px)] border border-[var(--pt-border)] text-[var(--pt-text)] transition-colors duration-150 hover:bg-[var(--pt-secondary)] focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-[var(--pt-primary)] md:hidden"
   310→          >
   311→            {menuOpen ? <X className="size-5" /> : <Menu className="size-5" />}
   312→          </button>
   313→        </div>
   314→
   315→        {/* قائمة الهاتف المنسدلة */}
   316→        {menuOpen ? (
   317→          <nav aria-label="قائمة الهاتف" className="border-t border-[var(--pt-border)] bg-[var(--pt-surface)] px-4 py-3 md:hidden">
   318→            <div className="flex flex-col gap-1">
   319→              {navLinks('text-start')}
   320→              {nav.cta ? (
   321→                <PButton
   322→                  size="md"
   323→                  className="mt-2 w-full"
   324→                  onClick={() => {
   325→                    go(nav.cta!.pageId)
   326→                    setMenuOpen(false)
   327→                  }}
   328→                >
   329→                  {nav.cta.label}
   330→                </PButton>
   331→              ) : null}
   332→            </div>
   333→          </nav>
   334→        ) : null}
   335→      </header>
   336→
   337→      {/* المحتوى */}
   338→      <main className="flex-1">
   339→        <BlockList page={page} ctx={ctx} selectedBlockId={selectedBlockId} onBlockSelect={onBlockSelect} />
   340→      </main>
   341→
   342→      {/* التذييل */}
   343→      <footer className="mt-auto border-t border-[var(--pt-border)] bg-[var(--pt-surface)]">
   344→        <div className="mx-auto w-full max-w-6xl px-4 py-8 md:px-8">
   345→          <div className="flex flex-col items-center justify-between gap-4 sm:flex-row">
   346→            <div className="flex items-center gap-2">
   347→              {logo ? (
   348→                <img
   349→                  src={logo}
   350→                  alt={nav.brand}
   351→                  loading="lazy"
   352→                  decoding="async"
   353→                  className="h-7 w-auto max-w-[100px] object-contain"
   354→                />
   355→              ) : (
   356→                <>
   357→                  <span className="size-6 rounded-[calc(var(--pt-radius)-4px)]" style={{ background: 'var(--pt-primary)' }} />
   358→                  <span className="text-sm font-bold text-[var(--pt-text)]">{nav.brand}</span>
   359→                </>
   360→              )}
   361→              <PBadge tone="primary">بيانات تجريبية</PBadge>
   362→            </div>
   363→            <nav aria-label="تنقل التذييل" className="flex flex-wrap items-center justify-center gap-x-4 gap-y-2">
   364→              {nav.items.map((item) => (
   365→                <button
   366→                  key={`f-${item.pageId}`}
   367→                  type="button"
   368→                  onClick={() => go(item.pageId)}
   369→                  className="min-h-10 cursor-pointer text-[13px] text-[var(--pt-muted)] transition-colors duration-150 hover:text-[var(--pt-primary)] focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-[var(--pt-primary)]"
   370→                >
   371→                  {item.label}
   372→                </button>
   373→              ))}
   374→            </nav>
   375→          </div>
   376→          {nav.footerNote ? (
   377→            <div className="mt-4 flex flex-col items-center gap-2 border-t border-[var(--pt-border)] pt-4 text-center">
   378→              {logo ? (
   379→                <img
   380→                  src={logo}
   381→                  alt=""
   382→                  loading="lazy"
   383→                  decoding="async"
   384→                  className="h-6 w-auto max-w-[90px] object-contain opacity-80"
   385→                />
   386→              ) : null}
   387→              <p className="text-xs text-[var(--pt-muted)]">{nav.footerNote}</p>
   388→            </div>
   389→          ) : null}
   390→        </div>
   391→      </footer>
   392→
   393→      {/* زر العودة للخلف — كل صفحة غير الرئيسية */}
   394→      {!isHome && ctx.interactive && ctx.onNavigate && homeId ? (
   395→        <FloatingBackButton label="عودة" onBack={() => go(homeId)} />
   396→      ) : null}
   397→    </div>
   398→  )
   399→}
