    57→/** مزج لونين محلياً — ratio = وزن اللون الثاني (0..1) */
    58→function ptMix(a: string, b: string, ratio: number): string {
    59→  const expand = (hex: string): [number, number, number] | null => {
    60→    const m = /^#([0-9a-f]{3}|[0-9a-f]{6})$/i.exec(hex.trim())
    61→    if (!m) return null
    62→    let h = m[1]
    63→    if (h.length === 3) h = h.split('').map((c) => c + c).join('')
    64→    return [parseInt(h.slice(0, 2), 16), parseInt(h.slice(2, 4), 16), parseInt(h.slice(4, 6), 16)]
    65→  }
    66→  const ca = expand(a)
    67→  const cb = expand(b)
    68→  if (!ca || !cb) return a
    69→  const t = Math.max(0, Math.min(1, ratio))
    70→  const clamp = (v: number) => Math.max(0, Math.min(255, Math.round(v)))
    71→  const mixed = [0, 1, 2].map((i) => clamp(ca[i] + (cb[i] - ca[i]) * t))
    72→  return `#${mixed.map((v) => v.toString(16).padStart(2, '0')).join('')}`
    73→}

    75→/**
    76→ * متغيرات CSS لألوان قسم واحد (bgColor/textColor اختياريان):
    77→ * - bgColor: خلفية + --pt-bg + --pt-surface/--pt-border مشتقان حسب السطوع.
    78→ * - textColor: --pt-text + --pt-muted ممزوج نحو الخلفية 0.38.
    79→ * - نص تلقائي ذكي: مع bgColor بلا textColor → نص داكن على الخلفية
    80→ *   الفاتحة (سطوع > 0.5) وأبيض على الداكنة (+ ثانوي مشتق).
    81→ * تُرجع null عندما لا ألوان مخصّصة (سلوك قديم مطابق بالبايت).
    82→ */
    83→export function sectionVars(bgColor: unknown, textColor: unknown): CSSProperties | null {
    84→  const bg = sectionHexColor(bgColor)
    85→  const tx = sectionHexColor(textColor)
    86→  if (!bg && !tx) return null
    87→
    88→  const vars: Record<string, string> = {}
    89→  if (bg) {
    90→    const bright = hexLuminance(bg) > 0.55
    91→    vars['--pt-bg'] = bg
    92→    vars['--pt-surface'] = bright ? hexLighten(bg, 0.42) : hexDarken(bg, 0.42)
    93→    vars['--pt-border'] = bright ? hexLighten(bg, 0.18) : hexDarken(bg, 0.18)
    94→    vars.background = bg
    95→  }
    96→
    97→  let text = tx
    98→  if (bg && !tx) text = hexLuminance(bg) > 0.5 ? '#17151f' : '#ffffff'
    99→  if (text) {
   100→    vars['--pt-text'] = text
   101→    vars['--pt-muted'] = ptMix(text, bg || '#ffffff', 0.38)
   102→  }
   103→  vars.color = 'var(--pt-text)'
   104→
   105→  return vars as CSSProperties
   106→}

   192→/** قائمة المكوّنات مع أغلفة وضع التحرير (outline) عند توفّر onBlockSelect */
   193→function BlockList({
   194→  page,
   195→  ctx,
   196→  selectedBlockId,
   197→  onBlockSelect,
   198→}: {
   199→  page: ProjectPage
   200→  ctx: BlockCtx
   201→  selectedBlockId?: string | null
   202→  onBlockSelect?: (blockId: string | null) => void
   203→}): ReactElement {
   204→  return (
   205→    <>
   206→      {page.blocks.map((block, index) => {
   207→        const label = BLOCK_REGISTRY[block.type]?.label ?? block.type
   208→        const selected = selectedBlockId === block.id
   209→        const dimmed = selectedBlockId != null && selectedBlockId !== block.id
   210→
   211→        if (!onBlockSelect) {
   212→          // حركة دخول متدرّجة في وضع المعاينة التفاعلي فقط (بلا وضع التحرير)
   213→          if (ctx.interactive) {
   214→            return (
   215→              <div key={block.id} className="pt-rise" style={{ animationDelay: `${Math.min(index, 6) * 60}ms` }}>
   216→                {renderBlock(block, ctx)}
   217→              </div>
   218→            )
   219→          }
   220→          return <div key={block.id}>{renderBlock(block, ctx)}</div>
   221→        }
   222→
   223→        return (
   224→          <div
   225→            key={block.id}
   226→            role="button"
   227→            tabIndex={0}
   228→            aria-pressed={selected}
   229→            onClick={(e) => {
   230→              e.stopPropagation()
   231→              onBlockSelect(selected ? null : block.id)
   232→            }}
   233→            onKeyDown={(e) => {
   234→              if (e.key === 'Enter' || e.key === ' ') {
   235→                e.preventDefault()
   236→                e.stopPropagation()
   237→                onBlockSelect(selected ? null : block.id)
   238→              }
   239→            }}
   240→            className={`relative group/block cursor-pointer transition-opacity duration-200 ${dimmed ? 'opacity-55' : ''}`}
   241→          >
   242→            {/* المحتوى معطّل المؤشر في وضع التحرير */}
   243→            <div className="pointer-events-none">{renderBlock(block, ctx)}</div>
   244→
   245→            {/* إطار التحديد */}
   246→            <span
   247→              aria-hidden
   248→              className={`pointer-events-none absolute inset-0 z-[6] rounded-[var(--pt-radius)] border-2 transition-opacity duration-150 ${
   249→                selected ? 'opacity-100' : 'opacity-0 group-hover/block:opacity-100'
   250→              }`}
   251→              style={{ borderColor: 'var(--pt-primary)' }}
   252→            />
   253→
   254→            {/* شارة اسم المكوّن */}
   255→            <span
   256→              className={`pointer-events-none absolute top-1 start-1 z-[7] rounded px-2 py-0.5 text-[11px] font-bold text-white transition-opacity duration-150 ${
   257→                selected ? 'opacity-100' : 'opacity-0 group-hover/block:opacity-100'
   258→              }`}
   259→              style={{ background: 'var(--pt-primary)' }}
   260→            >
   261→              {label}
   262→            </span>
   263→          </div>
   264→        )
   265→      })}
   266→    </>
   267→  )
   268→}

   292→function SiteShell({
   293→  schema,
   294→  page,
   295→  ctx,
   296→  selectedBlockId,
   297→  onBlockSelect,
   298→}: {
   299→  schema: ProjectSchema
   300→  page: ProjectPage
   301→  ctx: BlockCtx
   302→  selectedBlockId?: string | null
   303→  onBlockSelect?: (blockId: string | null) => void
   304→}): ReactElement {
   305→  const [menuOpen, setMenuOpen] = useState(false)
   306→  const nav = schema.navigation
   307→  const logo = ctx.schema.branding?.logo
   308→  // خلفية المشروع كلّه (Task 9) — الجذر يصبح شفافاً والصورة خلفه
   309→  const projectBg = ctx.schema.branding?.bgImage
   310→  const homeId = schema.pages.find((pg) => pg.type === 'landing')?.id ?? schema.pages[0]?.id
   311→  const go = (pageId: string) => {
   312→    if (ctx.onNavigate) ctx.onNavigate(pageId)
   313→  }
   314→  const isHome = page.type === 'landing' || page.id === homeId
   315→
   316→  const navLinks = (extraClass: string) => (
   317→    <>
   318→      {nav.items.map((item) => {
   319→        const active = item.pageId === page.id
   320→        return (
   321→          <button
   322→            key={item.pageId}
   323→            type="button"
   324→            onClick={() => {
   325→              go(item.pageId)
   326→              setMenuOpen(false)
   327→            }}
   328→            className={`min-h-10 cursor-pointer rounded-[calc(var(--pt-radius)-2px)] px-3.5 text-sm font-medium transition-colors duration-150 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-[var(--pt-primary)] ${
   329→              active ? 'text-[var(--pt-primary)]' : 'text-[var(--pt-text)] hover:text-[var(--pt-primary)]'
   330→            } ${extraClass}`}

   332→            {item.label}

   340→    <div className={`flex min-h-screen w-full flex-col ${projectBg ? 'bg-transparent' : 'bg-[var(--pt-bg)]'} text-[var(--pt-text)]`}>
   341→      {/* الترويسة */}
   342→      <header className="sticky top-0 z-30 border-b border-[var(--pt-border)] bg-[var(--pt-bg)]/95 backdrop-blur-sm">
   343→        <div className="mx-auto flex w-full max-w-6xl items-center justify-between gap-3 px-4 py-3 md:px-8">

   345→            type="button"
   346→            onClick={() => homeId && go(homeId)}
   347→            aria-label={`الذهاب إلى ${nav.brand}`}
   348→            className="flex min-h-10 cursor-pointer items-center gap-2 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-[var(--pt-primary)]"

   350→            {logo ? (
   351→              <img
   352→                src={logo}
   353→                alt={nav.brand}
   354→                loading="lazy"
   355→                decoding="async"
   356→                className="h-8 w-auto max-w-[120px] object-contain md:h-9"
   357→              />
   358→            ) : (
   359→              <>
   360→                <span className="size-8 rounded-[calc(var(--pt-radius)-2px)]" style={{ background: 'var(--pt-primary)' }} />
   361→                <span className="text-base font-extrabold text-[var(--pt-text)] md:text-lg">{nav.brand}</span>
   362→              </>
   363→            )}
   364→          </button>
   365→
   366→          {/* تنقل سطح المكتب */}

   370→              <span className="ms-2">
   371→                <PButton size="sm" onClick={() => go(nav.cta!.pageId)}>
   372→                  {nav.cta.label}

   380→            type="button"
   381→            aria-label={menuOpen ? 'إغلاق القائمة' : 'فتح القائمة'}

   383→            onClick={() => setMenuOpen((v) => !v)}
   384→            className="flex size-10 cursor-pointer items-center justify-center rounded-[calc(var(--pt-radius)-2px)] border border-[var(--pt-border)] text-[var(--pt-text)] transition-colors duration-150 hover:bg-[var(--pt-secondary)] focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-[var(--pt-primary)] md:hidden"

   421→            <div className="flex items-center gap-2">
   422→              {logo ? (
   423→                <img
   424→                  src={logo}
   425→                  alt={nav.brand}
   426→                  loading="lazy"
   427→                  decoding="async"
   428→                  className="h-7 w-auto max-w-[100px] object-contain"
   429→                />
   430→              ) : (
   431→                <>
   432→                  <span className="size-6 rounded-[calc(var(--pt-radius)-4px)]" style={{ background: 'var(--pt-primary)' }} />
   433→                  <span className="text-sm font-bold text-[var(--pt-text)]">{nav.brand}</span>
   434→                </>
   435→              )}
   436→              <PBadge tone="primary">بيانات تجريبية</PBadge>
   437→            </div>
   438→            <nav aria-label="تنقل التذييل" className="flex flex-wrap items-center justify-center gap-x-4 gap-y-2">

   440→                <button
   441→                  key={`f-${item.pageId}`}
   442→                  type="button"
   443→                  onClick={() => go(item.pageId)}
   444→                  className="min-h-10 cursor-pointer text-[13px] text-[var(--pt-muted)] transition-colors duration-150 hover:text-[var(--pt-primary)] focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-[var(--pt-primary)]"
   445→                >
   446→                  {item.label}
   447→                </button>
   448→              ))}
   449→            </nav>
   450→          </div>
   451→          {nav.footerNote ? (
   452→            <div className="mt-4 flex flex-col items-center gap-2 border-t border-[var(--pt-border)] pt-4 text-center">
   453→              {logo ? (
   454→                <img

   458→                  decoding="async"
   459→                  className="h-6 w-auto max-w-[90px] object-contain opacity-80"

   462→              <p className="text-xs text-[var(--pt-muted)]">{nav.footerNote}</p>

   927→  /** وضع التحرير: أغلفة تحديد حول كل مكوّن */
   928→  onBlockSelect?: (blockId: string | null) => void
   929→  /** روابط التنقل بين صفحات النموذج */
   930→  onNavigate?: (pageId: string) => void
   931→  /** وضع المعاينة: النماذج والأزرار حيّة */

   939→    schema.pages[0]
   940→
   941→  const ctx: BlockCtx = {

   950→  // خلفية المشروع كلّه (Task 9) — تُمرَّر للنطاق فترسم خلف كل الصفحات
   951→  const brandingBg = schema.branding?.bgImage

   958→      bgOverlay={schema.branding?.bgOverlay ?? 'medium'}
   959→    >

   961→        <div className="flex min-h-96 items-center justify-center bg-transparent p-8 text-center text-sm text-[var(--pt-muted)]">
   962→          لا توجد صفحات في هذا النموذج بعد.
   963→        </div>

   965→        <AuthShell {...shellProps} />

   967→        <DashboardShell {...shellProps} />

   970→      )}
   971→    </PrototypeThemeScope>
