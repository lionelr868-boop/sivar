    73→    73→        {children}

   116→   116→export function NewProjectFlow() {
   117→   117→  const openProject = useAppStore((s) => s.openProject)
   118→   118→  const setDashboardSection = useAppStore((s) => s.setDashboardSection)
   119→   119→
   120→   120→  // ---------- حالة المسار ----------
   121→   121→  const [step, setStep] = useState<Step>('idea')
   122→   122→  const [idea, setIdea] = useState('')
   123→   123→  const [type, setType] = useState<ProjectType | ''>('')
   124→   124→  const [audience, setAudience] = useState('')
   125→   125→  const [style, setStyle] = useState<ProjectStyle | ''>('')
   126→   126→  const [colorMode, setColorMode] = useState<'auto' | 'custom'>('auto')
   127→   127→  const [colors, setColors] = useState({ primary: '#7c3aed', secondary: '#0d9488', accent: '#d97706' })
   128→   128→  const [optionsOpen, setOptionsOpen] = useState(false)
   129→   129→
   130→   130→  const [analysis, setAnalysis] = useState<IdeaAnalysis | null>(null)
   131→   131→  const [projectName, setProjectName] = useState('')
   132→   132→  const [selectedPages, setSelectedPages] = useState<SuggestedPage[]>([])
   133→   133→  const [editMode, setEditMode] = useState(false)
   134→   134→
   135→   135→  const [analyzeStage, setAnalyzeStage] = useState(0)
   136→   136→  const [buildStage, setBuildStage] = useState(0)
   137→   137→  const [buildError, setBuildError] = useState(false)
   138→   138→
   139→   139→  // مؤقّتات المراحل — تُنظّف عند إلغاء التثبيت
   140→   140→  const analyzeTimer = useRef<ReturnType<typeof setInterval> | null>(null)
   141→   141→  const buildTimer = useRef<ReturnType<typeof setInterval> | null>(null)
   142→   142→  useEffect(
   143→   143→    () => () => {
   144→   144→      if (analyzeTimer.current) clearInterval(analyzeTimer.current)
   145→   145→      if (buildTimer.current) clearInterval(buildTimer.current)
   146→   146→    },
   147→   147→    [],
   148→   148→  )
   149→   149→
   150→   150→  const ideaReady = idea.trim().length >= MIN_IDEA_LENGTH
   151→   151→
   152→   152→  const analyzeStageState = (i: number): 'pending' | 'active' | 'done' =>
   153→   153→    i < analyzeStage ? 'done' : i === analyzeStage ? 'active' : 'pending'
   154→   154→
   155→   155→  // ---------- الخطوة 2: التحليل ----------
   156→   156→  const startAnalysis = async () => {
   157→   157→    if (!ideaReady) return
   158→   158→    setStep('analyzing')
   159→   159→    setAnalyzeStage(0)
   160→   160→    if (analyzeTimer.current) clearInterval(analyzeTimer.current)
   161→   161→    analyzeTimer.current = setInterval(() => {
   162→   162→      setAnalyzeStage((s) => Math.min(s + 1, ANALYZE_STAGES.length - 1))
   163→   163→    }, 900)
   164→   164→
   165→   165→    try {
   166→   166→      const [result] = await Promise.all([api.analyzeIdea(idea.trim()), delay(2400)])
   167→   167→      if (analyzeTimer.current) clearInterval(analyzeTimer.current)
   168→   168→      setAnalysis(result.analysis)
   169→   169→      setProjectName(result.analysis.name)
   170→   170→      setSelectedPages(result.analysis.suggestedPages)
   171→   171→      setStep('review')
   172→   172→    } catch {
   173→   173→      if (analyzeTimer.current) clearInterval(analyzeTimer.current)
   174→   174→      toast.error(tNP.errorAnalyze)
   175→   175→      setStep('idea')
   176→   176→    }
   177→   177→  }
   178→   178→
   179→   179→  // ---------- الخطوة 4: البناء ----------
   180→   180→  const startBuilding = async () => {
   181→   181→    if (!analysis) return
   182→   182→    setStep('building')
   183→   183→    setBuildStage(0)
   184→   184→    setBuildError(false)
   185→   185→    if (buildTimer.current) clearInterval(buildTimer.current)
   186→   186→    buildTimer.current = setInterval(() => {
   187→   187→      setBuildStage((s) => Math.min(s + 1, BUILD_STAGES.length - 1))
   188→   188→    }, 700)
   189→   189→
   190→   190→    const finalAnalysis: IdeaAnalysis = {
   191→   191→      ...analysis,
   192→   192→      name: projectName.trim() || analysis.name,
   193→   193→      audience: audience.trim() || analysis.audience,
   194→   194→      suggestedPages: selectedPages.length > 0 ? selectedPages : analysis.suggestedPages,
   195→   195→    }
   196→   196→
   197→   197→    try {
   198→   198→      const [result] = await Promise.all([
   199→   199→        api.createProject({
   200→   200→          idea: idea.trim(),
   201→   201→          analysis: finalAnalysis,
   202→   202→          name: finalAnalysis.name,
   203→   203→          type: type || undefined,
   204→   204→          style: style || undefined,
   205→   205→          audience: finalAnalysis.audience,
   206→   206→          palette: colorMode === 'custom' ? colors : null,
   207→   207→        }),
   208→   208→        delay(2800),
   209→   209→      ])
   210→   210→      if (buildTimer.current) clearInterval(buildTimer.current)
   211→   211→      setStep('success')
   212→   212→      setTimeout(() => void openProject(result.project.id), 1200)
   213→   213→    } catch {
   214→   214→      if (buildTimer.current) clearInterval(buildTimer.current)
   215→   215→      setBuildError(true)
   216→   216→      toast.error(tNP.errorCreate)
   217→   217→    }
   218→   218→  }
   219→   219→
   220→   220→  const togglePage = (page: SuggestedPage) => {
   221→   221→    setSelectedPages((pages) => {
   222→   222→      const exists = pages.some((p) => p.name === page.name && p.type === page.type)
   223→   223→      return exists
   224→   224→        ? pages.filter((p) => !(p.name === page.name && p.type === page.type))
   225→   225→        : [...pages, page]
   226→   226→    })
   227→   227→  }
   228→   228→
   229→   229→  const stepKey = useMemo(() => `${step}${editMode ? '-edit' : ''}`, [step, editMode])
   230→   230→
   231→   231→  return (
   232→   232→    <div className="mx-auto w-full max-w-2xl space-y-4">
   233→   233→      {/* زر الرجوع الصغير */}
   234→   234→      <Button
   235→   235→        variant="ghost"
   236→   236→        className="h-10 px-3 text-muted-foreground"
   237→   237→        onClick={() => setDashboardSection('projects')}
   238→   238→      >
   239→   239→        <ArrowRight className="size-4" />
   240→   240→        {tNP.backToProjects}
   241→   241→      </Button>
   242→   242→
   243→   243→      <StepShell stepKey={stepKey}>
   244→   244→        {/* ================ الخطوة 1: الفكرة ================ */}
   245→   245→        {step === 'idea' ? (
   246→   246→          <div className="space-y-6">
   247→   247→            <div className="space-y-2 text-center">
   248→   248→              <span className="mx-auto grid size-14 place-items-center rounded-2xl bg-primary/10 text-primary">
   249→   249→                <Sparkles className="size-7" aria-hidden />
   250→   250→              </span>
   251→   251→              <h1 className="font-display text-2xl font-extrabold tracking-tight sm:text-3xl">
   252→   252→                {tNP.ideaTitle}
   253→   253→              </h1>
   254→   254→              <p className="text-sm text-muted-foreground sm:text-base">{tNP.ideaSubtitle}</p>
   255→   255→            </div>
   256→   256→
   257→   257→            <div className="space-y-2.5">
   258→   258→              <Label htmlFor="new-idea" className="sr-only">
   259→   259→                {tNP.ideaTitle}
   260→   260→              </Label>
   261→   261→              <Textarea
   262→   262→                id="new-idea"
   263→   263→                value={idea}
   264→   264→                onChange={(e) => setIdea(e.target.value)}
   265→   265→                placeholder={tNP.ideaPlaceholder}
   266→   266→                className="min-h-40 text-base leading-relaxed sm:text-lg"
   267→   267→              />
   268→   268→              <p
   269→   269→                className={`text-xs ${ideaReady ? 'text-muted-foreground' : 'text-amber-600 dark:text-amber-400'}`}
   270→   270→              >
   271→   271→                {tNP.ideaTooShort}
   272→   272→              </p>
   273→   273→            </div>
   274→   274→
   275→   275→            {/* خيارات إضافية */}
   276→   276→            <Collapsible open={optionsOpen} onOpenChange={setOptionsOpen}>
   277→   277→              <CollapsibleTrigger className="group flex min-h-11 items-center gap-1.5 rounded-lg text-sm font-semibold text-primary transition-colors hover:text-primary/80 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-ring">
   278→   278→                {tNP.moreOptions}
   279→   279→                <span className="text-xs font-normal text-muted-foreground">
   280→   280→                  ({t.common.optional})
   281→   281→                </span>
   282→   282→                <ChevronDown
   283→   283→                  aria-hidden
   284→   284→                  className="size-4 text-muted-foreground transition-transform duration-200 group-data-[state=open]:rotate-180"
   285→   285→                />
   286→   286→              </CollapsibleTrigger>
   287→   287→              <CollapsibleContent>
   288→   288→                <div className="mt-4 space-y-6 rounded-xl border bg-muted/30 p-4 sm:p-5">
   289→   289→                  <p className="text-xs text-muted-foreground">{tNP.moreOptionsHint}</p>
   290→   290→
   291→   291→                  {/* نوع المشروع */}
   292→   292→                  <div className="space-y-2.5">
   293→   293→                    <p className="text-sm font-semibold">{tNP.typeLabel}</p>
   294→   294→                    <div className="flex flex-wrap gap-2">
   295→   295→                      {QUICK_TYPES.map((v) => {
   296→   296→                        const active = type === v
   297→   297→                        return (
   298→   298→                          <button
   299→   299→                            key={v}
   300→   300→                            type="button"
   301→   301→                            aria-pressed={active}
   302→   302→                            onClick={() => setType((cur) => (cur === v ? '' : v))}
   303→   303→                            className={`h-10 min-h-10 rounded-full border px-4 text-xs font-medium transition-colors focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-ring ${
   304→   304→                              active
   305→   305→                                ? 'border-primary bg-primary text-primary-foreground'
   306→   306→                                : 'border-border bg-background text-muted-foreground hover:bg-accent hover:text-foreground'
   307→   307→                            }`}
   308→   308→                          >
   309→   309→                            {PROJECT_TYPE_LABELS[v]}
   310→   310→                          </button>
   311→   311→                        )
   312→   312→                      })}
   313→   313→                    </div>
   314→   314→                  </div>
   315→   315→
   316→   316→                  {/* الجمهور */}
   317→   317→                  <div className="space-y-2.5">
   318→   318→                    <Label htmlFor="new-audience" className="text-sm font-semibold">
   319→   319→                      {tNP.audienceLabel}
   320→   320→                    </Label>
   321→   321→                    <Input
   322→   322→                      id="new-audience"
   323→   323→                      value={audience}
   324→   324→                      onChange={(e) => setAudience(e.target.value)}
   325→   325→                      placeholder={tNP.audiencePlaceholder}
   326→   326→                      className="bg-background"
   327→   327→                    />
   328→   328→                  </div>
   329→   329→
   330→   330→                  {/* النمط */}
   331→   331→                  <div className="space-y-2.5">
   332→   332→                    <p className="text-sm font-semibold">{tNP.styleLabel}</p>
   333→   333→                    <div className="flex flex-wrap gap-2">
   334→   334→                      {QUICK_STYLES.map((v) => {
   335→   335→                        const active = style === v
   336→   336→                        return (
   337→   337→                          <button
   338→   338→                            key={v}
   339→   339→                            type="button"
   340→   340→                            aria-pressed={active}
   341→   341→                            onClick={() => setStyle((cur) => (cur === v ? '' : v))}
   342→   342→                            className={`h-10 min-h-10 rounded-full border px-4 text-xs font-medium transition-colors focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-ring ${
   343→   343→                              active
   344→   344→                                ? 'border-primary bg-primary text-primary-foreground'
   345→   345→                                : 'border-border bg-background text-muted-foreground hover:bg-accent hover:text-foreground'
   346→   346→                            }`}
   347→   347→                          >
   348→   348→                            {PROJECT_STYLE_LABELS[v]}
   349→   349→                          </button>
   350→   350→                        )
   351→   351→                      })}
   352→   352→                    </div>
   353→   353→                  </div>
   354→   354→
   355→   355→                  {/* الألوان */}
   356→   356→                  <div className="space-y-2.5">
   357→   357→                    <p className="text-sm font-semibold">{tNP.colorsLabel}</p>
   358→   358→                    <div className="flex flex-wrap items-center gap-2">
   359→   359→                      <button
   360→   360→                        type="button"
   361→   361→                        aria-pressed={colorMode === 'auto'}
   362→   362→                        onClick={() => setColorMode('auto')}
   363→   363→                        className={`h-10 min-h-10 rounded-full border px-4 text-xs font-medium transition-colors focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-ring ${
   364→   364→                          colorMode === 'auto'
   365→   365→                            ? 'border-primary bg-primary text-primary-foreground'
   366→   366→                            : 'border-border bg-background text-muted-foreground hover:bg-accent hover:text-foreground'
   367→   367→                        }`}
   368→   368→                      >
   369→   369→                        {tNP.colorsAuto}
   370→   370→                      </button>
   371→   371→                      <button
   372→   372→                        type="button"
   373→   373→                        aria-pressed={colorMode === 'custom'}
   374→   374→                        onClick={() => setColorMode('custom')}
   375→   375→                        className={`h-10 min-h-10 rounded-full border px-4 text-xs font-medium transition-colors focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-ring ${
   376→   376→                          colorMode === 'custom'
   377→   377→                            ? 'border-primary bg-primary text-primary-foreground'
   378→   378→                            : 'border-border bg-background text-muted-foreground hover:bg-accent hover:text-foreground'
   379→   379→                        }`}
   380→   380→                      >
   381→   381→                        {tNP.colorsCustom}
   382→   382→                      </button>
   383→   383→                    </div>
   384→   384→                    {colorMode === 'custom' ? (
   385→   385→                      <div className="grid grid-cols-3 gap-3">
   386→   386→                        {(
   387→   387→                          [
   388→   388→                            { key: 'primary', label: tNP.colorPrimary },
   389→   389→                            { key: 'secondary', label: tNP.colorSecondary },
   390→   390→                            { key: 'accent', label: tNP.colorAccent },
   391→   391→                          ] as const
   392→   392→                        ).map((c) => (
   393→   393→                          <div key={c.key} className="space-y-1.5">
   394→   394→                            <Label htmlFor={`color-${c.key}`} className="text-xs font-medium text-muted-foreground">
   395→   395→                              {c.label}
   396→   396→                            </Label>
   397→   397→                            <input
   398→   398→                              id={`color-${c.key}`}
   399→   399→                              type="color"
   400→   400→                              value={colors[c.key]}
   401→   401→                              onChange={(e) => setColors((prev) => ({ ...prev, [c.key]: e.target.value }))}
   402→   402→                              className="h-11 w-full cursor-pointer rounded-lg border bg-background p-1.5"
   403→   403→                              aria-label={c.label}
   404→   404→                            />
   405→   405→                          </div>
   406→   406→                        ))}
   407→   407→                      </div>
   408→   408→                    ) : null}
   409→   409→                  </div>
   410→   410→                </div>
   411→   411→              </CollapsibleContent>
   412→   412→            </Collapsible>
   413→   413→
   414→   414→            <Button
   415→   415→              size="lg"
   416→   416→              className="h-12 w-full text-base"
   417→   417→              disabled={!ideaReady}
   418→   418→              onClick={() => void startAnalysis()}
   419→   419→            >
   420→   420→              {tNP.submitAnalyze}
   421→   421→            </Button>
   422→   422→          </div>
   423→   423→        ) : null}
   424→   424→
   425→   425→        {/* ================ الخطوة 2: التحليل ================ */}
   426→   426→        {step === 'analyzing' ? (
   427→   427→          <div className="space-y-6 py-8">
   428→   428→            <div className="space-y-3 text-center">
   429→   429→              <Loader2
   430→   430→                className="mx-auto size-10 animate-spin text-primary"
   431→   431→                aria-hidden
   432→   432→              />
