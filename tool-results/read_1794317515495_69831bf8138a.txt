   240→// ---------- seed helpers ----------
   241→/** تدوير الصفوف — seed 0 يعني بدون تغيير */
   242→function rot<T>(rows: T[], seed: number): T[] {
   243→  if (!seed || rows.length === 0) return [...rows]
   244→  const r = ((seed % rows.length) + rows.length) % rows.length
   245→  return [...rows.slice(r), ...rows.slice(0, r)]
   246→}
   247→
   248→/** اهتزاز رقمي بسيط — seed 0 يعني بدون تغيير */
   249→function jit(v: number, seed: number, pct = 0.08): number {
   250→  if (!seed) return v
   251→  const f = 1 + (((seed % 7) - 3) * pct) / 3
   252→  return Math.max(0, Math.round(v * f))
   253→}
   254→
   255→const NUMERIC_COLUMN_TYPES = new Set<string>(['number', 'currency', 'progress'])
   256→
   257→/** بناء كيان مع تطبيق seed (تدوير + اهتزاز الأرقام) — نسخة مستقلة عن الأعمدة */
   258→function makeEntity(
   259→  columns: EntityColumn[],
   260→  baseRows: Array<Record<string, unknown>>,
   261→  seed: number
   262→): EntityData {
   263→  const cols = columns.map((c) => ({ ...c }))
   264→  const rows = rot(baseRows, seed).map((row) => {
   265→    const out: Record<string, unknown> = {}
   266→    for (const col of cols) {
   267→      const v = row[col.key]
   268→      if (NUMERIC_COLUMN_TYPES.has(col.type) && typeof v === 'number') {
   269→        out[col.key] = jit(v, seed)
   270→      } else {
   271→        out[col.key] = v
   272→      }
   273→    }
   274→    return out
   275→  })
   276→  return { columns: cols, rows }
   277→}
