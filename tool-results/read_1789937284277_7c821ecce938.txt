   117→/** قائمة المكوّنات مع أغلفة وضع التحرير (outline) عند توفّر onBlockSelect */
   118→function BlockList({
   119→  page,
   120→  ctx,
   121→  selectedBlockId,
   122→  onBlockSelect,
   123→}: {
   124→  page: ProjectPage
   125→  ctx: BlockCtx
   126→  selectedBlockId?: string | null
   127→  onBlockSelect?: (blockId: string | null) => void
   128→}): ReactElement {
   129→  return (
   130→    <>
   131→      {page.blocks.map((block, index) => {
   132→        const label = BLOCK_REGISTRY[block.type]?.label ?? block.type
   133→        const selected = selectedBlockId === block.id
   134→        const dimmed = selectedBlockId != null && selectedBlockId !== block.id
   135→
   136→        if (!onBlockSelect) {
   137→          // حركة دخول متدرّجة في وضع المعاينة التفاعلي فقط (بلا وضع التحرير)
   138→          if (ctx.interactive) {
   139→            return (
   140→              <div key={block.id} className="pt-rise" style={{ animationDelay: `${Math.min(index, 6) * 60}ms` }}>
   141→                {renderBlock(block, ctx)}
   142→              </div>
   143→            )
   144→          }
   145→          return <div key={block.id}>{renderBlock(block, ctx)}</div>
   146→        }
   147→
   148→        return (
   149→          <div
   150→            key={block.id}
   151→            role="button"
   152→            tabIndex={0}
   153→            aria-pressed={selected}
   154→            onClick={(e) => {
   155→              e.stopPropagation()
   156→              onBlockSelect(selected ? null : block.id)
   157→            }}
   158→            onKeyDown={(e) => {
   159→              if (e.key === 'Enter' || e.key === ' ') {
   160→                e.preventDefault()
   161→                e.stopPropagation()
   162→                onBlockSelect(selected ? null : block.id)
   163→              }
   164→            }}
   165→            className={`relative group/block cursor-pointer transition-opacity duration-200 ${dimmed ? 'opacity-55' : ''}`}
   166→          >
   167→            {/* المحتوى معطّل المؤشر في وضع التحرير */}
   168→            <div className="pointer-events-none">{renderBlock(block, ctx)}</div>
   169→
   170→            {/* إطار التحديد */}
   171→            <span
   172→              aria-hidden
   173→              className={`pointer-events-none absolute inset-0 z-[6] rounded-[var(--pt-radius)] border-2 transition-opacity duration-150 ${
   174→                selected ? 'opacity-100' : 'opacity-0 group-hover/block:opacity-100'
   175→              }`}
   176→              style={{ borderColor: 'var(--pt-primary)' }}
   177→            />
   178→
   179→            {/* شارة اسم المكوّن */}
   180→            <span
   181→              className={`pointer-events-none absolute top-1 start-1 z-[7] rounded px-2 py-0.5 text-[11px] font-bold text-white transition-opacity duration-150 ${
   182→                selected ? 'opacity-100' : 'opacity-0 group-hover/block:opacity-100'
   183→              }`}
   184→              style={{ background: 'var(--pt-primary)' }}
   185→            >
   186→              {label}
   187→            </span>
   188→          </div>
   189→        )
   190→      })}
   191→    </>
   192→  )
   193→}

   217→function SiteShell({
   218→  schema,
   219→  page,
   220→  ctx,
   221→  selectedBlockId,
   222→  onBlockSelect,

   224→  schema: ProjectSchema
   225→  page: ProjectPage

   228→  onBlockSelect?: (blockId: string | null) => void

   235→    if (ctx.onNavigate) ctx.onNavigate(pageId)
   236→  }
   237→  const isHome = page.type === 'landing' || page.id === homeId
   238→
   239→  const navLinks = (extraClass: string) => (
   240→    <>
   241→      {nav.items.map((item) => {
   242→        const active = item.pageId === page.id
   243→        return (
   244→          <button
   245→            key={item.pageId}
   246→            type="button"
   247→            onClick={() => {
   248→              go(item.pageId)
   249→              setMenuOpen(false)
   250→            }}
   251→            className={`min-h-10 cursor-pointer rounded-[calc(var(--pt-radius)-2px)] px-3.5 text-sm font-medium transition-colors duration-150 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-[var(--pt-primary)] ${
   252→              active ? 'text-[var(--pt-primary)]' : 'text-[var(--pt-text)] hover:text-[var(--pt-primary)]'
   253→            } ${extraClass}`}
   254→          >
   255→            {item.label}
   256→          </button>
   257→        )
   258→      })}
   259→    </>
   260→  )
   261→
   262→  return (
   263→    <div className="flex min-h-screen w-full flex-col bg-[var(--pt-bg)] text-[var(--pt-text)]">
   264→      {/* الترويسة */}
   265→      <header className="sticky top-0 z-30 border-b border-[var(--pt-border)] bg-[var(--pt-bg)]/95 backdrop-blur-sm">
   266→        <div className="mx-auto flex w-full max-w-6xl items-center justify-between gap-3 px-4 py-3 md:px-8">
   267→          <button
   268→            type="button"
   269→            onClick={() => homeId && go(homeId)}
   270→            aria-label={`الذهاب إلى ${nav.brand}`}
   271→            className="flex min-h-10 cursor-pointer items-center gap-2 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-[var(--pt-primary)]"
   272→          >
   273→            {logo ? (
   274→              <img
   275→                src={logo}
   276→                alt={nav.brand}
   277→                loading="lazy"
   278→                decoding="async"
   279→                className="h-8 w-auto max-w-[120px] object-contain md:h-9"
   280→              />
   281→            ) : (
   282→              <>
   283→                <span className="size-8 rounded-[calc(var(--pt-radius)-2px)]" style={{ background: 'var(--pt-primary)' }} />
   284→                <span className="text-base font-extrabold text-[var(--pt-text)] md:text-lg">{nav.brand}</span>
   285→              </>
   286→            )}
   287→          </button>
   288→
   289→          {/* تنقل سطح المكتب */}
   290→          <nav aria-label="التنقل الرئيسي" className="hidden items-center gap-1 md:flex">
   291→            {navLinks('')}
   292→            {nav.cta ? (
   293→              <span className="ms-2">
   294→                <PButton size="sm" onClick={() => go(nav.cta!.pageId)}>
   295→                  {nav.cta.label}
   296→                </PButton>
   297→              </span>
   298→            ) : null}
   299→          </nav>
   300→
   301→          {/* زر القائمة (الهاتف) */}
   302→          <button
   303→            type="button"
   304→            aria-label={menuOpen ? 'إغلاق القائمة' : 'فتح القائمة'}
   305→            aria-expanded={menuOpen}
   306→            onClick={() => setMenuOpen((v) => !v)}
   307→            className="flex size-10 cursor-pointer items-center justify-center rounded-[calc(var(--pt-radius)-2px)] border border-[var(--pt-border)] text-[var(--pt-text)] transition-colors duration-150 hover:bg-[var(--pt-secondary)] focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-[var(--pt-primary)] md:hidden"
   308→          >
   309→            {menuOpen ? <X className="size-5" /> : <Menu className="size-5" />}
   310→          </button>
   311→        </div>
   312→
   313→        {/* قائمة الهاتف المنسدلة */}
   314→        {menuOpen ? (
   315→          <nav aria-label="قائمة الهاتف" className="border-t border-[var(--pt-border)] bg-[var(--pt-surface)] px-4 py-3 md:hidden">
   316→            <div className="flex flex-col gap-1">
   317→              {navLinks('text-start')}
   318→              {nav.cta ? (
   319→                <PButton
   320→                  size="md"
   321→                  className="mt-2 w-full"
   322→                  onClick={() => {
   323→                    go(nav.cta!.pageId)
   324→                    setMenuOpen(false)
   325→                  }}
   326→                >
   327→                  {nav.cta.label}
   328→                </PButton>
   329→              ) : null}
   330→            </div>
   331→          </nav>
   332→        ) : null}
   333→      </header>
   334→
   335→      {/* المحتوى */}
   336→      <main className="flex-1">
   337→        <BlockList page={page} ctx={ctx} selectedBlockId={selectedBlockId} onBlockSelect={onBlockSelect} />
   338→      </main>
   339→
   340→      {/* التذييل */}
   341→      <footer className="mt-auto border-t border-[var(--pt-border)] bg-[var(--pt-surface)]">
   342→        <div className="mx-auto w-full max-w-6xl px-4 py-8 md:px-8">
   343→          <div className="flex flex-col items-center justify-between gap-4 sm:flex-row">

   347→                  src={logo}
   348→                  alt={nav.brand}
   349→                  loading="lazy"
   350→                  decoding="async"
   351→                  className="h-7 w-auto max-w-[100px] object-contain"
   352→                />
   353→              ) : (
   354→                <>
   355→                  <span className="size-6 rounded-[calc(var(--pt-radius)-4px)]" style={{ background: 'var(--pt-primary)' }} />
   356→                  <span className="text-sm font-bold text-[var(--pt-text)]">{nav.brand}</span>
   357→                </>
   358→              )}
   359→              <PBadge tone="primary">بيانات تجريبية</PBadge>
   360→            </div>
   361→            <nav aria-label="تنقل التذييل" className="flex flex-wrap items-center justify-center gap-x-4 gap-y-2">
   362→              {nav.items.map((item) => (
   363→                <button
   364→                  key={`f-${item.pageId}`}
   365→                  type="button"
   366→                  onClick={() => go(item.pageId)}
   367→                  className="min-h-10 cursor-pointer text-[13px] text-[var(--pt-muted)] transition-colors duration-150 hover:text-[var(--pt-primary)] focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-[var(--pt-primary)]"
   368→                >
   369→                  {item.label}
   370→                </button>
   371→              ))}
   372→            </nav>
   373→          </div>
   374→          {nav.footerNote ? (
   375→            <div className="mt-4 flex flex-col items-center gap-2 border-t border-[var(--pt-border)] pt-4 text-center">
   376→              {logo ? (
   377→                <img
   378→                  src={logo}
   379→                  alt=""
   380→                  loading="lazy"
   381→                  decoding="async"
   382→                  className="h-6 w-auto max-w-[90px] object-contain opacity-80"
   383→                />
   384→              ) : null}
   385→              <p className="text-xs text-[var(--pt-muted)]">{nav.footerNote}</p>
   386→            </div>
   387→          ) : null}
   388→        </div>
   389→      </footer>
   390→
   391→      {/* زر العودة للخلف — كل صفحة غير الرئيسية */}

   393→        <FloatingBackButton label="عودة" onBack={() => go(homeId)} />
   394→      ) : null}
   395→    </div>
   396→  )
   397→}

   673→  schema,

   740→        <nav aria-label="تنقل لوحة التحكم" className="flex-1 space-y-1 overflow-y-auto p-3">

   832→// PrototypeRenderer — العقد المثبّت

   834→export function PrototypeRenderer({
   835→  schema,
   836→  pageId,
   837→  selectedBlockId,
   838→  onBlockSelect,
   839→  onNavigate,
   840→  interactive,
   841→  mini,
   842→}: {
   843→  schema: ProjectSchema
   844→  pageId: string
   845→  selectedBlockId?: string | null
   846→  /** وضع التحرير: أغلفة تحديد حول كل مكوّن */
   847→  onBlockSelect?: (blockId: string | null) => void
   848→  /** روابط التنقل بين صفحات النموذج */
   849→  onNavigate?: (pageId: string) => void
   850→  /** وضع المعاينة: النماذج والأزرار حيّة */
   851→  interactive?: boolean
   852→  /** معاينة مصغّرة — بديل خفيف لأقسام HTML المخصصة */
   853→  mini?: boolean
   854→}): ReactElement {
   855→  const page =
   856→    schema.pages.find((pg) => pg.id === pageId) ??
   857→    schema.pages.find((pg) => pg.slug === pageId) ??
   858→    schema.pages[0]
   859→
   860→  const ctx: BlockCtx = {
   861→    schema,
   862→    interactive: interactive ?? false,
   863→    mini,
   864→    onNavigate,
   865→  }
   866→
   867→  const shellProps = { schema, page, ctx, selectedBlockId, onBlockSelect }
   868→
   869→  return (
   870→    <PrototypeThemeScope theme={schema.theme} className="w-full">
   871→      {!page ? (
   872→        <div className="flex min-h-96 items-center justify-center bg-[var(--pt-bg)] p-8 text-center text-sm text-[var(--pt-muted)]">
   873→          لا توجد صفحات في هذا النموذج بعد.
   874→        </div>
   875→      ) : page.type === 'auth' ? (
   876→        <AuthShell {...shellProps} />
   877→      ) : page.type === 'dashboard' || page.type === 'table' || page.type === 'form' || page.type === 'settings' || page.type === 'profile' ? (
   878→        <DashboardShell {...shellProps} />
   879→      ) : (
   880→        <SiteShell {...shellProps} />
   881→      )}
   882→    </PrototypeThemeScope>
   883→  )
   884→}
