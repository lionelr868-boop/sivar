   121→export function PagesPanel({ onNavigated }: { onNavigated?: () => void }) {
   122→  const schema = useAppStore((s) => s.schema)
   123→  const selectedPageId = useAppStore((s) => s.selectedPageId)
   124→  const selectedBlockId = useAppStore((s) => s.selectedBlockId)
   125→  const selectPage = useAppStore((s) => s.selectPage)
   126→  const selectBlock = useAppStore((s) => s.selectBlock)
   127→  const applySchemaFn = useAppStore((s) => s.applySchemaFn)
   128→
   129→  const [addOpen, setAddOpen] = useState(false)
   130→  const [styleOpen, setStyleOpen] = useState(false)
   131→  const [renameTarget, setRenameTarget] = useState<string | null>(null)
   132→  const [renameDraft, setRenameDraft] = useState('')
   133→  const [deleteTarget, setDeleteTarget] = useState<string | null>(null)
   134→
   135→  const page = useMemo(() => schema?.pages.find((p) => p.id === selectedPageId) ?? null, [schema, selectedPageId])
   136→  const entityKeys = useMemo(() => (schema ? Object.keys(schema.mockData.entities) : []), [schema])
   137→
   138→  if (!schema) return null
   139→
   140→  const handleAddBlock = (blockType: BlockType) => {
   141→    if (!page) return
   142→    try {
   143→      let newId = ''
   144→      applySchemaFn((s) => {
   145→        const res = addBlockToPage(s, page.id, blockType)
   146→        newId = res.blockId
   147→        return res.schema
   148→      })
   149→      selectBlock(newId)
   150→      toast.success(t.builder.blockAddedToast.replace('{name}', BLOCK_REGISTRY[blockType].label))
   151→    } catch (e) {
   152→      toast.error((e as Error).message)
   153→    }
   154→  }
   155→
   156→  const commitRename = () => {
   157→    if (!renameTarget) return
   158→    const name = renameDraft.trim()
   159→    if (name) {
   160→      // (الجولة 13) إعادة التسمية من المحرر تُزامن عنوان الشريط الجانبي وقائمة الموقع تلقائياً
   161→      applySchemaFn((s) =>
   162→        syncPageNavigation(
   163→          {
   164→            ...s,
   165→            pages: s.pages.map((p) => (p.id === renameTarget ? { ...p, name, title: p.title ?? name } : p)),
   166→          },
   167→          renameTarget
   168→        )
   169→      )
   170→    }
   171→    setRenameTarget(null)
   172→  }
   173→
   174→  return (
   175→    <div className="flex h-full min-h-0 flex-col">
   176→      {/* الصفحات */}
   177→      <div className="flex items-center justify-between px-3 pb-1.5 pt-3">
   178→        <h3 className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">{t.builder.pagesPanel}</h3>
   179→        <Button variant="ghost" size="sm" className="h-7 gap-1 px-2 text-xs" onClick={() => setAddOpen(true)}>
   180→          <Plus className="size-3.5" />
   181→          {t.builder.addPage}
   182→        </Button>
   183→      </div>
   184→
   185→      <ul className="max-h-64 overflow-y-auto px-2 pb-2">
   186→        {schema.pages.map((p) => {
   187→          const Icon = PAGE_ICONS[p.type] ?? FileText
   188→          const isActive = p.id === selectedPageId
   189→          return (
   190→            <li key={p.id} className="group/page">
   191→              {renameTarget === p.id ? (
   192→                <Input
   193→                  autoFocus
   194→                  value={renameDraft}
   195→                  onChange={(e) => setRenameDraft(e.target.value)}
   196→                  onBlur={commitRename}
   197→                  onKeyDown={(e) => {
   198→                    if (e.key === 'Enter') commitRename()

   200→                  }}
   201→                  className="h-9 text-sm"
   202→                  aria-label={t.builder.renamePage}
   203→                />
   204→              ) : (
   205→                <div
   206→                  className={cn(
   207→                    'flex items-center gap-2 rounded-lg px-2 py-1.5 text-sm transition-colors',
   208→                    isActive ? 'bg-secondary font-medium text-secondary-foreground' : 'text-muted-foreground hover:bg-accent/60 hover:text-foreground'
   209→                  )}
   210→                >
   211→                  <button
   212→                    type="button"
   213→                    className="flex min-w-0 flex-1 items-center gap-2 text-start"
   214→                    onClick={() => {
   215→                      selectPage(p.id)
   216→                      onNavigated?.()
   217→                    }}
   218→                  >
   219→                    <Icon className="size-4 shrink-0" />
   220→                    <span className="truncate">{p.name}</span>
   221→                  </button>
   222→                  <button
   223→                    type="button"
   224→                    className="hidden rounded p-1 text-muted-foreground/70 hover:text-foreground group-hover/page:block"
   225→                    onClick={() => {
   226→                      setRenameTarget(p.id)
   227→                      setRenameDraft(p.name)
   228→                    }}
   229→                    aria-label={t.builder.renamePage}
   230→                  >
   231→                    <FileText className="size-3.5" />
   232→                  </button>
   233→                  <button
   234→                    type="button"
   235→                    className="hidden rounded p-1 text-muted-foreground/70 hover:text-destructive group-hover/page:block disabled:opacity-30"
   236→                    onClick={() => setDeleteTarget(p.id)}
   237→                    disabled={schema.pages.length <= 1}
   238→                    aria-label={t.builder.deletePage}
   239→                  >
   240→                    <Trash2 className="size-3.5" />
   241→                  </button>
   242→                </div>
   243→              )}
   244→            </li>
   245→          )
   246→        })}
   247→      </ul>
   248→
   249→      {/* مكوّنات الصفحة الحالية */}
   250→      <div className="mt-2 flex min-h-0 flex-1 flex-col border-t">
   251→        <div className="flex flex-wrap items-center justify-between gap-1.5 px-3 pb-1.5 pt-2.5">
   252→          <h3 className="min-w-0 flex-1 truncate text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   253→            {t.builder.componentsPanel}
   254→          </h3>
   255→          <div className="flex shrink-0 items-center gap-1.5">
   256→            {page?.type === 'landing' ? (
   257→              <Button
   258→                variant="outline"
   259→                size="sm"
   260→                className="h-7 gap-1 px-2 text-xs"
   261→                onClick={() => setStyleOpen(true)}
   262→                title={t.builder.changeStyleBtn}
   263→              >
   264→                <Wand2 className="size-3.5 text-primary" />
   265→                {t.builder.changeStyleBtn}
   266→              </Button>
   267→            ) : null}
   268→            <DropdownMenu>
   269→              <DropdownMenuTrigger asChild>
   270→                <Button variant="outline" size="sm" className="h-7 gap-1 px-2 text-xs">
   271→                  <Plus className="size-3.5" />
   272→                  {t.builder.addBlock}
   273→                </Button>
   274→              </DropdownMenuTrigger>
   275→              <DropdownMenuContent side="left" align="start" className="w-60 max-h-96 overflow-y-auto">
   276→                {(
   277→                  [
   278→                    { key: 'marketing', label: t.builder.marketingBlocks },
   279→                    { key: 'dashboard', label: t.builder.dashboardBlocks },
   280→                    { key: 'auth', label: t.builder.authBlocks },
   281→                    { key: 'states', label: t.builder.stateBlocks },
   282→                  ] as const
   283→                ).map((group) => {
   284→                  const defs = Object.values(BLOCK_REGISTRY).filter((d) => d.category === group.key)
   285→                  if (!defs.length) return null
   286→                  return (
   287→                    <DropdownMenuLabel key={group.key}>{group.label}</DropdownMenuLabel>
   288→                  )
   289→                })}
   290→                <DropdownMenuSeparator />
   291→                {Object.values(BLOCK_REGISTRY)
   292→                  .sort((a, b) => a.category.localeCompare(b.category))
   293→                  .map((def) => (
   294→                    <DropdownMenuItem key={def.type} onClick={() => handleAddBlock(def.type)} className="gap-2">
   295→                      <span className="text-xs font-semibold text-primary">{def.label}</span>
   296→                      <span className="truncate text-xs text-muted-foreground">{def.description}</span>
   297→                    </DropdownMenuItem>
   298→                  ))}
   299→              </DropdownMenuContent>
   300→            </DropdownMenu>
   301→          </div>
   302→        </div>
   303→
   304→        {page && page.blocks.length > 0 ? (
   305→          <ul className="min-h-0 flex-1 overflow-y-auto px-2 pb-3">
   306→            {page.blocks.map((block, idx) => {
   307→              const def = BLOCK_REGISTRY[block.type]
   308→              const isSel = block.id === selectedBlockId
   309→              return (
   310→                <li
   311→                  key={block.id}
   312→                  className={cn(
   313→                    'group/blk mb-1 flex items-center gap-1 rounded-lg border px-2 py-1.5 text-sm transition-colors',
   314→                    isSel ? 'border-primary/50 bg-secondary/70 font-medium' : 'border-transparent text-muted-foreground hover:border-border hover:bg-accent/40'
   315→                  )}
   316→                >
   317→                  <button
   318→                    type="button"
   319→                    className="min-w-0 flex-1 truncate text-start"

   321→                    title={def?.description}
   322→                  >
   323→                    {def?.label ?? block.type}
   324→                  </button>
   325→                  <div className="flex shrink-0 items-center">
   326→                    <button
   327→                      type="button"
   328→                      className="rounded p-0.5 text-muted-foreground/60 hover:text-foreground disabled:opacity-25"
   329→                      disabled={idx === 0}
   330→                      onClick={() =>
   331→                        applySchemaFn((s) => ({
   332→                          ...s,
   333→                          pages: s.pages.map((pg) =>
   334→                            pg.id === page.id
   335→                              ? {
   336→                                  ...pg,
   337→                                  blocks: (() => {
   338→                                    const blocks = [...pg.blocks]
   339→                                    const [moved] = blocks.splice(idx, 1)
   340→                                    blocks.splice(idx - 1, 0, moved)
   341→                                    return blocks
   342→                                  })(),
   343→                                }
   344→                              : pg
   345→                          ),
   346→                        }))
   347→                      }
   348→                      aria-label={t.builder.moveUp}
   349→                    >
   350→                      <ChevronUp className="size-3.5" />
   351→                    </button>
   352→                    <button
   353→                      type="button"
   354→                      className="rounded p-0.5 text-muted-foreground/60 hover:text-foreground disabled:opacity-25"
   355→                      disabled={idx === page.blocks.length - 1}
   356→                      onClick={() =>
   357→                        applySchemaFn((s) => ({
   358→                          ...s,
   359→                          pages: s.pages.map((pg) =>
   360→                            pg.id === page.id
   361→                              ? {
   362→                                  ...pg,
   363→                                  blocks: (() => {
   364→                                    const blocks = [...pg.blocks]
   365→                                    const [moved] = blocks.splice(idx, 1)
   366→                                    blocks.splice(idx + 1, 0, moved)
   367→                                    return blocks
   368→                                  })(),
   369→                                }
   370→                              : pg
   371→                          ),
   372→                        }))
   373→                      }
   374→                      aria-label={t.builder.moveDown}
   375→                    >
   376→                      <ChevronDown className="size-3.5" />
   377→                    </button>
   378→                    <button
   379→                      type="button"
   380→                      className="rounded p-0.5 text-muted-foreground/60 hover:text-destructive"
   381→                      onClick={() => {
   382→                        applySchemaFn((s) => ({
   383→                          ...s,
   384→                          pages: s.pages.map((pg) => (pg.id === page.id ? { ...pg, blocks: pg.blocks.filter((b) => b.id !== block.id) } : pg)),
   385→                        }))
   386→                        if (isSel) selectBlock(null)
   387→                      }}
   388→                      aria-label={t.builder.deleteBlock}
   389→                    >
   390→                      <Trash2 className="size-3.5" />
   391→                    </button>
   392→                  </div>
   393→                </li>
   394→              )
   395→            })}
   396→          </ul>
   397→        ) : (
   398→          <div className="mx-2 mb-3 rounded-lg border border-dashed p-4 text-center">
   399→            <p className="text-sm font-medium text-muted-foreground">{t.builder.noBlocks}</p>
   400→            <p className="mt-1 text-xs text-muted-foreground/70">{t.builder.noBlocksHint}</p>
   401→          </div>
   402→        )}
   403→      </div>
   404→
   405→      {/* نافذة إضافة صفحة */}
   406→      <AddPageDialog
   407→        open={addOpen}
   408→        onOpenChange={setAddOpen}
   409→        entityKeys={entityKeys}
   410→        onCreate={(opts) => {
   411→          let pageId = ''
   412→          let inSidebar = false
   413→          applySchemaFn((s) => {
   414→            const res = addPageToSchema(s, {
   415→              name: opts.name,
   416→              type: opts.type,
   417→              authVariant: opts.authVariant,
   418→              entity: opts.entity,
   419→              landingVariant: opts.landingVariant,
   420→            })
   421→            pageId = res.pageId
   422→            // (الجولة 13) هل أُدرجت الصفحة الجديدة في الشريط الجانبي للوحة التحكم؟
   423→            inSidebar = res.schema.dashboardNav.items.some((it) => it.pageId === res.pageId)
   424→            return res.schema
   425→          })
   426→          if (pageId) {
   427→            selectPage(pageId)
   428→            const toastMsg = inSidebar
   429→              ? t.builder.pageAddedSidebarToast.replace('{name}', opts.name)
   430→              : t.builder.pageAddedToast.replace('{name}', opts.name)
   431→            toast.success(toastMsg)
   432→            onNavigated?.()
   433→          }
   434→        }}
   435→        onCreateHtml={(opts) => {
   436→          let pageId = ''
   437→          let inSidebar = false
   438→          applySchemaFn((s) => {
   439→            const res = addHtmlPageToSchema(s, { name: opts.name, html: opts.html, placement: opts.placement, height: 640 })
   440→            pageId = res.pageId
   441→            // (الجولة 13) صفحة HTML داخل اللوحة تُدرج في الشريط الجانبي أيضاً
   442→            inSidebar = res.schema.dashboardNav.items.some((it) => it.pageId === res.pageId)
   443→            return res.schema
   444→          })
   445→          if (pageId) {
   446→            selectPage(pageId)
   447→            const toastMsg = inSidebar
   448→              ? t.builder.pageAddedSidebarToast.replace('{name}', opts.name)
   449→              : t.builder.pageAddedToast.replace('{name}', opts.name)
   450→            toast.success(toastMsg)
   451→            onNavigated?.()
   452→            // حفظ في مكتبة HTML — غير حاجز (لا يمنع إنشاء الصفحة)
   453→            if (opts.saveToLibrary && opts.snippetName) {
   454→              api
   455→                .createSnippet({
   456→                  name: opts.snippetName,
   457→                  kind: opts.placement === 'dashboard' ? 'dashboard' : 'landing',
   458→                  html: opts.html,
   459→                })
   460→                .then(() => toast.success(t.builder.htmlSavedToLibrary))
   461→                .catch((e) => toast.error((e as Error).message || t.builder.htmlSaveLibraryError))
   462→            }
   463→          }
   464→        }}
   465→      />
   466→
   467→      {/* نافذة تغيير نمط صفحة الهبوط */}
   468→      {page && page.type === 'landing' ? (
   469→        <VariantStyleDialog page={page} open={styleOpen} onOpenChange={setStyleOpen} />
   470→      ) : null}
   471→
   472→      {/* تأكيد حذف الصفحة */}
   473→      <AlertDialog open={!!deleteTarget} onOpenChange={(o) => !o && setDeleteTarget(null)}>
   474→        <AlertDialogContent>
   475→          <AlertDialogHeader>
   476→            <AlertDialogTitle>{t.builder.deletePageTitle}</AlertDialogTitle>
   477→            <AlertDialogDescription>
   478→              {t.builder.deletePageDesc.replace('{name}', schema.pages.find((p) => p.id === deleteTarget)?.name ?? '')}
   479→            </AlertDialogDescription>
   480→          </AlertDialogHeader>
   481→          <AlertDialogFooter>
   482→            <AlertDialogCancel>{t.common.cancel}</AlertDialogCancel>
   483→            <AlertDialogAction
   484→              className="bg-destructive text-white hover:bg-destructive/90"
   485→              onClick={() => {
   486→                if (deleteTarget) {
   487→                  applySchemaFn((s) => deletePageFromSchema(s, deleteTarget))
   488→                  if (deleteTarget === selectedPageId) {
   489→                    const next = schema.pages.find((p) => p.id !== deleteTarget)
   490→                    if (next) selectPage(next.id)
   491→                  }
   492→                }
   493→                setDeleteTarget(null)
   494→              }}
   495→            >
   496→              {t.common.delete}
   497→            </AlertDialogAction>
   498→          </AlertDialogFooter>
   499→        </AlertDialogContent>
   500→      </AlertDialog>
   501→    </div>
   502→  )
   503→}

   509→  variant,

   514→  selected: boolean
   515→  onSelect: () => void

   517→  return (
   518→    <button

   520→      onClick={onSelect}

   629→function AddPageDialog({
   630→  open,
   631→  onOpenChange,
   632→  entityKeys,
   633→  onCreate,
   634→  onCreateHtml,
   635→}: {
   636→  open: boolean
   637→  onOpenChange: (o: boolean) => void
   638→  entityKeys: string[]
   639→  onCreate: (opts: { name: string; type: PageType; authVariant?: AuthVariant; entity?: string; landingVariant?: LandingVariantId }) => void
   640→  onCreateHtml: (opts: { name: string; html: string; placement: HtmlPlacement; saveToLibrary: boolean; snippetName: string }) => void
   641→}) {
   642→  const [type, setType] = useState<PageType>('landing')
   643→  const [authVariant, setAuthVariant] = useState<AuthVariant>('login')
   644→  const [name, setName] = useState('')
   645→  const [entity, setEntity] = useState<string>('')

   651→  const [html, setHtml] = useState('')
   652→  const [fileName, setFileName] = useState<string | null>(null)

   655→  const [saveToLibrary, setSaveToLibrary] = useState(false)

   767→          <DialogDescription>{t.builder.addPageSubtitle}</DialogDescription>

   775→            const Icon = opt.icon
   776→            return (
   777→              <button
   778→                key={opt.kind === 'html' ? 'html' : `${opt.type}-${opt.authVariant ?? 'x'}`}
   779→                type="button"
   780→                onClick={() => {
   781→                  if (opt.kind === 'html') {
   782→                    setIsHtml(true)
   783→                  } else {
   784→                    setIsHtml(false)
   785→                    setType(opt.type)
   786→                    if (opt.authVariant) setAuthVariant(opt.authVariant)
   787→                  }
   788→                  setName('')
   789→                }}
   790→                aria-pressed={isActive}
   791→                className={cn(
   792→                  'flex flex-col items-center gap-1.5 rounded-lg border p-3 text-center transition-colors',
   793→                  isActive ? 'border-primary bg-secondary/70' : 'border-border hover:bg-accent/50'
   794→                )}
   795→              >
   796→                <Icon className={cn('size-5', isActive ? 'text-primary' : 'text-muted-foreground')} />
   797→                <span className="text-sm font-medium">{opt.label}</span>
   798→                <span className="text-[11px] leading-tight text-muted-foreground">{opt.desc}</span>
   799→              </button>
   800→            )
   801→          })}
   802→        </div>
   803→
   804→        {isHtml ? (
   805→          /* ===== لوحة صفحة HTML ===== */
   806→          <div className="space-y-3">
   807→            <div className="space-y-1.5">
   808→              <Label htmlFor="new-page-name">{t.builder.pageName}</Label>
   809→              <Input id="new-page-name" value={name} placeholder={defaultName} onChange={(e) => setName(e.target.value)} />
   810→            </div>
   811→
   812→            <div className="space-y-1.5">
   813→              <Label className="text-xs">{t.builder.htmlSourceLabel}</Label>
   814→              <Tabs value={htmlTab} onValueChange={(v) => setHtmlTab(v as HtmlSourceTab)}>
   815→                <TabsList className="grid w-full grid-cols-3">
   816→                  <TabsTrigger value="paste" className="text-xs">{t.builder.htmlPasteTab}</TabsTrigger>
   817→                  <TabsTrigger value="upload" className="text-xs">{t.builder.htmlUploadTab}</TabsTrigger>
   818→                  <TabsTrigger value="library" className="text-xs">{t.builder.htmlLibraryTab}</TabsTrigger>
   819→                </TabsList>
   820→
   821→                {/* لصق الكود */}
   822→                <TabsContent value="paste" className="mt-2">
   823→                  <Textarea

   827→                      setSelectedSnippetId(null)
   828→                    }}
   829→                    dir="ltr"
   830→                    rows={8}
   831→                    spellCheck={false}
   832→                    placeholder={t.builder.htmlPlaceholder}
