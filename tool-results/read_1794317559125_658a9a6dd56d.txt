    58→/** مزج لونين محلياً — ratio = وزن اللون الثاني (0..1) */
    59→function ptMix(a: string, b: string, ratio: number): string {
    60→  const expand = (hex: string): [number, number, number] | null => {
    61→    const m = /^#([0-9a-f]{3}|[0-9a-f]{6})$/i.exec(hex.trim())
    62→    if (!m) return null
    63→    let h = m[1]
    64→    if (h.length === 3) h = h.split('').map((c) => c + c).join('')
    65→    return [parseInt(h.slice(0, 2), 16), parseInt(h.slice(2, 4), 16), parseInt(h.slice(4, 6), 16)]
    66→  }
    67→  const ca = expand(a)
    68→  const cb = expand(b)
    69→  if (!ca || !cb) return a
    70→  const t = Math.max(0, Math.min(1, ratio))
    71→  const clamp = (v: number) => Math.max(0, Math.min(255, Math.round(v)))
    72→  const mixed = [0, 1, 2].map((i) => clamp(ca[i] + (cb[i] - ca[i]) * t))
    73→  return `#${mixed.map((v) => v.toString(16).padStart(2, '0')).join('')}`
    74→}

    76→/**
    77→ * متغيرات CSS لألوان قسم واحد (bgColor/textColor اختياريان):
    78→ * - bgColor: خلفية + --pt-bg + --pt-surface/--pt-border مشتقان حسب السطوع.
    79→ * - textColor: --pt-text + --pt-muted ممزوج نحو الخلفية 0.38.
    80→ * - نص تلقائي ذكي: مع bgColor بلا textColor → نص داكن على الخلفية
    81→ *   الفاتحة (سطوع > 0.5) وأبيض على الداكنة (+ ثانوي مشتق).
    82→ * تُرجع null عندما لا ألوان مخصّصة (سلوك قديم مطابق بالبايت).
    83→ */
    84→export function sectionVars(bgColor: unknown, textColor: unknown): CSSProperties | null {
    85→  const bg = sectionHexColor(bgColor)
    86→  const tx = sectionHexColor(textColor)
    87→  if (!bg && !tx) return null
    88→
    89→  const vars: Record<string, string> = {}
    90→  if (bg) {
    91→    const bright = hexLuminance(bg) > 0.55
    92→    vars['--pt-bg'] = bg
    93→    vars['--pt-surface'] = bright ? hexLighten(bg, 0.42) : hexDarken(bg, 0.42)
    94→    vars['--pt-border'] = bright ? hexLighten(bg, 0.18) : hexDarken(bg, 0.18)
    95→    vars.background = bg
    96→  }
    97→
    98→  let text = tx
    99→  if (bg && !tx) text = hexLuminance(bg) > 0.5 ? '#17151f' : '#ffffff'
   100→  if (text) {
   101→    vars['--pt-text'] = text
   102→    vars['--pt-muted'] = ptMix(text, bg || '#ffffff', 0.38)
   103→  }
   104→  vars.color = 'var(--pt-text)'
   105→
   106→  return vars as CSSProperties
   107→}

   193→/** قائمة المكوّنات مع أغلفة وضع التحرير (outline) عند توفّر onBlockSelect */
   194→function BlockList({
   195→  page,
   196→  ctx,
   197→  selectedBlockId,
   198→  onBlockSelect,
   199→}: {
   200→  page: ProjectPage
   201→  ctx: BlockCtx
   202→  selectedBlockId?: string | null
   203→  onBlockSelect?: (blockId: string | null) => void
   204→}): ReactElement {
   205→  return (
   206→    <>
   207→      {page.blocks.map((block, index) => {
   208→        const label = BLOCK_REGISTRY[block.type]?.label ?? block.type
   209→        const selected = selectedBlockId === block.id
   210→        const dimmed = selectedBlockId != null && selectedBlockId !== block.id
   211→
   212→        if (!onBlockSelect) {
   213→          // حركة دخول متدرّجة في وضع المعاينة التفاعلي فقط (بلا وضع التحرير)
   214→          if (ctx.interactive) {
   215→            return (
   216→              <div key={block.id} className="pt-rise" style={{ animationDelay: `${Math.min(index, 6) * 60}ms` }}>
   217→                {renderBlock(block, ctx)}
   218→              </div>
   219→            )
   220→          }
   221→          return <div key={block.id}>{renderBlock(block, ctx)}</div>
   222→        }
   223→
   224→        return (
   225→          <div
   226→            key={block.id}
   227→            role="button"
   228→            tabIndex={0}
   229→            aria-pressed={selected}
   230→            onClick={(e) => {
   231→              e.stopPropagation()
   232→              onBlockSelect(selected ? null : block.id)
   233→            }}
   234→            onKeyDown={(e) => {
   235→              if (e.key === 'Enter' || e.key === ' ') {
   236→                e.preventDefault()
   237→                e.stopPropagation()
   238→                onBlockSelect(selected ? null : block.id)
   239→              }
   240→            }}
   241→            className={`relative group/block cursor-pointer transition-opacity duration-200 ${dimmed ? 'opacity-55' : ''}`}
   242→          >
   243→            {/* المحتوى معطّل المؤشر في وضع التحرير */}
   244→            <div className="pointer-events-none">{renderBlock(block, ctx)}</div>
   245→
   246→            {/* إطار التحديد */}
   247→            <span
   248→              aria-hidden
   249→              className={`pointer-events-none absolute inset-0 z-[6] rounded-[var(--pt-radius)] border-2 transition-opacity duration-150 ${
   250→                selected ? 'opacity-100' : 'opacity-0 group-hover/block:opacity-100'
   251→              }`}
   252→              style={{ borderColor: 'var(--pt-primary)' }}
   253→            />
   254→
   255→            {/* شارة اسم المكوّن */}
   256→            <span
   257→              className={`pointer-events-none absolute top-1 start-1 z-[7] rounded px-2 py-0.5 text-[11px] font-bold text-white transition-opacity duration-150 ${
   258→                selected ? 'opacity-100' : 'opacity-0 group-hover/block:opacity-100'
   259→              }`}
   260→              style={{ background: 'var(--pt-primary)' }}
   261→            >
   262→              {label}
   263→            </span>
   264→          </div>
   265→        )
   266→      })}
   267→    </>
   268→  )
   269→}

   293→function SiteShell({
   294→  schema,
   295→  page,
   296→  ctx,
   297→  selectedBlockId,
   298→  onBlockSelect,
   299→}: {
   300→  schema: ProjectSchema
   301→  page: ProjectPage
   302→  ctx: BlockCtx

   304→  onBlockSelect?: (blockId: string | null) => void

   308→  const logo = ctx.schema.branding?.logo

   311→  const homeId = schema.pages.find((pg) => pg.type === 'landing')?.id ?? schema.pages[0]?.id

   313→    if (ctx.onNavigate) ctx.onNavigate(pageId)

   323→            key={item.pageId}
   324→            type="button"
   325→            onClick={() => {

   327→              setMenuOpen(false)

   357→                className="h-8 w-auto max-w-[120px] object-contain md:h-9"
