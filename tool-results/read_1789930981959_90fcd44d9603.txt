    25→/** تهريب HTML لكل نص قادم من المستخدم */
    26→function esc(value: unknown): string {
    27→  return String(value ?? '')
    28→    .replace(/&/g, '&amp;')
    29→    .replace(/</g, '&lt;')
    30→    .replace(/>/g, '&gt;')
    31→    .replace(/"/g, '&quot;')
    32→    .replace(/'/g, '&#39;')
    33→}

    35→/** تنسيق الأرقام بالأسلوب الجزائري (فاصلة عشرية) مع بديل آمن */
    36→function fmtNum(v: number): string {
    37→  if (!Number.isFinite(v)) return '0'
    38→  try {
    39→    const s = v.toLocaleString('ar-DZ')
    40→    return /^[\d\s.,'’]+$/.test(s) ? s : v.toLocaleString('en-US')
    41→  } catch {
    42→    return v.toLocaleString('en-US')
    43→  }
    44→}

    46→/** اختصار رقمي للمحاور (3.4م / 12ألف) — مطابق لعارض React */
    47→function compactNum(v: number): string {
    48→  if (Math.abs(v) >= 1_000_000) {
    49→    const m = v / 1_000_000
    50→    return `${Number.isInteger(m) ? m : m.toFixed(1)}م`
    51→  }
    52→  if (Math.abs(v) >= 1000) {
    53→    const k = v / 1000
    54→    return `${Number.isInteger(k) ? k : k.toFixed(1)}ألف`
    55→  }
    56→  return String(v)
    57→}

    59→/** قصّ النص لحدّ أقصى من المحارف (آمن مع أزواج UTF-16) للوصف والوسوم */
    60→function truncateText(value: string, max: number): string {
    61→  const t = (value ?? '').trim()
    62→  const chars = Array.from(t)
    63→  if (chars.length <= max) return t
    64→  return `${chars.slice(0, Math.max(0, max - 1)).join('')}…`
    65→}

    78→/** تحويل data URL لصورة إلى { buffer, ext, mime } — null إن لم تكن صالحة */
    79→export function dataUrlToAsset(dataUrl: string): { buffer: Uint8Array; ext: string; mime: string } | null {
    80→  const raw = typeof dataUrl === 'string' ? dataUrl.trim() : ''
    81→  if (!/^data:image\//i.test(raw)) return null
    82→  const comma = raw.indexOf(',')
    83→  if (comma < 0) return null
    84→  const header = raw.slice(5, comma)
    85→  const payload = raw.slice(comma + 1)
    86→  const mime = (header.split(';')[0] ?? '').toLowerCase()
    87→  let ext: string | undefined
    88→  if (mime === 'image/png') ext = 'png'
    89→  else if (mime === 'image/jpeg' || mime === 'image/jpg') ext = 'jpg'
    90→  else if (mime === 'image/webp') ext = 'webp'
    91→  else if (mime === 'image/gif') ext = 'gif'
    92→  else if (mime === 'image/svg+xml') ext = 'svg'
    93→  if (!ext) return null
    94→  const isBase64 = /;base64$/i.test(header)
    95→  let buffer: Uint8Array
    96→  try {
    97→    buffer = isBase64
    98→      ? Buffer.from(payload, 'base64')
    99→      : Buffer.from(decodeURIComponent(payload), 'utf8')
   100→  } catch {
   101→    return null
   102→  }
   103→  if (buffer.byteLength === 0) return null
   104→  return { buffer, ext, mime: LOGO_MIME_BY_EXT[ext] ?? 'image/png' }
   105→}

   107→// قُرّاء خصائص آمنة (مطابقة لعارض React)
   108→function S(props: Record<string, unknown>, key: string, fallback = ''): string {
   109→  const v = props?.[key]
   110→  return typeof v === 'string' ? v : typeof v === 'number' && Number.isFinite(v) ? String(v) : fallback
   111→}
   112→function N(props: Record<string, unknown>, key: string, fallback = 0): number {
   113→  const v = props?.[key]
   114→  if (typeof v === 'number' && Number.isFinite(v)) return v
   115→  if (typeof v === 'string' && v.trim() !== '' && Number.isFinite(Number(v))) return Number(v)
   116→  return fallback
   117→}
   118→function B(props: Record<string, unknown>, key: string, fallback = false): boolean {
   119→  const v = props?.[key]
   120→  if (typeof v === 'boolean') return v
   121→  if (v === 'true') return true
   122→  if (v === 'false') return false
   123→  return fallback
   124→}
   125→function L(props: Record<string, unknown>, key: string): Array<Record<string, unknown>> {
   126→  const v = props?.[key]
   127→  if (!Array.isArray(v)) return []
   128→  return v.filter((it): it is Record<string, unknown> => !!it && typeof it === 'object' && !Array.isArray(it))
   129→}
   130→function iS(item: Record<string, unknown>, key: string, fallback = ''): string {
   131→  return S(item, key, fallback)
   132→}
   133→function iN(item: Record<string, unknown>, key: string, fallback = 0): number {
   134→  return N(item, key, fallback)
   135→}
   136→function iB(item: Record<string, unknown>, key: string, fallback = false): boolean {
   137→  return B(item, key, fallback)
   138→}

   148→function hashName(name: string): number {
   149→  let h = 0
   150→  for (let i = 0; i < name.length; i++) {
   151→    h = (h * 31 + name.charCodeAt(i)) >>> 0
   152→  }
   153→  return h
   154→}

   156→function initialsOf(name: string): string {
   157→  const parts = name.trim().split(/\s+/).filter(Boolean)
   158→  if (parts.length === 0) return '؟'
   159→  if (parts.length === 1) return parts[0].slice(0, 2)
   160→  return (parts[0][0] ?? '') + (parts[1][0] ?? '')
   161→}

   189→function parseHex(hex: string): [number, number, number] | null {
   190→  const m = /^#([0-9a-f]{3}|[0-9a-f]{6})$/i.exec(hex.trim())
   191→  if (!m) return null
   192→  let h = m[1]
   193→  if (h.length === 3) h = h.split('').map((c) => c + c).join('')
   194→  return [parseInt(h.slice(0, 2), 16), parseInt(h.slice(2, 4), 16), parseInt(h.slice(4, 6), 16)]
   195→}
   196→function toHex(r: number, g: number, b: number): string {
   197→  const clamp = (v: number) => Math.max(0, Math.min(255, Math.round(v)))
   198→  return `#${[r, g, b].map((v) => clamp(v).toString(16).padStart(2, '0')).join('')}`
   199→}
   200→function mixHex(a: string, b: string, ratio: number): string {
   201→  const ca = parseHex(a)
   202→  const cb = parseHex(b)
   203→  if (!ca || !cb) return a
   204→  const t = Math.max(0, Math.min(1, ratio))
   205→  return toHex(ca[0] + (cb[0] - ca[0]) * t, ca[1] + (cb[1] - ca[1]) * t, ca[2] + (cb[2] - ca[2]) * t)
   206→}
   207→function hexLighten(hex: string, ratio: number): string {
   208→  return mixHex(hex, '#ffffff', ratio)
   209→}
   210→function hexDarken(hex: string, ratio: number): string {
   211→  return mixHex(hex, '#000000', ratio)
   212→}

   214→const SHADOW_CSS: Record<string, string> = {
   215→  none: 'none',
   216→  sm: '0 1px 2px rgba(0,0,0,.08)',
   217→  md: '0 4px 12px rgba(0,0,0,.08)',

   221→function themeVarsCss(theme: ProjectTheme): string {
   222→  const dark = theme.mode === 'dark'
   223→  const bg = dark ? hexDarken(theme.background, 0.72) : theme.background
   224→  const surface = dark ? hexLighten(bg, 0.09) : theme.surface
   225→  const text = dark ? hexLighten(theme.text, 0.88) : theme.text
   226→  const muted = dark ? hexLighten(theme.text, 0.6) : theme.mutedText
   227→  const border = dark ? hexLighten(bg, 0.16) : theme.border
   228→  const secondary = dark ? hexDarken(theme.secondary, 0.78) : theme.secondary
   229→  const primary = dark ? hexLighten(theme.primary, 0.18) : theme.primary
   230→  const accent = dark ? hexLighten(theme.accent, 0.12) : theme.accent
   231→  const radius = Math.max(0, Math.min(40, Number(theme.radius) || 0))
   232→  const shadow = SHADOW_CSS[theme.shadow] ?? SHADOW_CSS.md
   233→  const fontName = (theme.fontFamily || 'IBM Plex Sans Arabic').replace(/'/g, '')
   234→  return [
   235→    `:root{`,
   236→    `--pt-primary:${primary};`,
   237→    `--pt-primary-fg:${theme.primaryForeground || '#ffffff'};`,
   238→    `--pt-secondary:${secondary};`,
   239→    `--pt-accent:${accent};`,
   240→    `--pt-bg:${bg};`,
   241→    `--pt-surface:${surface};`,
   242→    `--pt-text:${text};`,
   243→    `--pt-muted:${muted};`,
   244→    `--pt-border:${border};`,
   245→    `--pt-radius:${radius}px;`,
   246→    `--pt-shadow:${shadow};`,
   247→    `--pt-font:'${fontName}','IBM Plex Sans Arabic','Tajawal','Cairo',sans-serif;`,
   248→    `}`,
   249→  ].join('')
   250→}

   729→    form.addEventListener('submit', function (e) {
   730→      e.preventDefault()
   731→      var redirect = form.getAttribute('data-redirect')
   732→      if (redirect) {
   733→        window.location.href = redirect
   734→        return
   735→      }
   736→      var success = form.getAttribute('data-success') || 'تم الإرسال بنجاح'
   737→      var alertEl = document.createElement('div')
   738→      alertEl.className = 'pt-alert pt-alert--success'
   739→      alertEl.setAttribute('role', 'status')
   740→      alertEl.textContent = '✓ ' + success
   741→      form.replaceWith(alertEl)
   742→    })
   743→  })
   744→
   745→  // بحث الجداول
   746→  document.querySelectorAll('[data-table-search]').forEach(function (input) {
   747→    input.addEventListener('input', function () {
   748→      var table = document.getElementById(input.getAttribute('data-table-search'))
   749→      if (!table) return
   750→      var q = input.value.trim().toLowerCase()
   751→      var rows = table.querySelectorAll('tbody tr')
   752→      var visible = 0
   753→      rows.forEach(function (row) {
   754→        var match = !q || row.textContent.toLowerCase().indexOf(q) !== -1
   755→        row.style.display = match ? '' : 'none'
   756→        if (match) visible++
   757→      })
   758→      var counter = document.getElementById(input.getAttribute('data-table-search') + '-count')
   759→      if (counter) counter.textContent = 'عرض ' + visible + ' من ' + rows.length + ' صف'
   760→    })
   761→  })
   762→
   763→  // مفاتيح الإعدادات
   764→  document.querySelectorAll('[data-switch]').forEach(function (btn) {
   765→    btn.addEventListener('click', function () {
   766→      var on = btn.classList.toggle('on')
   767→      btn.setAttribute('aria-checked', on ? 'true' : 'false')

   891→function topRoundedRect(x: number, y: number, w: number, h: number, r: number): string {

   896→function barChartSvg(data: ChartDatum[], gid: string): string {
   897→  const W = 600
   898→  const H = 260
   899→  const padX = 40
   900→  const padTop = 16
   901→  const padBottom = 38
   902→  const rtl = [...data].reverse()
   903→  const max = Math.max(...rtl.map((d) => d.value), 1)
   904→  const areaW = W - padX * 2
   905→  const areaH = H - padTop - padBottom
   906→  const baseline = padTop + areaH
   907→  const slot = areaW / rtl.length
   908→  const barW = Math.min(slot * 0.58, 56)
   909→  const skipLabels = rtl.length > 8
   910→
   911→  let svg = `<svg viewBox="0 0 ${W} ${H}" role="img" aria-label="رسم بياني بالأعمدة">`
   912→  svg += `<defs><linearGradient id="bar-${gid}" x1="0" y1="0" x2="0" y2="1"><stop offset="0%" style="stop-color:var(--pt-primary);stop-opacity:1"/><stop offset="100%" style="stop-color:var(--pt-primary);stop-opacity:.62"/></linearGradient></defs>`
   913→
   914→  for (const f of [0, 0.25, 0.5, 0.75, 1]) {
   915→    const y = baseline - areaH * f
   916→    svg += `<line x1="${padX}" y1="${y}" x2="${W - padX}" y2="${y}" style="stroke:var(--pt-border);stroke-width:1"${f === 0 ? '' : ' stroke-dasharray="4 6"'}/>`
   917→    svg += `<text x="${W - padX + 8}" y="${y + 3.5}" text-anchor="start" style="fill:var(--pt-muted);font-size:10px">${esc(compactNum(max * f))}</text>`
   918→  }
   919→  rtl.forEach((d, i) => {
   920→    const h = Math.max(2, (d.value / max) * areaH)
   921→    const cx = padX + slot * i + slot / 2
   922→    const x = cx - barW / 2
   923→    const y = baseline - h
   924→    svg += `<g><title>${esc(d.label)}: ${esc(d.value.toLocaleString('en-US'))}</title><path d="${topRoundedRect(x, y, barW, h, 6)}" style="fill:url(#bar-${gid})"/></g>`
   925→  })
   926→  rtl.forEach((d, i) => {
   927→    if (skipLabels && i % 2 === 1) return
   928→    const cx = padX + slot * i + slot / 2
   929→    svg += `<text x="${cx}" y="${H - 12}" text-anchor="middle" style="fill:var(--pt-muted);font-size:10.5px">${esc(d.label)}</text>`
   930→  })
   931→  svg += '</svg>'
   932→  return svg
   933→}

   935→function smoothPath(pts: Array<{ x: number; y: number }>): string {
   936→  if (pts.length === 0) return ''
   937→  if (pts.length === 1) return `M${pts[0].x},${pts[0].y}`
   938→  let d = `M${pts[0].x},${pts[0].y}`
   939→  for (let i = 0; i < pts.length - 1; i++) {
   940→    const p0 = pts[i - 1] ?? pts[i]
   941→    const p1 = pts[i]
   942→    const p2 = pts[i + 1]
   943→    const p3 = pts[i + 2] ?? p2
   944→    const c1x = p1.x + (p2.x - p0.x) / 6
   945→    const c1y = p1.y + (p2.y - p0.y) / 6
   946→    const c2x = p2.x - (p3.x - p1.x) / 6
   947→    const c2y = p2.y - (p3.y - p1.y) / 6
   948→    d += ` C${c1x.toFixed(2)},${c1y.toFixed(2)} ${c2x.toFixed(2)},${c2y.toFixed(2)} ${p2.x.toFixed(2)},${p2.y.toFixed(2)}`
   949→  }
   950→  return d
   951→}

   953→function lineChartSvg(data: ChartDatum[], gid: string): string {
   954→  const W = 600
   955→  const H = 260
   956→  const padX = 40
   957→  const padTop = 16
   958→  const padBottom = 38
   959→  const rtl = [...data].reverse()
   960→  if (rtl.length < 2) return ''
   961→  const values = rtl.map((d) => d.value)
   962→  const max = Math.max(...values)
   963→  const min = Math.min(...values)
   964→  const span = max - min || max || 1
   965→  const areaW = W - padX * 2
   966→  const areaH = H - padTop - padBottom
   967→  const baseline = padTop + areaH
   968→
   969→  const pts = rtl.map((d, i) => ({
   970→    x: padX + (areaW / (rtl.length - 1)) * i,
   971→    y: padTop + areaH - ((d.value - min + span * 0.06) / (span * 1.12)) * areaH,
   972→  }))
   973→  const line = smoothPath(pts)
   974→  const area = `${line} L${pts[pts.length - 1].x},${baseline} L${pts[0].x},${baseline} Z`
   975→
   976→  let svg = `<svg viewBox="0 0 ${W} ${H}" role="img" aria-label="رسم بياني خطي">`
   977→  svg += `<defs><linearGradient id="area-${gid}" x1="0" y1="0" x2="0" y2="1"><stop offset="0%" style="stop-color:var(--pt-primary);stop-opacity:.26"/><stop offset="100%" style="stop-color:var(--pt-primary);stop-opacity:.02"/></linearGradient></defs>`
   978→  for (const f of [0, 0.25, 0.5, 0.75, 1]) {
   979→    const y = baseline - areaH * f
   980→    svg += `<line x1="${padX}" y1="${y}" x2="${W - padX}" y2="${y}" style="stroke:var(--pt-border);stroke-width:1"${f === 0 ? '' : ' stroke-dasharray="4 6"'}/>`
   981→    svg += `<text x="${W - padX + 8}" y="${y + 3.5}" text-anchor="start" style="fill:var(--pt-muted);font-size:10px">${esc(compactNum(min + span * f))}</text>`
   982→  }
   983→  svg += `<path d="${area}" style="fill:url(#area-${gid})"/>`
   984→  svg += `<path d="${line}" style="fill:none;stroke:var(--pt-primary);stroke-width:2.5;stroke-linecap:round"/>`
   985→  pts.forEach((p, i) => {
   986→    svg += `<g><title>${esc(rtl[i].label)}: ${esc(rtl[i].value.toLocaleString('en-US'))}</title><circle cx="${p.x.toFixed(2)}" cy="${p.y.toFixed(2)}" r="3.5" style="fill:var(--pt-primary);stroke:var(--pt-surface);stroke-width:2"/></g>`
   987→  })
   988→  const skipLabels = rtl.length > 8
   989→  rtl.forEach((d, i) => {
   990→    if (skipLabels && i % 2 === 1) return
   991→    const x = padX + (areaW / (rtl.length - 1)) * i
   992→    svg += `<text x="${x.toFixed(2)}" y="${H - 12}" text-anchor="middle" style="fill:var(--pt-muted);font-size:10.5px">${esc(d.label)}</text>`
   993→  })
   994→  svg += '</svg>'
   995→  return svg
   996→}

   998→function donutChartSvg(data: ChartDatum[]): string {
   999→  const raw = data.filter((d) => d.value > 0)
  1000→  if (raw.length === 0) return ''
  1001→  const rows =
  1002→    raw.length > 6
  1003→      ? [...raw.slice(0, 5), { label: 'أخرى', value: raw.slice(5).reduce((a, d) => a + d.value, 0) }]
  1004→      : raw
  1005→  const total = rows.reduce((a, d) => a + d.value, 0)
  1006→  const R = 60
  1007→  const C = 2 * Math.PI * R
  1008→
  1009→  let cum = 0
  1010→  const segments = rows.map((d) => {
  1011→    const frac = d.value / total
  1012→    const seg = { label: d.label, value: d.value, frac, start: cum }
  1013→    cum += frac
  1014→    return seg
  1015→  })
  1016→
  1017→  const compactTotal = total >= 1000 ? total.toLocaleString('en-US') : String(total)
  1018→
  1019→  let svg = `<div class="blk-donut-wrap"><div class="blk-donut"><svg viewBox="0 0 160 160" style="transform:rotate(-90deg)" role="img" aria-label="رسم بياني دائري">`
  1020→  svg += `<circle cx="80" cy="80" r="${R}" style="fill:none;stroke:var(--pt-border);stroke-width:20"/>`
  1021→  segments.forEach((seg, i) => {
  1022→    const len = seg.frac * C
  1023→    const offset = -seg.start * C
  1024→    const color = CHART_PALETTE[i % CHART_PALETTE.length]
  1025→    svg += `<g><title>${esc(seg.label)}: ${esc(seg.value.toLocaleString('en-US'))} (${Math.round(seg.frac * 100)}%)</title><circle cx="80" cy="80" r="${R}" style="fill:none;stroke:${color};stroke-width:20;stroke-dasharray:${len} ${C - len};stroke-dashoffset:${offset}"/></g>`
  1026→  })
  1027→  svg += `</svg><div class="blk-donut__center"><span class="blk-donut__total">${esc(compactTotal)}</span><span class="blk-donut__total-label">الإجمالي</span></div></div>`
  1028→  svg += `<ul class="blk-donut__legend">`
  1029→  segments.forEach((seg, i) => {
  1030→    const color = CHART_PALETTE[i % CHART_PALETTE.length]
  1031→    svg += `<li><span class="dot" style="background:${color}"></span><span>${esc(seg.label)}</span><span class="pct">${Math.round(seg.frac * 100)}%</span></li>`

  1089→function applyReveal(html: string, delayMs: number): string {
  1090→  const gt = html.indexOf('>')
  1091→  if (gt < 0) return html
  1092→  const head = html.slice(0, gt)
  1093→  const rest = html.slice(gt)
  1094→  const classIdx = head.indexOf('class="')
  1095→  const withClass =
  1096→    classIdx >= 0
  1097→      ? `${head.slice(0, classIdx)}class="pt-reveal ${head.slice(classIdx + 7)}`
  1098→      : `${head} class="pt-reveal"`
  1099→  const styleAttr = delayMs > 0 ? ` style="transition-delay:${delayMs}ms"` : ''
  1100→  return `${withClass}${styleAttr}${rest}`
  1101→}
