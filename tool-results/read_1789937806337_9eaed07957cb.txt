    41→function esc(value: unknown): string {
    42→  return String(value ?? '')
    43→    .replace(/&/g, '&amp;')

    51→function fmtNum(v: number): string {
    52→  if (!Number.isFinite(v)) return '0'
    53→  try {
    54→    const s = v.toLocaleString('ar-DZ')
    55→    return /^[\d\s.,'’]+$/.test(s) ? s : v.toLocaleString('en-US')
    56→  } catch {
    57→    return v.toLocaleString('en-US')
    58→  }
    59→}

    61→/** اختصار رقمي للمحاور (3.4م / 12ألف) — مطابق لعارض React */
    62→function compactNum(v: number): string {
    63→  if (Math.abs(v) >= 1_000_000) {
    64→    const m = v / 1_000_000
    65→    return `${Number.isInteger(m) ? m : m.toFixed(1)}م`
    66→  }
    67→  if (Math.abs(v) >= 1000) {
    68→    const k = v / 1000
    69→    return `${Number.isInteger(k) ? k : k.toFixed(1)}ألف`
    70→  }
    71→  return String(v)
    72→}

    74→/** قصّ النص لحدّ أقصى من المحارف (آمن مع أزواج UTF-16) للوصف والوسوم */
    75→function truncateText(value: string, max: number): string {
    76→  const t = (value ?? '').trim()
    77→  const chars = Array.from(t)
    78→  if (chars.length <= max) return t
    79→  return `${chars.slice(0, Math.max(0, max - 1)).join('')}…`

    93→/** تحويل data URL لصورة إلى { buffer, ext, mime } — null إن لم تكن صالحة */
    94→export function dataUrlToAsset(dataUrl: string): { buffer: Uint8Array; ext: string; mime: string } | null {
    95→  const raw = typeof dataUrl === 'string' ? dataUrl.trim() : ''
    96→  if (!/^data:image\//i.test(raw)) return null
    97→  const comma = raw.indexOf(',')
    98→  if (comma < 0) return null
    99→  const header = raw.slice(5, comma)
   100→  const payload = raw.slice(comma + 1)
   101→  const mime = (header.split(';')[0] ?? '').toLowerCase()
   102→  let ext: string | undefined
   103→  if (mime === 'image/png') ext = 'png'
   104→  else if (mime === 'image/jpeg' || mime === 'image/jpg') ext = 'jpg'
   105→  else if (mime === 'image/webp') ext = 'webp'
   106→  else if (mime === 'image/gif') ext = 'gif'
   107→  else if (mime === 'image/svg+xml') ext = 'svg'
   108→  if (!ext) return null
   109→  const isBase64 = /;base64$/i.test(header)
   110→  let buffer: Uint8Array
   111→  try {
   112→    buffer = isBase64
   113→      ? Buffer.from(payload, 'base64')
   114→      : Buffer.from(decodeURIComponent(payload), 'utf8')
   115→  } catch {
   116→    return null
   117→  }
   118→  if (buffer.byteLength === 0) return null
   119→  return { buffer, ext, mime: LOGO_MIME_BY_EXT[ext] ?? 'image/png' }
   120→}

   122→// قُرّاء خصائص آمنة (مطابقة لعارض React)
   123→function S(props: Record<string, unknown>, key: string, fallback = ''): string {
   124→  const v = props?.[key]
   125→  return typeof v === 'string' ? v : typeof v === 'number' && Number.isFinite(v) ? String(v) : fallback
   126→}
   127→function N(props: Record<string, unknown>, key: string, fallback = 0): number {
   128→  const v = props?.[key]
   129→  if (typeof v === 'number' && Number.isFinite(v)) return v
   130→  if (typeof v === 'string' && v.trim() !== '' && Number.isFinite(Number(v))) return Number(v)
   131→  return fallback
   132→}
   133→function B(props: Record<string, unknown>, key: string, fallback = false): boolean {
   134→  const v = props?.[key]
   135→  if (typeof v === 'boolean') return v
   136→  if (v === 'true') return true
   137→  if (v === 'false') return false
   138→  return fallback
   139→}
   140→function L(props: Record<string, unknown>, key: string): Array<Record<string, unknown>> {
   141→  const v = props?.[key]
   142→  if (!Array.isArray(v)) return []
   143→  return v.filter((it): it is Record<string, unknown> => !!it && typeof it === 'object' && !Array.isArray(it))
   144→}
   145→function iS(item: Record<string, unknown>, key: string, fallback = ''): string {
   146→  return S(item, key, fallback)

   148→function iN(item: Record<string, unknown>, key: string, fallback = 0): number {
   149→  return N(item, key, fallback)

   151→function iB(item: Record<string, unknown>, key: string, fallback = false): boolean {

   163→function hashName(name: string): number {
   164→  let h = 0
   165→  for (let i = 0; i < name.length; i++) {
   166→    h = (h * 31 + name.charCodeAt(i)) >>> 0
   167→  }
   168→  return h

   171→function initialsOf(name: string): string {
   172→  const parts = name.trim().split(/\s+/).filter(Boolean)
   173→  if (parts.length === 0) return '؟'
   174→  if (parts.length === 1) return parts[0].slice(0, 2)
   175→  return (parts[0][0] ?? '') + (parts[1][0] ?? '')
   176→}

   243→function parseHex(hex: string): [number, number, number] | null {
   244→  const m = /^#([0-9a-f]{3}|[0-9a-f]{6})$/i.exec(hex.trim())
   245→  if (!m) return null
   246→  let h = m[1]
   247→  if (h.length === 3) h = h.split('').map((c) => c + c).join('')
   248→  return [parseInt(h.slice(0, 2), 16), parseInt(h.slice(2, 4), 16), parseInt(h.slice(4, 6), 16)]
   249→}
   250→function toHex(r: number, g: number, b: number): string {
   251→  const clamp = (v: number) => Math.max(0, Math.min(255, Math.round(v)))
   252→  return `#${[r, g, b].map((v) => clamp(v).toString(16).padStart(2, '0')).join('')}`
   253→}
   254→function mixHex(a: string, b: string, ratio: number): string {
   255→  const ca = parseHex(a)
   256→  const cb = parseHex(b)
   257→  if (!ca || !cb) return a
   258→  const t = Math.max(0, Math.min(1, ratio))
   259→  return toHex(ca[0] + (cb[0] - ca[0]) * t, ca[1] + (cb[1] - ca[1]) * t, ca[2] + (cb[2] - ca[2]) * t)
   260→}
   261→function hexLighten(hex: string, ratio: number): string {
   262→  return mixHex(hex, '#ffffff', ratio)
   263→}
   264→function hexDarken(hex: string, ratio: number): string {
   265→  return mixHex(hex, '#000000', ratio)
   266→}
