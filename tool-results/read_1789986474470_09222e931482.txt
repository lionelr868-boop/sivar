    12→export type FieldDefinition =
    13→  | {
    14→      key: string
    15→      label: string
    16→      type: 'text' | 'textarea' | 'number' | 'color' | 'switch' | 'select' | 'code'
    17→      options?: { value: string; label: string }[]
    18→    }
    19→  | { key: string; label: string; type: 'page' }
    20→  | { key: string; label: string; type: 'image'; accept?: string }
    21→  | { key: string; label: string; type: 'list'; itemFields: FieldDefinition[]; itemLabel?: string }
    22→
    23→export interface BlockDefinition {
    24→  type: BlockType
    25→  label: string
    26→  category: 'marketing' | 'dashboard' | 'auth' | 'states'

    31→  defaults: () => Record<string, unknown>
    32→  /** حقول التحرير (مجموعة سهلة للاستعمال في المحرر) */
    33→  fields: FieldDefinition[]
    34→}
