    61→async function callLLM(systemPrompt: string, userPrompt: string): Promise<string | null> {
    62→  try {
    63→    const zai = await ZAI.create()
    64→    const completion = await Promise.race([
    65→      zai.chat.completions.create({
    66→        messages: [
    67→          { role: 'assistant', content: systemPrompt },
    68→          { role: 'user', content: userPrompt },
    69→        ],
    70→        thinking: { type: 'disabled' },
    71→      }),
    72→      new Promise<never>((_, reject) => {
    73→        setTimeout(() => reject(new Error('timeout')), LLM_TIMEOUT_MS)
    74→      }),
    75→    ])
    76→    const text = completion.choices[0]?.message?.content
    77→    return typeof text === 'string' && text.trim() ? text : null
    78→  } catch {
    79→    return null
    80→  }
    81→}

    83→function extractJsonObject(text: string): Record<string, unknown> | null {
    84→  let t = text.trim()
    85→  t = t.replace(/^```(?:json)?\s*/i, '').replace(/```\s*$/i, '')
    86→  const start = t.indexOf('{')
    87→  const end = t.lastIndexOf('}')
    88→  if (start === -1 || end === -1 || end <= start) return null
    89→  try {
    90→    const parsed = JSON.parse(t.slice(start, end + 1)) as unknown
    91→    return parsed && typeof parsed === 'object' && !Array.isArray(parsed) ? (parsed as Record<string, unknown>) : null
    92→  } catch {
    93→    return null
    94→  }
    95→}

   186→function resolvePageRef(
   187→  schema: ProjectSchema,
   188→  args: Record<string, unknown>,
   189→  keys: string[]
   190→): string | null {
   191→  for (const key of keys) {
   192→    const v = args[key]
   193→    if (typeof v !== 'string' || !v.trim()) continue
   194→    const ref = v.trim()
   195→    const page = schema.pages.find((p) => p.id === ref) ??
   196→      schema.pages.find((p) => p.name === ref || p.slug === ref) ??
   197→      schema.pages.find((p) => p.name.includes(ref) || ref.includes(p.name))
   198→    if (page) return page.id
   199→  }
   200→  return null
   201→}

   203→function isRecord(v: unknown): v is Record<string, unknown> {
   204→  return !!v && typeof v === 'object' && !Array.isArray(v)
   205→}

   207→function mapArabicColors(patch: Record<string, unknown>): Record<string, unknown> {
   208→  const out: Record<string, unknown> = { ...patch }
   209→  for (const key of THEME_COLOR_KEYS) {
   210→    const v = out[key]
   211→    if (typeof v === 'string') {
   212→      const trimmed = v.trim()
   213→      if (/^#([0-9a-f]{3}|[0-9a-f]{6}|[0-9a-f]{8})$/i.test(trimmed)) {
   214→        out[key] = trimmed.toLowerCase()
   215→      } else if (COLOR_NAME_TO_HEX[trimmed]) {
   216→        out[key] = COLOR_NAME_TO_HEX[trimmed]
   217→      } else {
   218→        delete out[key]
   219→      }
   220→    }
   221→  }
   222→  return out
   223→}

   225→function sanitizeNavItems(
   226→  schema: ProjectSchema,
   227→  raw: unknown
   228→): Array<{ label: string; pageId: string }> | null {
   229→  if (!Array.isArray(raw)) return null
   230→  const items: Array<{ label: string; pageId: string }> = []
   231→  for (const item of raw) {
   232→    if (!isRecord(item)) continue
   233→    const label = typeof item.label === 'string' && item.label.trim() ? item.label.trim().slice(0, 40) : ''
   234→    const pageId =
   235→      (typeof item.pageId === 'string' && schema.pages.find((p) => p.id === item.pageId)?.id) ??
   236→      resolvePageRef(schema, item, ['pageName', 'page', 'name'])
   237→    if (label && pageId) items.push({ label, pageId })
   238→  }
   239→  return items
   240→}

   242→function sanitizeActions(schema: ProjectSchema, rawActions: unknown): AgentAction[] {
   243→  if (!Array.isArray(rawActions)) return []
   244→  const actions: AgentAction[] = []
   245→  for (const item of rawActions.slice(0, MAX_ACTIONS)) {
   246→    if (!isRecord(item)) continue
   247→    const tool = typeof item.tool === 'string' ? item.tool : ''
   248→    if (!TOOL_NAMES.includes(tool)) continue
   249→    const args: Record<string, unknown> = isRecord(item.args) ? { ...item.args } : {}
   250→
   251→    switch (tool as AgentToolName) {
   252→      case 'create_page': {
   253→        const type = typeof args.type === 'string' && PAGE_TYPES.includes(args.type) ? args.type : null
   254→        if (!type) continue
   255→        const name = typeof args.name === 'string' && args.name.trim() ? args.name.trim().slice(0, 40) : ''
   256→        const entity =
   257→          typeof args.entity === 'string' && ENTITY_KEYS.includes(args.entity as (typeof ENTITY_KEYS)[number])
   258→            ? args.entity
   259→            : undefined
   260→        const authVariant =
   261→          typeof args.authVariant === 'string' && ['login', 'signup', 'forgot'].includes(args.authVariant)
   262→            ? (args.authVariant as AuthVariant)
   263→            : undefined
   264→        const clean: Record<string, unknown> = { name, type, entity, authVariant }
   265→        if (
   266→          type === 'landing' &&
   267→          typeof args.landingVariant === 'string' &&
   268→          LANDING_VARIANT_IDS.includes(args.landingVariant)
   269→        ) {
   270→          clean.landingVariant = args.landingVariant
   271→        }
   272→        actions.push({ tool: 'create_page', args: clean })
   273→        break
   274→      }
   275→      case 'delete_page': {
   276→        const pageId = resolvePageRef(schema, args, ['pageId', 'pageName', 'page', 'name'])
   277→        if (!pageId) continue
   278→        actions.push({ tool: 'delete_page', args: { pageId } })
   279→        break
   280→      }
   281→      case 'update_page': {
   282→        const pageId = resolvePageRef(schema, args, ['pageId', 'page', 'pageName'])
   283→        if (!pageId) continue
   284→        const clean: Record<string, unknown> = { pageId }
   285→        if (typeof args.name === 'string' && args.name.trim()) clean.name = args.name.trim().slice(0, 40)
   286→        if (typeof args.title === 'string' && args.title.trim()) clean.title = args.title.trim().slice(0, 80)
   287→        if (typeof args.subtitle === 'string') clean.subtitle = args.subtitle.slice(0, 160)
   288→        if (typeof args.entity === 'string' && ENTITY_KEYS.includes(args.entity as (typeof ENTITY_KEYS)[number])) {
   289→          clean.entity = args.entity
   290→        }
   291→        actions.push({ tool: 'update_page', args: clean })
   292→        break
   293→      }
   294→      case 'add_block': {
   295→        const pageId = resolvePageRef(schema, args, ['pageId', 'page', 'pageName'])
   296→        const blockType = typeof args.blockType === 'string' ? args.blockType : (typeof args.type === 'string' ? args.type : '')
   297→        if (!pageId || !BLOCK_TYPES.includes(blockType)) continue
   298→        const clean: Record<string, unknown> = { pageId, blockType }
   299→        if (isRecord(args.props)) clean.props = args.props
   300→        if (typeof args.position === 'number' && Number.isFinite(args.position)) clean.position = Math.round(args.position)
   301→        actions.push({ tool: 'add_block', args: clean })
   302→        break
   303→      }
   304→      case 'update_block':
   305→      case 'update_content': {
   306→        const pageId = resolvePageRef(schema, args, ['pageId', 'page', 'pageName'])
   307→        const blockId = typeof args.blockId === 'string' && args.blockId.trim() ? args.blockId.trim() : ''
   308→        if (!pageId || !blockId) continue
   309→        const patch = isRecord(args.props) ? args.props : isRecord(args.patch) ? args.patch : isRecord(args.content) ? args.content : null
   310→        if (!patch) continue
   311→        actions.push({ tool: tool as AgentToolName, args: { pageId, blockId, props: patch } })
   312→        break
   313→      }
   314→      case 'delete_block': {
   315→        const pageId = resolvePageRef(schema, args, ['pageId', 'page', 'pageName'])
   316→        const blockId = typeof args.blockId === 'string' && args.blockId.trim() ? args.blockId.trim() : ''
   317→        if (!pageId || !blockId) continue
   318→        actions.push({ tool: 'delete_block', args: { pageId, blockId } })
   319→        break
   320→      }
   321→      case 'move_block': {
   322→        const pageId = resolvePageRef(schema, args, ['pageId', 'page', 'pageName'])
   323→        const blockId = typeof args.blockId === 'string' && args.blockId.trim() ? args.blockId.trim() : ''
   324→        if (!pageId || !blockId) continue
   325→        const clean: Record<string, unknown> = { pageId, blockId }
   326→        if (typeof args.position === 'number' && Number.isFinite(args.position)) clean.position = Math.round(args.position)
   327→        else if (typeof args.direction === 'string' && ['up', 'down'].includes(args.direction)) clean.direction = args.direction
   328→        else continue
   329→        actions.push({ tool: 'move_block', args: clean })
   330→        break
   331→      }
   332→      case 'update_theme': {
   333→        const patch = isRecord(args.theme)
   334→          ? args.theme
   335→          : isRecord(args.patch)
   336→            ? args.patch
   337→            : args
   338→        const mapped = mapArabicColors(patch)
   339→        if (typeof mapped.radius !== 'undefined') {
   340→          const r = Number(mapped.radius)
   341→          if (Number.isFinite(r)) mapped.radius = Math.max(0, Math.min(40, Math.round(r)))
   342→          else delete mapped.radius
   343→        }
   344→        if (typeof mapped.shadow === 'string' && !['none', 'sm', 'md', 'lg'].includes(mapped.shadow)) delete mapped.shadow
   345→        if (typeof mapped.fontFamily === 'string' && !['IBM Plex Sans Arabic', 'Tajawal', 'Cairo'].includes(mapped.fontFamily)) delete mapped.fontFamily
   346→        if (typeof mapped.mode === 'string' && !['light', 'dark'].includes(mapped.mode)) delete mapped.mode
   347→        // نتجاهل الإجراء إن لم تبقَ أي قيمة صالحة
   348→        if (Object.keys(mapped).length === 0) continue
   349→        // applyActions يقرأ patch ?? args — نضع القيم تحت مفتاح patch
   350→        actions.push({ tool: 'update_theme', args: { patch: mapped } })
   351→        break
   352→      }
   353→      case 'update_navigation': {
   354→        const clean: Record<string, unknown> = {}
   355→        if (typeof args.brand === 'string' && args.brand.trim()) clean.brand = args.brand.trim().slice(0, 40)
   356→        const items = sanitizeNavItems(schema, args.items)
   357→        if (Array.isArray(items)) clean.items = items
   358→        if (isRecord(args.cta)) {
   359→          const cta = args.cta
   360→          const label = typeof cta.label === 'string' && cta.label.trim() ? cta.label.trim().slice(0, 40) : ''
   361→          const pageId =
   362→            (typeof cta.pageId === 'string' && schema.pages.find((p) => p.id === cta.pageId)?.id) ??
   363→            resolvePageRef(schema, cta, ['pageName', 'page', 'name'])
   364→          if (label && pageId) clean.cta = { label, pageId }
   365→        }
   366→        if (Object.keys(clean).length === 0) continue
   367→        actions.push({ tool: 'update_navigation', args: clean })
   368→        break
   369→      }
   370→      case 'update_dashboard_nav': {
   371→        const items = sanitizeNavItems(schema, args.items)
   372→        if (!items || items.length === 0) continue
   373→        actions.push({
   374→          tool: 'update_dashboard_nav',
   375→          args: {
   376→            items: items.map((it) => ({ label: it.label, pageId: it.pageId, icon: 'file-text' })),
   377→          },
   378→        })
   379→        break
   380→      }
   381→      case 'rename_project': {
   382→        const name = typeof args.name === 'string' && args.name.trim() ? args.name.trim().slice(0, 40) : ''
   383→        if (!name) continue
   384→        actions.push({ tool: 'rename_project', args: { name } })
   385→        break
   386→      }
   387→      case 'generate_mock_data': {
   388→        const entity =
   389→          typeof args.entity === 'string' && ENTITY_KEYS.includes(args.entity as (typeof ENTITY_KEYS)[number])
   390→            ? args.entity
   391→            : ''
   392→        actions.push({ tool: 'generate_mock_data', args: entity ? { entity } : {} })
   393→        break
   394→      }
   395→      case 'change_landing_variant': {
   396→        const variant =
   397→          typeof args.variant === 'string' && LANDING_VARIANT_IDS.includes(args.variant)
   398→            ? args.variant
   399→            : typeof args.landingVariant === 'string' && LANDING_VARIANT_IDS.includes(args.landingVariant)
   400→              ? args.landingVariant
   401→              : ''
   402→        if (!variant) continue
   403→        const pageId = resolvePageRef(schema, args, ['pageId', 'page', 'pageName', 'name'])
   404→        const clean: Record<string, unknown> = { variant }
   405→        if (pageId) clean.pageId = pageId
   406→        actions.push({ tool: 'change_landing_variant', args: clean })
   407→        break
   408→      }
   409→      default:
   410→        break
   411→    }
   412→  }
   413→  return actions.slice(0, MAX_ACTIONS)
   414→}

   426→function pageIntentFromText(text: string): PageIntent | null {
   427→  if (/تسعير|أسعار|الأسعار|باقات/.test(text)) return { type: 'pricing', name: 'الأسعار' }
   428→  if (/تسجيل\s*الدخول|دخول/.test(text)) return { type: 'auth', name: 'تسجيل الدخول', authVariant: 'login' }
   429→  if (/حساب\s*جديد|إنشاء\s*حساب|اشتراك/.test(text)) return { type: 'auth', name: 'إنشاء حساب', authVariant: 'signup' }
   430→  if (/فواتير/.test(text)) return { type: 'table', name: 'الفواتير', entity: 'invoices' }
   431→  if (/عملاء|زبائن|زبناء/.test(text)) return { type: 'table', name: 'العملاء', entity: 'customers' }
   432→  if (/طلبات/.test(text)) return { type: 'table', name: 'الطلبات', entity: 'orders' }
   433→  if (/منتجات/.test(text)) return { type: 'table', name: 'المنتجات', entity: 'products' }
   434→  if (/حجوزات/.test(text)) return { type: 'table', name: 'الحجوزات', entity: 'bookings' }
   435→  if (/إعدادات|اعدادات/.test(text)) return { type: 'settings', name: 'الإعدادات' }
   436→  if (/ملف\s*شخصي|بروفايل|بروفيل/.test(text)) return { type: 'profile', name: 'الملف الشخصي' }
   437→  if (/أسئلة|اسئلة|faq/.test(text)) return { type: 'faq', name: 'الأسئلة الشائعة' }
   438→  // صفحة مخصصة: نستخرج الاسم من نهاية الطلب
   439→  const rest = text
   440→    .replace(/أضف|اضف|أنشئ|انشئ|جديد|جديدة|صفحة|صفحةً|لي|لى|لـ|لل|ال|من|في|بـ|بعنوان/g, ' ')
   441→    .replace(/\s+/g, ' ')
   442→    .trim()
   443→  if (rest.length >= 2) {
   444→    return { type: 'custom', name: rest.slice(0, 40) }
   445→  }
   446→  return { type: 'custom', name: 'صفحة جديدة' }
   447→}

   449→function extractAfterPageKeyword(message: string): string {
   450→  const idx = message.indexOf('صفحة')
   451→  if (idx === -1) return ''
   452→  return message
   453→    .slice(idx + 'صفحة'.length)
   454→    .replace(/^(?:\s|ل|ال|لل|لـ)+/, '')
   455→    .replace(/^(?:تسعير|أسعار|الأسعار|باقات|دخول|تسجيل الدخول|حساب جديد|إنشاء حساب|فواتير|عملاء|زبائن|طلبات|منتجات|حجوزات|إعدادات|ملف شخصي|أسئلة|شخصية)\s*/, '')
   456→    .trim()
   457→}

   459→function findPageByIntent(schema: ProjectSchema, text: string): ProjectPage | null {
   460→  const term = extractAfterPageKeyword(text)
   461→  if (term && term.length >= 2) {
   462→    const byName = schema.pages.find(
   463→      (p) => p.name.includes(term) || term.includes(p.name) || p.slug.includes(term)
   464→    )
   465→    if (byName) return byName
   466→  }
   467→  const intent = pageIntentFromText(text)
   468→  if (intent) {
   469→    const byType = schema.pages.find((p) => p.type === intent.type && (p.entity ?? '') === (intent.entity ?? ''))
   470→    if (byType) return byType
   471→  }
   472→  return null
   473→}

   512→function heuristicRun(schema: ProjectSchema, message: string): AgentRunResult {
   513→  const text = message.trim()
   514→
   515→  // ---------- حذف صفحة ----------
   516→  if (/احذف|امسح|حذف/.test(text) && /صفحة/.test(text)) {
   517→    const page = findPageByIntent(schema, text)
   518→    if (page) {
   519→      return {
   520→        explanation: `حذفت صفحة «${page.name}» بناءً على طلبك، ونظّفت قوائم التنقل المرتبطة بها.`,
   521→        actions: [{ tool: 'delete_page', args: { pageId: page.id } }],
   522→        fallback: true,
   523→      }
   524→    }
   525→    return {
   526→      explanation: 'لم أجد الصفحة المطلوب حذفها — تأكد من كتابة اسمها كما يظهر في قائمة الصفحات.',
   527→      actions: [],
   528→      fallback: true,
   529→    }
   530→  }
   531→
   532→  // ---------- إضافة صفحة ----------
   533→  if (/أضف|اضف|أنشئ|انشئ|أريد|اريد/.test(text) && /صفحة/.test(text)) {
   534→    const intent = pageIntentFromText(text)
   535→    if (intent) {
   536→      const label = intent.type === 'pricing' ? 'التسعير' : intent.name
   537→      return {
   538→        explanation: `أضفت صفحة «${label}» مع محتوى جاهز قابل للتعديل.`,
   539→        actions: [
   540→          {
   541→            tool: 'create_page',
   542→            args: { name: intent.name, type: intent.type, entity: intent.entity, authVariant: intent.authVariant },
   543→          },
   544→        ],
   545→        fallback: true,
   546→      }
   547→    }
   548→  }
   549→
   550→  // ---------- إضافة صفحة هبوط بنمط محدد ----------
   551→  if (/أضف|اضف|أنشئ|انشئ|أريد|اريد/.test(text) && /صفحة\s*هبوط|هبوط\s*بنمط/.test(text)) {
   552→    const variant = detectLandingVariant(text)
   553→    const variantName = LANDING_VARIANTS.find((v) => v.id === variant)?.name ?? variant
   554→    return {
   555→      explanation: `أنشأت صفحة هبوط جديدة بنمط «${variantName}» — يمكنك تغيير النمط لاحقاً من لوحة الصفحات.`,
   556→      actions: [
   557→        {
   558→          tool: 'create_page',
   559→          args: { name: 'صفحة هبوط جديدة', type: 'landing', landingVariant: variant },
   560→        },
   561→      ],
   562→      fallback: true,
   563→    }
   564→  }
   565→
   566→  // ---------- تغيير نمط صفحة الهبوط ----------
   567→  if (
   568→    /(بدّل|بدل|غيّر|غير|حَوّل|حوّل|اجعل|صمّم|صمم)/.test(text) &&
   569→    /(الصفحة\s*الرئيسية|صفحة\s*الهبوط|الهبوط|الواجهة|النمط)/.test(text) &&
   570→    !/(لون|اللون|ألوان|الوان|ثيم)/.test(text)
   571→  ) {
   572→    const landing = schema.pages.find((p) => p.type === 'landing')
   573→    if (landing) {
   574→      const variant = detectLandingVariant(text)
   575→      const variantName = LANDING_VARIANTS.find((v) => v.id === variant)?.name ?? variant
   576→      return {
   577→        explanation: `أعدت بناء صفحة «${landing.name}» بنمط «${variantName}» — محتوى النصوص قابل للتعديل من لوحة الخصائص.`,
   578→        actions: [{ tool: 'change_landing_variant', args: { variant } }],
   579→        fallback: true,
   580→      }
   581→    }
   582→    return {
   583→      explanation: 'لا توجد صفحة هبوط في المشروع لتغيير نمطها — أضف أولاً صفحة هبوط جديدة.',
   584→      actions: [],
   585→      fallback: true,
   586→    }
   587→  }
   588→
   589→  // ---------- تغيير اللون ----------
   590→  if (/غيّر|غير|تغيير|بدّل|بدل|اجعل/.test(text) && /لون|اللون|ألوان|الوان|هوية|ثيم/.test(text)) {
   591→    const hex = detectColor(text) ?? COLOR_NAME_TO_HEX['أزرق']
   592→    return {
   593→      explanation: `غيّرت اللون الأساسي للهوية البصرية${detectColor(text) ? '' : ' (أزرق افتراضياً)'} — يمكنك معاينة النتيجة فوراً.`,
   594→      actions: [{ tool: 'update_theme', args: { patch: { primary: hex } } }],
   595→      fallback: true,
   596→    }
   597→  }
   598→
   599→  // ---------- إضافة مكوّنات ----------
   600→  if (/أضف|اضف|أنشئ|انشئ|اريد|أريد/.test(text)) {
   601→    const target = firstDashboardPage(schema)
   602→    if (target && /جدول|جداول/.test(text)) {
   603→      const entity = defaultEntityFor(schema.metadata.type)
   604→      return {
   605→        explanation: `أضفت مكوّن «جدول البيانات» إلى صفحة «${target.name}» ببيانات تجريبية جزائرية.`,
   606→        actions: [{ tool: 'add_block', args: { pageId: target.id, blockType: 'data-table', props: { entity, pageSize: 5 } } }],
   607→        fallback: true,
   608→      }
   609→    }
   610→    if (target && /faq|أسئلة|اسئلة|الأسئلة|سؤال|أسوال/.test(text)) {
   611→      return {
   612→        explanation: `أضفت قسم «الأسئلة الشائعة» إلى صفحة «${target.name}».`,
   613→        actions: [{ tool: 'add_block', args: { pageId: target.id, blockType: 'faq' } }],
   614→        fallback: true,
   615→      }
   616→    }
   617→    if (target && /إحصائية|احصائية|إحصائيات|احصائيات/.test(text)) {
   618→      return {
   619→        explanation: `أضفت صف بطاقات الإحصائيات إلى صفحة «${target.name}».`,
   620→        actions: [{ tool: 'add_block', args: { pageId: target.id, blockType: 'stats-row' } }],
   621→        fallback: true,
   622→      }
   623→    }
   624→    if (target && /رسم|بياني|إيرادات|ايرادات/.test(text)) {
   625→      return {
   626→        explanation: `أضفت رسماً بيانياً للإيرادات الشهرية إلى صفحة «${target.name}».`,
   627→        actions: [
   628→          {
   629→            tool: 'add_block',
   630→            args: { pageId: target.id, blockType: 'chart', props: { title: 'الإيرادات الشهرية', chartType: 'bar', dataKey: 'revenueMonthly' } },
   631→          },
   632→        ],
   633→        fallback: true,
   634→      }
   635→    }
   636→    if (target && /نشاطات|النشاطات|سجل|سجل النشاط/.test(text)) {
   637→      return {
   638→        explanation: `أضفت مكوّن «سجل النشاطات» إلى صفحة «${target.name}» بأحداث تجريبية جزائرية.`,
   639→        actions: [{ tool: 'add_block', args: { pageId: target.id, blockType: 'activity-feed' } }],
   640→        fallback: true,
   641→      }
   642→    }
   643→    if (target && /مهام|المهام|قائمة مهام|checklist/.test(text)) {
   644→      return {
   645→        explanation: `أضفت مكوّن «المهام» إلى صفحة «${target.name}» مع شريط تقدّم تفاعلي.`,
   646→        actions: [{ tool: 'add_block', args: { pageId: target.id, blockType: 'tasks-list' } }],
   647→        fallback: true,
   648→      }
   649→    }
   650→    if (target && /إجراءات سريعة|إجراء سريع|اختصارات/.test(text)) {
   651→      return {
   652→        explanation: `أضفت مكوّن «إجراءات سريعة» إلى صفحة «${target.name}» — يمكن ربط كل إجراء بصفحة من لوحة الخصائص.`,
   653→        actions: [{ tool: 'add_block', args: { pageId: target.id, blockType: 'quick-actions' } }],
   654→        fallback: true,
   655→      }
   656→    }
   657→    const landing = firstLandingPage(schema)
   658→    if (landing && /قسم|هيرو|hero|ترحيب/.test(text)) {
   659→      return {
   660→        explanation: `أضفت قسم «البطل» إلى صفحة «${landing.name}».`,
   661→        actions: [{ tool: 'add_block', args: { pageId: landing.id, blockType: 'hero' } }],
   662→        fallback: true,
   663→      }
   664→    }
   665→  }
   666→
   667→  // ---------- تصميم احترافي ----------
   668→  if (/احترافي|احترافية|أجمل|اجمل|أفضل|افضل/.test(text) && /تصميم|شكل|مظهر|اللون|الهوية/.test(text)) {
   669→    return {
   670→      explanation: 'رفعت مستوى التصميم: حواف أنعم (radius 12) وظلال أوضح لتبدو الواجهة أكثر احترافية.',
   671→      actions: [{ tool: 'update_theme', args: { patch: { radius: 12, shadow: 'md' } } }],
   672→      fallback: true,
   673→    }
   674→  }
   675→
   676→  // ---------- لم يفهم ----------
   677→  return {
   678→    explanation:
   679→      'لم أفهم الطلب تماماً — جرّب صياغة مثل: «أضف صفحة للتسعير» أو «غيّر اللون الأساسي إلى الأخضر»',
   680→    actions: [],
   681→    fallback: true,
   682→  }
   683→}

   685→// ============================================================
   686→// نقطة الدخول — runAgent
   687→// ============================================================
   688→export async function runAgent(
   689→  schema: ProjectSchema,
   690→  message: string,
   691→  history: AgentHistoryMessage[]
   692→): Promise<AgentRunResult> {
   693→  const trimmed = message.trim()
   694→  if (!trimmed) {
   695→    return {
   696→      explanation: 'اكتب طلبك وسأنفّذه فوراً — مثل: «أضف صفحة للتسعير».',
   697→      actions: [],
   698→      fallback: true,
   699→    }
   700→  }
   701→
   702→  const historyText = (history ?? [])
   703→    .slice(-8)
   704→    .map((m) => `${m.role === 'user' ? 'المستخدم' : 'المساعد'}: ${String(m.text ?? '').slice(0, 200)}`)
   705→    .join('\n')
   706→
   707→  const userPrompt = [
   708→    `طلب المستخدم: ${trimmed}`,
   709→    historyText ? `\nسياق المحادثة السابقة:\n${historyText}` : '',
   710→    `\nسياق المشروع الحالي (JSON):\n${buildContext(schema)}`,
   711→  ]
   712→    .filter(Boolean)
   713→    .join('\n')
   714→
   715→  const text = await callLLM(AGENT_SYSTEM_PROMPT, userPrompt)
   716→  if (text) {
   717→    const parsed = extractJsonObject(text)
   718→    if (parsed) {
   719→      const actions = sanitizeActions(schema, parsed.actions)
   720→      const explanation =
   721→        typeof parsed.explanation === 'string' && parsed.explanation.trim()
   722→          ? parsed.explanation.trim().slice(0, 400)
   723→          : ''
   724→      if (actions.length > 0) {
   725→        return { explanation: explanation || 'نفّذت تعديلاتك على المشروع.', actions, fallback: false }
   726→      }
   727→      if (explanation) {
   728→        // الـ LLM فهم الطلب لكنه لم يجد إجراءات (أو رفض) — نعيد شرحه
   729→        return { explanation, actions: [], fallback: false }
   730→      }
   731→    }
   732→  }
   733→
   734→  // فشل الـ LLM أو خرج تالفاً → البديل الاستدلالي
   735→  return heuristicRun(schema, trimmed)
   736→}
