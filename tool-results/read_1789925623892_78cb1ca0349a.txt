     8→import { useEffect, useMemo, useRef, useState } from 'react'
     9→import { AnimatePresence, motion } from 'framer-motion'
    10→import {
    11→  ArrowRight,
    12→  CheckCircle2,
    13→  ChevronDown,
    14→  Circle,
    15→  Loader2,
    16→  Sparkles,
    17→} from 'lucide-react'
    18→import { Badge } from '@/components/ui/badge'
    19→import { Button } from '@/components/ui/button'
    20→import { Card } from '@/components/ui/card'
    21→import { Checkbox } from '@/components/ui/checkbox'
    22→import {
    23→  Collapsible,
    24→  CollapsibleContent,
    25→  CollapsibleTrigger,
    26→} from '@/components/ui/collapsible'
    27→import { Input } from '@/components/ui/input'
    28→import { Label } from '@/components/ui/label'
    29→import { Textarea } from '@/components/ui/textarea'
    30→import { toast } from 'sonner'
    31→import { useAppStore } from '@/store/app-store'
    32→import * as api from '@/lib/api-client'
    33→import { t } from '@/lib/i18n'
    34→import { PAGE_TYPE_LABELS, PROJECT_STYLE_LABELS, PROJECT_TYPE_LABELS } from '@/lib/types'
    35→import type { IdeaAnalysis, ProjectStyle, ProjectType, SuggestedPage } from '@/lib/types'
    36→
    37→const tNP = t.newProject
    38→
    39→type Step = 'idea' | 'analyzing' | 'review' | 'building' | 'success'
    40→
    41→const MIN_IDEA_LENGTH = 10
    42→
    43→// 8 أنواع شائعة كخيارات سريعة (التلقائي = الذكاء الاصطناعي يحدد)
    44→const QUICK_TYPES: ProjectType[] = [
    45→  'ecommerce',
    46→  'saas',
    47→  'booking',
    48→  'restaurant',
    49→  'education',
    50→  'health',
    51→  'services',
    52→  'fintech',
    53→]
    54→
    55→const QUICK_STYLES: ProjectStyle[] = ['modern', 'minimal', 'professional', 'luxury', 'youthful', 'tech']
    56→
    57→const ANALYZE_STAGES = [tNP.analyzeStage1, tNP.analyzeStage2, tNP.analyzeStage3]
    58→const BUILD_STAGES = [tNP.buildStep1, tNP.buildStep2, tNP.buildStep3, tNP.buildStep4]
    59→
    60→const delay = (ms: number) => new Promise<void>((resolve) => setTimeout(resolve, ms))
    61→
    62→/** حاوية خطوة مع انتقال سلس (دخول من اليسار في RTL) */
    63→function StepShell({ stepKey, children }: { stepKey: string; children: React.ReactNode }) {
    64→  return (
    65→    <AnimatePresence mode="wait" initial={false}>
    66→      <motion.div
    67→        key={stepKey}
    68→        initial={{ opacity: 0, x: -28 }}
    69→        animate={{ opacity: 1, x: 0 }}
    70→        exit={{ opacity: 0, x: 28 }}
    71→        transition={{ duration: 0.28, ease: 'easeOut' }}
    72→      >
    73→        {children}
    74→      </motion.div>
    75→    </AnimatePresence>
    76→  )
    77→}
    78→
    79→/** عنصر قائمة تحقّق (قيد التنفيذ / منجز / معلّق) */
    80→function ChecklistItem({
    81→  label,
    82→  state,
    83→}: {
    84→  label: string
    85→  state: 'pending' | 'active' | 'done'
    86→}) {
    87→  return (
    88→    <li
    89→      className={`flex min-h-11 items-center gap-3 rounded-lg px-3 transition-colors duration-300 ${
    90→        state === 'done' ? 'bg-emerald-500/10' : state === 'active' ? 'bg-secondary' : ''
    91→      }`}
    92→    >
    93→      {state === 'done' ? (
    94→        <CheckCircle2 className="size-5 shrink-0 text-emerald-600 dark:text-emerald-400" aria-hidden />
    95→      ) : state === 'active' ? (
    96→        <Loader2 className="size-5 shrink-0 animate-spin text-primary" aria-hidden />
    97→      ) : (
    98→        <Circle className="size-5 shrink-0 text-muted-foreground/50" aria-hidden />
    99→      )}
   100→      <span
   101→        className={`text-sm font-medium ${
   102→          state === 'done' ? 'text-foreground' : state === 'active' ? 'text-foreground' : 'text-muted-foreground'
   103→        }`}
   104→      >
   105→        {label}
   106→      </span>
   107→      {state === 'done' ? (
   108→        <span className="ms-auto text-sm font-bold text-emerald-600 dark:text-emerald-400" aria-hidden>
   109→          ✓
   110→        </span>
   111→      ) : null}
   112→    </li>
   113→  )
   114→}
   115→
   116→export function NewProjectFlow() {
   117→  const openProject = useAppStore((s) => s.openProject)
   118→  const setDashboardSection = useAppStore((s) => s.setDashboardSection)
   119→
   120→  // ---------- حالة المسار ----------
   121→  const [step, setStep] = useState<Step>('idea')
   122→  const [idea, setIdea] = useState('')
   123→  const [type, setType] = useState<ProjectType | ''>('')
   124→  const [audience, setAudience] = useState('')
   125→  const [style, setStyle] = useState<ProjectStyle | ''>('')
   126→  const [colorMode, setColorMode] = useState<'auto' | 'custom'>('auto')
   127→  const [colors, setColors] = useState({ primary: '#7c3aed', secondary: '#0d9488', accent: '#d97706' })
   128→  const [optionsOpen, setOptionsOpen] = useState(false)
   129→
   130→  const [analysis, setAnalysis] = useState<IdeaAnalysis | null>(null)
   131→  const [projectName, setProjectName] = useState('')
   132→  const [selectedPages, setSelectedPages] = useState<SuggestedPage[]>([])
   133→  const [editMode, setEditMode] = useState(false)
   134→
   135→  const [analyzeStage, setAnalyzeStage] = useState(0)
   136→  const [buildStage, setBuildStage] = useState(0)
   137→  const [buildError, setBuildError] = useState(false)
   138→
   139→  // مؤقّتات المراحل — تُنظّف عند إلغاء التثبيت
   140→  const analyzeTimer = useRef<ReturnType<typeof setInterval> | null>(null)
   141→  const buildTimer = useRef<ReturnType<typeof setInterval> | null>(null)
   142→  useEffect(
   143→    () => () => {
   144→      if (analyzeTimer.current) clearInterval(analyzeTimer.current)
   145→      if (buildTimer.current) clearInterval(buildTimer.current)
   146→    },
   147→    [],
   148→  )
   149→
   150→  const ideaReady = idea.trim().length >= MIN_IDEA_LENGTH
   151→
   152→  const analyzeStageState = (i: number): 'pending' | 'active' | 'done' =>
   153→    i < analyzeStage ? 'done' : i === analyzeStage ? 'active' : 'pending'
   154→
   155→  // ---------- الخطوة 2: التحليل ----------
   156→  const startAnalysis = async () => {
   157→    if (!ideaReady) return
   158→    setStep('analyzing')
   159→    setAnalyzeStage(0)
   160→    if (analyzeTimer.current) clearInterval(analyzeTimer.current)
   161→    analyzeTimer.current = setInterval(() => {
   162→      setAnalyzeStage((s) => Math.min(s + 1, ANALYZE_STAGES.length - 1))
   163→    }, 900)
   164→
   165→    try {
   166→      const [result] = await Promise.all([api.analyzeIdea(idea.trim()), delay(2400)])
   167→      if (analyzeTimer.current) clearInterval(analyzeTimer.current)
   168→      setAnalysis(result.analysis)
   169→      setProjectName(result.analysis.name)
   170→      setSelectedPages(result.analysis.suggestedPages)
   171→      setStep('review')
   172→    } catch {
   173→      if (analyzeTimer.current) clearInterval(analyzeTimer.current)
   174→      toast.error(tNP.errorAnalyze)
   175→      setStep('idea')
   176→    }
   177→  }
   178→
   179→  // ---------- الخطوة 4: البناء ----------
   180→  const startBuilding = async () => {
   181→    if (!analysis) return
   182→    setStep('building')
   183→    setBuildStage(0)
   184→    setBuildError(false)
   185→    if (buildTimer.current) clearInterval(buildTimer.current)
   186→    buildTimer.current = setInterval(() => {
   187→      setBuildStage((s) => Math.min(s + 1, BUILD_STAGES.length - 1))
   188→    }, 700)
   189→
   190→    const finalAnalysis: IdeaAnalysis = {
   191→      ...analysis,
   192→      name: projectName.trim() || analysis.name,
   193→      audience: audience.trim() || analysis.audience,
   194→      suggestedPages: selectedPages.length > 0 ? selectedPages : analysis.suggestedPages,
   195→    }
   196→
   197→    try {
   198→      const [result] = await Promise.all([
   199→        api.createProject({
   200→          idea: idea.trim(),
   201→          analysis: finalAnalysis,
   202→          name: finalAnalysis.name,
   203→          type: type || undefined,
   204→          style: style || undefined,
   205→          audience: finalAnalysis.audience,
   206→          palette: colorMode === 'custom' ? colors : null,
   207→        }),
   208→        delay(2800),
   209→      ])
   210→      if (buildTimer.current) clearInterval(buildTimer.current)
   211→      setStep('success')
   212→      setTimeout(() => void openProject(result.project.id), 1200)
   213→    } catch {
   214→      if (buildTimer.current) clearInterval(buildTimer.current)
   215→      setBuildError(true)
   216→      toast.error(tNP.errorCreate)
   217→    }
   218→  }
   219→
   220→  const togglePage = (page: SuggestedPage) => {
   221→    setSelectedPages((pages) => {
   222→      const exists = pages.some((p) => p.name === page.name && p.type === page.type)
   223→      return exists
   224→        ? pages.filter((p) => !(p.name === page.name && p.type === page.type))
   225→        : [...pages, page]
   226→    })
   227→  }
   228→
   229→  const stepKey = useMemo(() => `${step}${editMode ? '-edit' : ''}`, [step, editMode])
   230→
   231→  return (
   232→    <div className="mx-auto w-full max-w-2xl space-y-4">
   233→      {/* زر الرجوع الصغير */}
   234→      <Button
   235→        variant="ghost"
   236→        className="h-10 px-3 text-muted-foreground"
   237→        onClick={() => setDashboardSection('projects')}
   238→      >
   239→        <ArrowRight className="size-4" />
   240→        {tNP.backToProjects}
   241→      </Button>
   242→
   243→      <StepShell stepKey={stepKey}>
   244→        {/* ================ الخطوة 1: الفكرة ================ */}
   245→        {step === 'idea' ? (
   246→          <div className="space-y-6">
   247→            <div className="space-y-2 text-center">
   248→              <span className="mx-auto grid size-14 place-items-center rounded-2xl bg-primary/10 text-primary">
   249→                <Sparkles className="size-7" aria-hidden />
   250→              </span>
   251→              <h1 className="font-display text-2xl font-extrabold tracking-tight sm:text-3xl">
   252→                {tNP.ideaTitle}
   253→              </h1>
   254→              <p className="text-sm text-muted-foreground sm:text-base">{tNP.ideaSubtitle}</p>
   255→            </div>
   256→
   257→            <div className="space-y-2.5">
   258→              <Label htmlFor="new-idea" className="sr-only">
   259→                {tNP.ideaTitle}
   260→              </Label>
   261→              <Textarea
   262→                id="new-idea"
   263→                value={idea}
   264→                onChange={(e) => setIdea(e.target.value)}
   265→                placeholder={tNP.ideaPlaceholder}
   266→                className="min-h-40 text-base leading-relaxed sm:text-lg"
   267→              />
   268→              <p
   269→                className={`text-xs ${ideaReady ? 'text-muted-foreground' : 'text-amber-600 dark:text-amber-400'}`}
   270→              >
   271→                {tNP.ideaTooShort}
   272→              </p>
   273→            </div>
   274→
   275→            {/* خيارات إضافية */}
   276→            <Collapsible open={optionsOpen} onOpenChange={setOptionsOpen}>
   277→              <CollapsibleTrigger className="group flex min-h-11 items-center gap-1.5 rounded-lg text-sm font-semibold text-primary transition-colors hover:text-primary/80 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-ring">
   278→                {tNP.moreOptions}
   279→                <span className="text-xs font-normal text-muted-foreground">
   280→                  ({t.common.optional})
   281→                </span>
   282→                <ChevronDown
   283→                  aria-hidden
   284→                  className="size-4 text-muted-foreground transition-transform duration-200 group-data-[state=open]:rotate-180"
   285→                />
   286→              </CollapsibleTrigger>
   287→              <CollapsibleContent>
   288→                <div className="mt-4 space-y-6 rounded-xl border bg-muted/30 p-4 sm:p-5">
   289→                  <p className="text-xs text-muted-foreground">{tNP.moreOptionsHint}</p>
   290→
   291→                  {/* نوع المشروع */}
   292→                  <div className="space-y-2.5">
   293→                    <p className="text-sm font-semibold">{tNP.typeLabel}</p>
   294→                    <div className="flex flex-wrap gap-2">
   295→                      {QUICK_TYPES.map((v) => {
   296→                        const active = type === v
   297→                        return (
   298→                          <button
   299→                            key={v}
   300→                            type="button"
   301→                            aria-pressed={active}
   302→                            onClick={() => setType((cur) => (cur === v ? '' : v))}
   303→                            className={`h-10 min-h-10 rounded-full border px-4 text-xs font-medium transition-colors focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-ring ${
   304→                              active
   305→                                ? 'border-primary bg-primary text-primary-foreground'
   306→                                : 'border-border bg-background text-muted-foreground hover:bg-accent hover:text-foreground'
   307→                            }`}
   308→                          >
   309→                            {PROJECT_TYPE_LABELS[v]}
   310→                          </button>
   311→                        )
   312→                      })}
   313→                    </div>
   314→                  </div>
   315→
   316→                  {/* الجمهور */}
   317→                  <div className="space-y-2.5">
   318→                    <Label htmlFor="new-audience" className="text-sm font-semibold">
   319→                      {tNP.audienceLabel}
   320→                    </Label>
   321→                    <Input
   322→                      id="new-audience"
   323→                      value={audience}
   324→                      onChange={(e) => setAudience(e.target.value)}
   325→                      placeholder={tNP.audiencePlaceholder}
   326→                      className="bg-background"
   327→                    />
   328→                  </div>
   329→
   330→                  {/* النمط */}
   331→                  <div className="space-y-2.5">
   332→                    <p className="text-sm font-semibold">{tNP.styleLabel}</p>
   333→                    <div className="flex flex-wrap gap-2">
   334→                      {QUICK_STYLES.map((v) => {
   335→                        const active = style === v
   336→                        return (
   337→                          <button
   338→                            key={v}
   339→                            type="button"
   340→                            aria-pressed={active}
   341→                            onClick={() => setStyle((cur) => (cur === v ? '' : v))}
   342→                            className={`h-10 min-h-10 rounded-full border px-4 text-xs font-medium transition-colors focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-ring ${
   343→                              active
   344→                                ? 'border-primary bg-primary text-primary-foreground'
   345→                                : 'border-border bg-background text-muted-foreground hover:bg-accent hover:text-foreground'
   346→                            }`}
   347→                          >
   348→                            {PROJECT_STYLE_LABELS[v]}
   349→                          </button>
   350→                        )
   351→                      })}
   352→                    </div>
   353→                  </div>
   354→
   355→                  {/* الألوان */}
   356→                  <div className="space-y-2.5">
   357→                    <p className="text-sm font-semibold">{tNP.colorsLabel}</p>
   358→                    <div className="flex flex-wrap items-center gap-2">
   359→                      <button
   360→                        type="button"
   361→                        aria-pressed={colorMode === 'auto'}
   362→                        onClick={() => setColorMode('auto')}
   363→                        className={`h-10 min-h-10 rounded-full border px-4 text-xs font-medium transition-colors focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-ring ${
   364→                          colorMode === 'auto'
   365→                            ? 'border-primary bg-primary text-primary-foreground'
   366→                            : 'border-border bg-background text-muted-foreground hover:bg-accent hover:text-foreground'
   367→                        }`}
   368→                      >
   369→                        {tNP.colorsAuto}
   370→                      </button>
   371→                      <button
   372→                        type="button"
   373→                        aria-pressed={colorMode === 'custom'}
   374→                        onClick={() => setColorMode('custom')}
   375→                        className={`h-10 min-h-10 rounded-full border px-4 text-xs font-medium transition-colors focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-ring ${
   376→                          colorMode === 'custom'
   377→                            ? 'border-primary bg-primary text-primary-foreground'
   378→                            : 'border-border bg-background text-muted-foreground hover:bg-accent hover:text-foreground'
   379→                        }`}
   380→                      >
   381→                        {tNP.colorsCustom}
   382→                      </button>
   383→                    </div>
   384→                    {colorMode === 'custom' ? (
   385→                      <div className="grid grid-cols-3 gap-3">
   386→                        {(
   387→                          [
   388→                            { key: 'primary', label: tNP.colorPrimary },
   389→                            { key: 'secondary', label: tNP.colorSecondary },
   390→                            { key: 'accent', label: tNP.colorAccent },
   391→                          ] as const
   392→                        ).map((c) => (
   393→                          <div key={c.key} className="space-y-1.5">
   394→                            <Label htmlFor={`color-${c.key}`} className="text-xs font-medium text-muted-foreground">
   395→                              {c.label}
   396→                            </Label>
   397→                            <input
   398→                              id={`color-${c.key}`}
   399→                              type="color"
   400→                              value={colors[c.key]}
   401→                              onChange={(e) => setColors((prev) => ({ ...prev, [c.key]: e.target.value }))}
   402→                              className="h-11 w-full cursor-pointer rounded-lg border bg-background p-1.5"
   403→                              aria-label={c.label}
   404→                            />
   405→                          </div>
   406→                        ))}
   407→                      </div>
   408→                    ) : null}
   409→                  </div>
   410→                </div>
   411→              </CollapsibleContent>
   412→            </Collapsible>
   413→
   414→            <Button
   415→              size="lg"
   416→              className="h-12 w-full text-base"
   417→              disabled={!ideaReady}
   418→              onClick={() => void startAnalysis()}
   419→            >
   420→              {tNP.submitAnalyze}
   421→            </Button>
   422→          </div>
   423→        ) : null}
   424→
   425→        {/* ================ الخطوة 2: التحليل ================ */}
   426→        {step === 'analyzing' ? (
   427→          <div className="space-y-6 py-8">
   428→            <div className="space-y-3 text-center">
   429→              <Loader2
   430→                className="mx-auto size-10 animate-spin text-primary"
   431→                aria-hidden
   432→              />
   433→              <h1 className="font-display text-2xl font-extrabold tracking-tight">
   434→                {tNP.analyzingTitle}
   435→              </h1>
   436→              <p className="text-sm text-muted-foreground">{tNP.analyzingSubtitle}</p>
   437→            </div>
   438→            <Card className="gap-0 p-4 sm:p-6">
   439→              <ul className="space-y-1.5">
   440→                {ANALYZE_STAGES.map((label, i) => (
   441→                  <ChecklistItem key={label} label={label} state={analyzeStageState(i)} />
   442→                ))}
   443→              </ul>
   444→            </Card>
   445→          </div>
   446→        ) : null}
   447→
   448→        {/* ================ الخطوة 3: المراجعة / التعديل ================ */}
   449→        {step === 'review' && analysis ? (
   450→          editMode ? (
   451→            <div className="space-y-6">
   452→              <div className="space-y-1.5">
   453→                <h1 className="font-display text-2xl font-extrabold tracking-tight">
   454→                  {tNP.editTitle}
   455→                </h1>
   456→                <p className="text-sm text-muted-foreground">{tNP.editHint}</p>
   457→              </div>
   458→
   459→              <Card className="gap-4 p-4 sm:p-6">
   460→                <div className="space-y-2.5">
   461→                  <Label htmlFor="project-name">{tNP.projectNameLabel}</Label>
   462→                  <Input
   463→                    id="project-name"
   464→                    value={projectName}
   465→                    onChange={(e) => setProjectName(e.target.value)}
   466→                    placeholder={tNP.projectNamePlaceholder || analysis.name}
   467→                  />
   468→                </div>
   469→
   470→                <div className="space-y-2.5">
   471→                  <div className="flex items-center justify-between">
   472→                    <p className="text-sm font-semibold">{tNP.suggestedPages}</p>
   473→                    <span className="text-xs text-muted-foreground">
   474→                      {selectedPages.length} {tNP.pagesCountSuffix}
   475→                    </span>
   476→                  </div>
   477→                  <div className="max-h-64 space-y-1.5 overflow-y-auto pe-1">
   478→                    {analysis.suggestedPages.map((page) => {
   479→                      const checked = selectedPages.some(
   480→                        (p) => p.name === page.name && p.type === page.type,
   481→                      )
   482→                      return (
   483→                        <label
   484→                          key={`${page.name}-${page.type}`}
   485→                          className={`flex min-h-11 cursor-pointer items-center gap-3 rounded-lg border px-3 transition-colors ${
   486→                            checked ? 'border-primary/40 bg-secondary/60' : 'border-border hover:bg-accent/50'
   487→                          }`}
   488→                        >
   489→                          <Checkbox
   490→                            checked={checked}
   491→                            onCheckedChange={() => togglePage(page)}
   492→                            aria-label={page.name}
   493→                          />
   494→                          <span className="text-sm font-medium">{page.name}</span>
   495→                          <Badge variant="outline" className="ms-auto text-[10px] font-normal text-muted-foreground">
   496→                            {PAGE_TYPE_LABELS[page.type]}
   497→                          </Badge>
   498→                        </label>
   499→                      )
   500→                    })}
   501→                  </div>
   502→                  {selectedPages.length === 0 ? (
   503→                    <p className="text-xs text-amber-600 dark:text-amber-400">{tNP.pagesRequiredHint}</p>
   504→                  ) : null}
   505→                </div>
   506→              </Card>
   507→
   508→              <div className="flex flex-col gap-2.5 sm:flex-row">
   509→                <Button
   510→                  variant="outline"
   511→                  size="lg"
   512→                  className="h-12 flex-1"
   513→                  onClick={() => setEditMode(false)}
   514→                >
   515→                  {tNP.backToReview}
   516→                </Button>
   517→                <Button
   518→                  size="lg"

   526→            </div>
   527→          ) : (

   537→              <Card className="gap-4 p-4 sm:p-6">

   541→                  </span>

   600→        {/* ================ الخطوة 4: البناء ================ */}

   663→      </StepShell>
