   299→   299→/** تحويل أسماء الألوان العربية إلى HEX + التحقق */
   300→   300→function normalizeThemeValue(key: string, value: unknown): { value?: unknown; error?: string } {
   301→   301→  switch (key) {
   302→   302→    case 'radius':
   303→   303→      if (typeof value === 'number' && Number.isFinite(value)) return { value: Math.max(0, Math.min(40, value)) }
   304→   304→      if (typeof value === 'string' && Number.isFinite(Number(value))) return { value: Math.max(0, Math.min(40, Number(value))) }
   305→   305→      return { error: 'قيمة نصف القطر غير صالحة' }
   306→   306→    case 'shadow': {
   307→   307→      const v = String(value)
   308→   308→      if (['none', 'sm', 'md', 'lg'].includes(v)) return { value: v }
   309→   309→      return { error: 'قيمة الظل غير صالحة (none/sm/md/lg)' }
   310→   310→    }
   311→   311→    case 'fontFamily': {
   312→   312→      const v = String(value)
   313→   313→      if (['IBM Plex Sans Arabic', 'Tajawal', 'Cairo'].includes(v)) return { value: v }
   314→   314→      return { error: 'الخط غير مدعوم (IBM Plex Sans Arabic / Tajawal / Cairo)' }
   315→   315→    }
   316→   316→    case 'mode': {
   317→   317→      const v = String(value)
   318→   318→      if (v === 'light' || v === 'dark') return { value: v }
   319→   319→      return { error: 'نمط العرض غير صالح' }
   320→   320→    }
   321→   321→    default: {
   322→   322→      // حقول الألوان
   323→   323→      const v = String(value).trim()
   324→   324→      if (/^#([0-9a-f]{3}|[0-9a-f]{6}|[0-9a-f]{8})$/i.test(v)) return { value: v.toLowerCase() }
   325→   325→      const mapped = COLOR_NAME_TO_HEX[v]
   326→   326→      if (mapped) return { value: mapped }
   327→   327→      return { error: `لون غير صالح: «${v}»` }
   328→   328→    }
   329→   329→  }
   330→   330→}

   425→   425→export function applyActions(schema: ProjectSchema, actions: AgentAction[]): ActionResult {

   428→   428→  const errors: string[] = []

   430→   430→  const pushError = (tool: string, msg: string) => errors.push(`${msg}`)
   431→   431→
   432→   432→  for (const action of actions ?? []) {
   433→   433→    const args = (action?.args ?? {}) as Record<string, unknown>
   434→   434→    try {
   435→   435→      switch (action.tool as AgentToolName) {
   436→   436→        // ---------- pages ----------
   437→   437→        case 'create_page': {
   438→   438→          const type = isPageType(args.type) ? args.type : null
   439→   439→          if (!type) {
   440→   440→            pushError(action.tool, 'نوع الصفحة غير صالح')
   441→   441→            break
   442→   442→          }
   443→   443→          const name = strArg(args, 'name', PAGE_TYPE_LABELS[type])
   444→   444→          const entity = strArg(args, 'entity') || undefined
   445→   445→          const authVariantRaw = strArg(args, 'authVariant')
   446→   446→          const authVariant: AuthVariant | undefined =
   447→   447→            authVariantRaw === 'login' || authVariantRaw === 'signup' || authVariantRaw === 'forgot'
   448→   448→              ? authVariantRaw
   449→   449→              : undefined
   450→   450→          const landingVariantRaw = strArg(args, 'landingVariant')
   451→   451→          const landingVariant = LANDING_VARIANTS.some((v) => v.id === landingVariantRaw) ? landingVariantRaw : undefined
   452→   452→          const { schema: next, pageId } = addPageToSchema(current, { name, type, entity, authVariant, landingVariant })
   453→   453→          current = next
   454→   454→          applied.push(
   455→   455→            landingVariant && type === 'landing'
   456→   456→              ? `تمت إضافة صفحة هبوط «${name}» بنمط «${getLandingVariant(landingVariant).name}»`
   457→   457→              : `تمت إضافة صفحة «${name}»`
   458→   458→          )
   459→   459→          void pageId
   460→   460→          break
   461→   461→        }
   462→   462→
   463→   463→        case 'delete_page': {
   464→   464→          if (current.pages.length <= 1) {
   465→   465→            pushError(action.tool, 'لا يمكن حذف الصفحة الوحيدة في المشروع')
   466→   466→            break
   467→   467→          }
   468→   468→          const page = findPageRef(current, args.pageId ?? args.name ?? args.page)
   469→   469→          if (!page) {
   470→   470→            pushError(action.tool, 'الصفحة غير موجودة')
   471→   471→            break
   472→   472→          }
   473→   473→          current = deletePageFromSchema(current, page.id)
   474→   474→          applied.push(`تم حذف صفحة «${page.name}»`)
   475→   475→          break
   476→   476→        }
   477→   477→
   478→   478→        case 'update_page': {
   479→   479→          const page = findPageRef(current, args.pageId ?? args.page ?? args.name)
   480→   480→          if (!page) {
   481→   481→            pushError(action.tool, 'الصفحة غير موجودة')
   482→   482→            break
   483→   483→          }
   484→   484→          const entity = strArg(args, 'entity')
   485→   485→          if (entity && !current.mockData.entities[entity]) {
   486→   486→            pushError(action.tool, `الكيان «${entity}» غير معروف في البيانات التجريبية`)
   487→   487→            break
   488→   488→          }
   489→   489→          const next = clone(current)
   490→   490→          const target = next.pages.find((p) => p.id === page.id) as ProjectPage
   491→   491→          const name = strArg(args, 'name')
   492→   492→          if (name) {
   493→   493→            target.name = name
   494→   494→            target.slug = slugifyFn(name)
   495→   495→            target.title = name
   496→   496→          }
   497→   497→          const title = strArg(args, 'title')
   498→   498→          if (title) target.title = title
   499→   499→          if (typeof args.subtitle === 'string') target.subtitle = args.subtitle
   500→   500→          if (entity) target.entity = entity
   501→   501→          current = next
   502→   502→          applied.push(`تم تحديث صفحة «${target.name}»`)
   503→   503→          break
   504→   504→        }
   505→   505→
   506→   506→        // ---------- blocks ----------
   507→   507→        case 'add_block': {
   508→   508→          const page = findPageRef(current, args.pageId ?? args.page)
   509→   509→          const blockType = args.blockType ?? args.type
   510→   510→          if (!isBlockType(blockType)) {
   511→   511→            pushError(action.tool, 'نوع المكوّن غير معروف')
   512→   512→            break
   513→   513→          }
   514→   514→          if (!page) {
   515→   515→            pushError(action.tool, 'الصفحة غير موجودة')
   516→   516→            break
   517→   517→          }
   518→   518→          const position = numArg(args, 'position')
   519→   519→          const { schema: next, blockId } = addBlockToPage(current, page.id, blockType, position)
   520→   520→          current = next
   521→   521→          applied.push(`تمت إضافة مكوّن «${blockLabel(blockType)}» إلى صفحة «${page.name}»`)
   522→   522→          void blockId
   523→   523→          break
   524→   524→        }
   525→   525→
   526→   526→        case 'update_block':
   527→   527→        case 'update_content': {
   528→   528→          const page = findPageRef(current, args.pageId ?? args.page)
   529→   529→          if (!page) {
   530→   530→            pushError(action.tool, 'الصفحة غير موجودة')
   531→   531→            break
   532→   532→          }
   533→   533→          const block = findBlock(page, args.blockId ?? args.id)
   534→   534→          if (!block) {
   535→   535→            pushError(action.tool, 'المكوّن غير موجود')
   536→   536→            break
   537→   537→          }
   538→   538→          const patch = (args.patch ?? args.props) as Record<string, unknown> | undefined
   539→   539→          if (!patch || typeof patch !== 'object' || Array.isArray(patch)) {
   540→   540→            pushError(action.tool, 'خصائص التحديث غير صالحة')
   541→   541→            break
   542→   542→          }
   543→   543→          current = updateBlockPropsInSchema(current, page.id, block.id, patch)
   544→   544→          const prefix = action.tool === 'update_content' ? 'محتوى' : 'مكوّن'
   545→   545→          applied.push(`تم تحديث ${prefix} «${blockLabel(block.type)}»`)
   546→   546→          break
   547→   547→        }
   548→   548→
   549→   549→        case 'delete_block': {
   550→   550→          const page = findPageRef(current, args.pageId ?? args.page)
   551→   551→          if (!page) {
   552→   552→            pushError(action.tool, 'الصفحة غير موجودة')
   553→   553→            break
   554→   554→          }
   555→   555→          const block = findBlock(page, args.blockId ?? args.id)
   556→   556→          if (!block) {
   557→   557→            pushError(action.tool, 'المكوّن غير موجود')
   558→   558→            break
   559→   559→          }
   560→   560→          current = deleteBlockFromSchema(current, page.id, block.id)
   561→   561→          applied.push(`تم حذف مكوّن «${blockLabel(block.type)}»`)
   562→   562→          break
   563→   563→        }
   564→   564→
   565→   565→        case 'move_block': {
   566→   566→          const page = findPageRef(current, args.pageId ?? args.page)
   567→   567→          if (!page) {
   568→   568→            pushError(action.tool, 'الصفحة غير موجودة')
   569→   569→            break
   570→   570→          }
   571→   571→          const block = findBlock(page, args.blockId ?? args.id)
   572→   572→          if (!block) {
   573→   573→            pushError(action.tool, 'المكوّن غير موجود')
   574→   574→            break
   575→   575→          }
   576→   576→          const idx = page.blocks.findIndex((b) => b.id === block.id)
   577→   577→          let position = numArg(args, 'position')
   578→   578→          const direction = strArg(args, 'direction')
   579→   579→          if (position === undefined && (direction === 'up' || direction === 'down')) {
   580→   580→            position = direction === 'up' ? idx - 1 : idx + 1
   581→   581→          }
   582→   582→          if (position === undefined || position < 0 || position >= page.blocks.length) {
   583→   583→            pushError(action.tool, 'الموضع الجديد غير صالح')
   584→   584→            break
   585→   585→          }
   586→   586→          current = moveBlockInSchema(current, page.id, block.id, position)
   587→   587→          applied.push(`تم نقل مكوّن «${blockLabel(block.type)}»`)
   588→   588→          break
   589→   589→        }
   590→   590→
   591→   591→        // ---------- landing variant ----------
   592→   592→        case 'change_landing_variant': {
   593→   593→          const variantRaw = strArg(args, 'variant') || strArg(args, 'landingVariant') || strArg(args, 'variantId')
   594→   594→          const variantMeta = LANDING_VARIANTS.find((v) => v.id === variantRaw)
   595→   595→          if (!variantMeta) {
   596→   596→            pushError(action.tool, 'نمط صفحة الهبوط غير معروف')
   597→   597→            break
   598→   598→          }
   599→   599→          const ref = args.pageId ?? args.page ?? args.name
   600→   600→          const page = ref ? findPageRef(current, ref) : current.pages.find((p) => p.type === 'landing')
   601→   601→          if (!page || page.type !== 'landing') {
   602→   602→            pushError(action.tool, 'لم أجد صفحة هبوط لتغيير نمطها')
   603→   603→            break
   604→   604→          }
   605→   605→          current = changeLandingVariantInSchema(current, page.id, variantMeta.id)
   606→   606→          applied.push(`تم تحويل صفحة «${page.name}» إلى نمط «${variantMeta.name}»`)
   607→   607→          break
   608→   608→        }
   609→   609→
   610→   610→        // ---------- theme / navigation ----------
   611→   611→        case 'update_theme': {
   612→   612→          const patch = (args.patch ?? args) as Record<string, unknown>
   613→   613→          const themePatch: Record<string, unknown> = {}
   614→   614→          let hadValid = false
   615→   615→          for (const key of THEME_KEYS) {
   616→   616→            if (patch[key] !== undefined) {
   617→   617→              const { value, error } = normalizeThemeValue(key, patch[key])
   618→   618→              if (error) {
   619→   619→                pushError(action.tool, error)
   620→   620→              } else if (value !== undefined) {
   621→   621→                themePatch[key] = value
   622→   622→                hadValid = true
   623→   623→              }
   624→   624→            }
   625→   625→          }
   626→   626→          if (!hadValid) {
   627→   627→            if (!errors.length) pushError(action.tool, 'لا توجد قيم صالحة لتحديث الهوية')
   628→   628→            break
   629→   629→          }
   630→   630→          current = updateThemeInSchema(current, themePatch as Partial<ProjectTheme>)
   631→   631→          applied.push('تم تحديث هوية الألوان')
   632→   632→          break
   633→   633→        }
   634→   634→
   635→   635→        case 'update_navigation': {
   636→   636→          const pageIds = new Set(current.pages.map((p) => p.id))
   637→   637→          const items = args.items as Array<{ label?: string; pageId?: string }> | undefined
   638→   638→          if (Array.isArray(items)) {
   639→   639→            const bad = items.find((it) => !it || typeof it.pageId !== 'string' || !pageIds.has(it.pageId))
   640→   640→            if (bad) {
   641→   641→              pushError(action.tool, 'عنصر تنقل يشير إلى صفحة غير موجودة')
   642→   642→              break
   643→   643→            }
   644→   644→          }
   645→   645→          const cta = args.cta as { label?: string; pageId?: string } | undefined
   646→   646→          if (cta && (typeof cta.pageId !== 'string' || !pageIds.has(cta.pageId))) {
   647→   647→            pushError(action.tool, 'زر التنقل يشير إلى صفحة غير موجودة')
   648→   648→            break
   649→   649→          }
   650→   650→          current = updateNavigationInSchema(current, {
   651→   651→            brand: typeof args.brand === 'string' ? args.brand : undefined,
   652→   652→            footerNote: typeof args.footerNote === 'string' ? args.footerNote : undefined,
   653→   653→            items: Array.isArray(items) ? (items as ProjectNavigation['items']) : undefined,
   654→   654→            cta: (cta as ProjectNavigation['cta']) ?? undefined,
   655→   655→          })
   656→   656→          applied.push('تم تحديث قائمة التنقل')
   657→   657→          break
   658→   658→        }
   659→   659→
   660→   660→        case 'update_dashboard_nav': {
   661→   661→          const pageIds = new Set(current.pages.map((p) => p.id))
   662→   662→          const items = args.items as Array<{ label?: string; pageId?: string; icon?: string }> | undefined
   663→   663→          if (!Array.isArray(items)) {
   664→   664→            pushError(action.tool, 'قائمة عناصر لوحة التحكم غير صالحة')
   665→   665→            break
   666→   666→          }
   667→   667→          const bad = items.find((it) => !it || typeof it.pageId !== 'string' || !pageIds.has(it.pageId))
   668→   668→          if (bad) {
   669→   669→            pushError(action.tool, 'عنصر لوحة التحكم يشير إلى صفحة غير موجودة')
   670→   670→            break
   671→   671→          }
   672→   672→          current = updateDashboardNavInSchema(current, items as DashboardNavItem[])
   673→   673→          applied.push('تم تحديث قائمة لوحة التحكم')
   674→   674→          break
   675→   675→        }
   676→   676→
   677→   677→        // ---------- project ----------
   678→   678→        case 'rename_project': {
   679→   679→          const name = strArg(args, 'name')
   680→   680→          if (!name) {
   681→   681→            pushError(action.tool, 'اسم المشروع غير صالح')
   682→   682→            break
   683→   683→          }
   684→   684→          current = renameProjectInSchema(current, name)
   685→   685→          applied.push(`تمت إعادة تسمية المشروع إلى «${name}»`)
   686→   686→          break
   687→   687→        }
   688→   688→
   689→   689→        case 'generate_mock_data': {
   690→   690→          const entity = strArg(args, 'entity')
   691→   691→          const seed = Math.max(1, Math.floor(Math.random() * 97))
   692→   692→          if (entity) {
   693→   693→            if (!ENTITY_LABELS[entity]) {
   694→   694→              pushError(action.tool, `الكيان «${entity}» غير معروف`)
   695→   695→              break
   696→   696→            }
   697→   697→            if (!refreshEntityIn(current, entity, seed)) {
   698→   698→              pushError(action.tool, 'تعذّر تحديث بيانات هذا الكيان')
   699→   699→              break
   700→   700→            }
   701→   701→            applied.push(`تم تحديث بيانات «${ENTITY_LABELS[entity]}»`)
   702→   702→          } else {
   703→   703→            let count = 0
   704→   704→            for (const key of Object.keys(current.mockData.entities)) {
   705→   705→              if (refreshEntityIn(current, key, seed + count)) count++
   706→   706→            }
   707→   707→            applied.push(`تم تحديث البيانات التجريبية (${count} كياناً)`)
   708→   708→          }
   709→   709→          break
   710→   710→        }
   711→   711→
   712→   712→        default:
   713→   713→          pushError(String(action?.tool ?? '?'), 'إجراء غير معروف')
   714→   714→          break
   715→   715→      }
   716→   716→    } catch (e) {
   717→   717→      pushError(action?.tool ?? '?', (e as Error).message || 'حدث خطأ أثناء تنفيذ الإجراء')
   718→   718→    }
   719→   719→  }
   720→   720→
   721→   721→  return { schema: current, applied, errors }
   722→   722→}
